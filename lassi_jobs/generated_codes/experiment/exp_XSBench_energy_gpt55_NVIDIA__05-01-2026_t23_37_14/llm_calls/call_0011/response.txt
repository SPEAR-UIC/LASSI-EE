void fancy_int(long a)
{
    if (a < 1000)
    {
        printf("%ld\n", a);
    }
    else if (a < 1000000)
    {
        printf("%ld,%03ld\n", a / 1000, a % 1000);
    }
    else if (a < 1000000000)
    {
        printf("%ld,%03ld,%03ld\n", a / 1000000, (a % 1000000) / 1000, a % 1000);
    }
    else
    {
        printf("%ld,%03ld,%03ld,%03ld\n", a / 1000000000, (a % 1000000000) / 1000000, (a % 1000000) / 1000, a % 1000);
    }
}

int double_compare(const void *a, const void *b)
{
    double A = *((double *)a);
    double B = *((double *)b);
    if (A > B)
    {
        return 1;
    }
    else if (A < B)
    {
        return -1;
    }
    else
    {
        return 0;
    }
}

int NGP_compare(const void *a, const void *b)
{
    NuclideGridPoint A = *((NuclideGridPoint *)a);
    NuclideGridPoint B = *((NuclideGridPoint *)b);
    if (A.energy > B.energy)
    {
        return 1;
    }
    else if (A.energy < B.energy)
    {
        return -1;
    }
    else
    {
        return 0;
    }
}

size_t estimate_mem_usage(Inputs in)
{
    size_t single_nuclide_grid = in.n_gridpoints * sizeof(NuclideGridPoint);
    size_t all_nuclide_grids = in.n_isotopes * single_nuclide_grid;
    size_t size_UEG = in.n_isotopes * in.n_gridpoints * sizeof(double) + in.n_isotopes * in.n_gridpoints * in.n_isotopes * sizeof(int);
    size_t size_hash_grid = in.hash_bins * in.n_isotopes * sizeof(int);
    size_t memtotal;
    if (in.grid_type == UNIONIZED)
    {
        memtotal = all_nuclide_grids + size_UEG;
    }
    else if (in.grid_type == NUCLIDE)
    {
        memtotal = all_nuclide_grids;
    }
    else
    {
        memtotal = all_nuclide_grids + size_hash_grid;
    }
    memtotal = ceil(memtotal / (1024.0 * 1024.0));
    return memtotal;
}

double get_time(void)

Inputs read_CLI(int argc, char *argv[])
{

            user_g = 1;
            input.n_gridpoints = atol(argv[i]);

        else if (strcmp(arg, "-m") == 0 && ++i < argc)
        {
            if (strcmp(argv[i], "history") == 0)
            {
                input.simulation_method = HISTORY_BASED;
            }
            else if (strcmp(argv[i], "event") == 0)
            {
                input.simulation_method = EVENT_BASED;
                if (default_lookups && default_particles)
                {
                    input.lookups = input.lookups * input.particles;
                    input.particles = 0;
                }
            }
            else
            {
                print_CLI_error();
            }
        }
        else if (strcmp(arg, "-l") == 0 && ++i < argc)
        {
            input.lookups = atoi(argv[i]);
            default_lookups = 0;
        }
        else if (strcmp(arg, "-h") == 0 && ++i < argc)
        {
            input.hash_bins = atoi(argv[i]);
        }
        else if (strcmp(arg, "-p") == 0 && ++i < argc)
        {
            input.particles = atoi(argv[i]);
            default_particles = 0;
        }
        else if (strcmp(arg, "-s") == 0 && ++i < argc)
        {
            input.HM = argv[i];
        }
        else if (strcmp(arg, "-G") == 0 && ++i < argc)
        {
            if (strcmp(argv[i], "unionized") == 0)
            {
                input.grid_type = UNIONIZED;
            }
            else if (strcmp(argv[i], "nuclide") == 0)
            {
                input.grid_type = NUCLIDE;
            }
            else if (strcmp(argv[i], "hash") == 0)
            {
                input.grid_type = HASH;
            }
            else
            {
                print_CLI_error();
            }
        }
        else if (strcmp(arg, "-b") == 0 && ++i < argc)
        {
            if (strcmp(argv[i], "read") == 0)
            {
                input.binary_mode = READ;
            }
            else if (strcmp(argv[i], "write") == 0)
            {
                input.binary_mode = WRITE;
            }
            else
            {
                print_CLI_error();
            }
        }
        else if (strcmp(arg, "-k") == 0 && ++i < argc)
        {
            input.kernel_id = atoi(argv[i]);
        }
        else
        {
            print_CLI_error();
        }
    }

    if (input.nthreads < 1 || input.n_isotopes < 1 || input.n_gridpoints < 1 || input.lookups < 1 || input.hash_bins < 1)
    {
        print_CLI_error();
    }
    if (strcasecmp(input.HM, "small") != 0 && strcasecmp(input.HM, "large") != 0 && strcasecmp(input.HM, "XL") != 0 && strcasecmp(input.HM, "XXL") != 0)
    {
        print_CLI_error();
    }

    }
    else if (strcasecmp(input.HM, "XL") == 0 && user_g == 0)

    }
    else if (strcasecmp(input.HM, "XXL") == 0 && user_g == 0)
    {
        input.n_gridpoints = 238847 * 2.1;
    }
    return input;

    }
    unsigned long long small = 0;
    unsigned long long large = 0;
    if (in.simulation_method == EVENT_BASED)
    {
        small = 945990;
        large = 952131;
    }
    else if (in.simulation_method == HISTORY_BASED)
    {
        small = 941535;
        large = 954318;
    }
    if (strcmp(in.HM, "large") == 0)
    {
        if (vhash == large)
        {
            is_invalid_result = 0;
        }
    }
    else if (strcmp(in.HM, "small") == 0)
    {
        if (vhash == small)
        {
            is_invalid_result = 0;
        }
    }
    if (mype == 0)
    {
        if (is_invalid_result)
        {
            printf("Verification checksum: %llu (WARNING - INAVALID CHECKSUM!)\n", vhash);
        }
        else
        {
            printf("Verification checksum: %llu (Valid)\n", vhash);
        }
        border_print();
    }
    return is_invalid_result;
}

double *load_concs(int *num_nucs, int max_num_nucs)
{
    uint64_t seed = STARTING_SEED * STARTING_SEED;
    double *concs = (double *)malloc(12 * max_num_nucs * sizeof(double));
    for (int i = 0; i < 12; i++)
    {
        for (int j = 0; j < num_nucs[i]; j++)
        {
            concs[i * max_num_nucs + j] = LCG_random_double(&seed);
        }
    }
    return concs;
}

__host__ __device__ long grid_search_nuclide(long n, double quarry, NuclideGridPoint *A, long low, long high)
{
    long lowerLimit = low;
    long upperLimit = high;
    long examinationPoint;
    long length = upperLimit - lowerLimit;
    while (length > 1)
    {
        examinationPoint = lowerLimit + (length / 2);
        if (A[examinationPoint].energy > quarry)
        {
            upperLimit = examinationPoint;
        }
        else
        {
            lowerLimit = examinationPoint;
        }
        length = upperLimit - lowerLimit;
    }
    return lowerLimit;
}

SimulationData grid_init_do_not_profile(Inputs in, int mype)
{
    SimulationData SD;
    memset(&SD, 0, sizeof(SimulationData));
    size_t nbytes = 0;
    uint64_t seed = 42;
    if (mype == 0)
    {
        printf("Intializing nuclide grids...\n");
    }
    SD.length_nuclide_grid = in.n_isotopes * in.n_gridpoints;
    SD.nuclide_grid = (NuclideGridPoint *)malloc(SD.length_nuclide_grid * sizeof(NuclideGridPoint));
    assert(SD.nuclide_grid != NULL);
    nbytes += SD.length_nuclide_grid * sizeof(NuclideGridPoint);
    for (int i = 0; i < SD.length_nuclide_grid; i++)
    {
        SD.nuclide_grid[i].energy = LCG_random_double(&seed);
        SD.nuclide_grid[i].total_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].elastic_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].absorbtion_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].fission_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].nu_fission_xs = LCG_random_double(&seed);
    }
    for (int i = 0; i < in.n_isotopes; i++)
    {
        qsort(&SD.nuclide_grid[i * in.n_gridpoints], in.n_gridpoints, sizeof(NuclideGridPoint), NGP_compare);
    }
    if (in.grid_type == NUCLIDE)
    {
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = 0;
    }
    if (in.grid_type == UNIONIZED)
    {
        if (mype == 0)
        {
            printf("Intializing unionized grid...\n");
        }
        SD.length_unionized_energy_array = in.n_isotopes * in.n_gridpoints;
        SD.unionized_energy_array = (double *)malloc(SD.length_unionized_energy_array * sizeof(double));
        assert(SD.unionized_energy_array != NULL);
        nbytes += SD.length_unionized_energy_array * sizeof(double);
        for (int i = 0; i < SD.length_unionized_energy_array; i++)
        {
            SD.unionized_energy_array[i] = SD.nuclide_grid[i].energy;
        }
        qsort(SD.unionized_energy_array, SD.length_unionized_energy_array, sizeof(double), double_compare);
        SD.length_index_grid = SD.length_unionized_energy_array * in.n_isotopes;
        SD.index_grid = (int *)malloc(SD.length_index_grid * sizeof(int));
        assert(SD.index_grid != NULL);
        nbytes += SD.length_index_grid * sizeof(int);
        int *idx_low = (int *)calloc(in.n_isotopes, sizeof(int));
        double *energy_high = (double *)malloc(in.n_isotopes * sizeof(double));
        assert(idx_low != NULL);
        assert(energy_high != NULL);
        for (int i = 0; i < in.n_isotopes; i++)
        {
            energy_high[i] = SD.nuclide_grid[i * in.n_gridpoints + 1].energy;
        }
        for (long e = 0; e < SD.length_unionized_energy_array; e++)
        {
            double unionized_energy = SD.unionized_energy_array[e];
            for (long i = 0; i < in.n_isotopes; i++)
            {
                if (unionized_energy < energy_high[i])
                {
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                }
                else if (idx_low[i] == in.n_gridpoints - 2)
                {
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                }
                else
                {
                    idx_low[i]++;
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                    energy_high[i] = SD.nuclide_grid[i * in.n_gridpoints + idx_low[i] + 1].energy;
                }
            }
        }
        free(idx_low);
        free(energy_high);
    }
    if (in.grid_type == HASH)
    {
        if (mype == 0)
        {
            printf("Intializing hash grid...\n");
        }
        SD.length_index_grid = in.hash_bins * in.n_isotopes;
        SD.index_grid = (int *)malloc(SD.length_index_grid * sizeof(int));
        assert(SD.index_grid != NULL);
        nbytes += SD.length_index_grid * sizeof(int);
        double du = 1.0 / in.hash_bins;
        for (long e = 0; e < in.hash_bins; e++)
        {
            double energy = e * du;
            for (long i = 0; i < in.n_isotopes; i++)
            {
                SD.index_grid[e * in.n_isotopes + i] = grid_search_nuclide(in.n_gridpoints, energy, SD.nuclide_grid + i * in.n_gridpoints, 0, in.n_gridpoints - 1);
            }
        }
    }
    if (mype == 0)
    {
        printf("Intializing material data...\n");
    }
    SD.num_nucs = load_num_nucs(in.n_isotopes);
    SD.length_num_nucs = 12;
    SD.mats = load_mats(SD.num_nucs, in.n_isotopes, &SD.max_num_nucs);
    SD.length_mats = SD.length_num_nucs * SD.max_num_nucs;
    SD.concs = load_concs(SD.num_nucs, SD.max_num_nucs);
    SD.length_concs = SD.length_mats;
    if (mype == 0)
    {
        printf("Intialization complete. Allocated %.0lf MB of data on CPU.\n", nbytes / 1024.0 / 1024.0);
    }
    return SD;
}

void allocate_and_copy_soa_field(double **dst, double *tmp, NuclideGridPoint *src, int len, int field)
{
    for (int i = 0; i < len; i++)
    {
        if (field == 0)
        {
            tmp[i] = src[i].energy;
        }
        else if (field == 1)
        {
            tmp[i] = src[i].total_xs;
        }
        else if (field == 2)
        {
            tmp[i] = src[i].elastic_xs;
        }
        else if (field == 3)
        {
            tmp[i] = src[i].absorbtion_xs;
        }
        else if (field == 4)
        {
            tmp[i] = src[i].fission_xs;
        }
        else
        {
            tmp[i] = src[i].nu_fission_xs;
        }
    }
    gpuErrchk(cudaMalloc((void **)dst, len * sizeof(double)));
    gpuErrchk(cudaMemcpy(*dst, tmp, len * sizeof(double), cudaMemcpyHostToDevice));
}

void build_device_soa(SimulationData SD)
{
    double *tmp = (double *)malloc(SD.length_nuclide_grid * sizeof(double));
    assert(tmp != NULL);
    allocate_and_copy_soa_field(&g_soa.energy, tmp, SD.nuclide_grid, SD.length_nuclide_grid, 0);
    allocate_and_copy_soa_field(&g_soa.total_xs, tmp, SD.nuclide_grid, SD.length_nuclide_grid, 1);
    allocate_and_copy_soa_field(&g_soa.elastic_xs, tmp, SD.nuclide_grid, SD.length_nuclide_grid, 2);
    allocate_and_copy_soa_field(&g_soa.absorbtion_xs, tmp, SD.nuclide_grid, SD.length_nuclide_grid, 3);
    allocate_and_copy_soa_field(&g_soa.fission_xs, tmp, SD.nuclide_grid, SD.length_nuclide_grid, 4);
    allocate_and_copy_soa_field(&g_soa.nu_fission_xs, tmp, SD.nuclide_grid, SD.length_nuclide_grid, 5);
    free(tmp);
}

__device__ __forceinline__ long grid_search(long n, double quarry, const double *__restrict__ A)
{
    long lowerLimit = 0;
    long upperLimit = n - 1;
    long examinationPoint;
    long length = upperLimit - lowerLimit;
    while (length > 1)
    {
        examinationPoint = lowerLimit + (length / 2);
        if (__ldg(&A[examinationPoint]) > quarry)
        {
            upperLimit = examinationPoint;
        }
        else
        {
            lowerLimit = examinationPoint;
        }
        length = upperLimit - lowerLimit;
    }
    return lowerLimit;
}

__device__ __forceinline__ long grid_search_nuclide_device_soa(long n, double quarry, const double *__restrict__ energy, long base, long low, long high)
{
    long lowerLimit = low;
    long upperLimit = high;
    long examinationPoint;
    long length = upperLimit - lowerLimit;
    while (length > 1)
    {
        examinationPoint = lowerLimit + (length / 2);
        if (__ldg(&energy[base + examinationPoint]) > quarry)
        {
            upperLimit = examinationPoint;
        }
        else
        {
            lowerLimit = examinationPoint;
        }
        length = upperLimit - lowerLimit;
    }
    return lowerLimit;
}

template<int GRID_TYPE>
__device__ __forceinline__ void micro_xs_device_t(double p_energy, int nuc, long n_isotopes, long n_gridpoints, const int *__restrict__ index_data, NuclideGridSoA soa, long idx, double *__restrict__ xs, int hash_bins)
{
    long nuc_base = nuc * n_gridpoints;
    long low_idx = 0;
    if constexpr (GRID_TYPE == UNIONIZED)
    {
        int grid_idx = __ldg(&index_data[idx * n_isotopes + nuc]);
        low_idx = nuc_base + grid_idx - (grid_idx == n_gridpoints - 1);
    }
    else if constexpr (GRID_TYPE == NUCLIDE)
    {
        long lower = grid_search_nuclide_device_soa(n_gridpoints, p_energy, soa.energy, nuc_base, 0, n_gridpoints - 1);
        low_idx = nuc_base + lower - (lower == n_gridpoints - 1);
    }
    else
    {
        int u_low = __ldg(&index_data[idx * n_isotopes + nuc]);
        int u_high;
        if (idx == hash_bins - 1)
        {
            u_high = n_gridpoints - 1;
        }
        else
        {
            u_high = __ldg(&index_data[(idx + 1) * n_isotopes + nuc]) + 1;
        }
        double e_low = __ldg(&soa.energy[nuc_base + u_low]);
        double e_high = __ldg(&soa.energy[nuc_base + u_high]);
        int lower;
        if (p_energy <= e_low)
        {
            lower = 0;
        }
        else if (p_energy >= e_high)
        {
            lower = n_gridpoints - 1;
        }
        else
        {
            lower = grid_search_nuclide_device_soa(n_gridpoints, p_energy, soa.energy, nuc_base, u_low, u_high);
        }
        low_idx = nuc_base + lower - (lower == n_gridpoints - 1);
    }

    long high_idx = low_idx + 1;
    double low_energy = __ldg(&soa.energy[low_idx]);
    double high_energy = __ldg(&soa.energy[high_idx]);
    double f = (high_energy - p_energy) / (high_energy - low_energy);

    double high_total = __ldg(&soa.total_xs[high_idx]);
    double low_total = __ldg(&soa.total_xs[low_idx]);
    double high_elastic = __ldg(&soa.elastic_xs[high_idx]);
    double low_elastic = __ldg(&soa.elastic_xs[low_idx]);
    double high_absorbtion = __ldg(&soa.absorbtion_xs[high_idx]);
    double low_absorbtion = __ldg(&soa.absorbtion_xs[low_idx]);
    double high_fission = __ldg(&soa.fission_xs[high_idx]);
    double low_fission = __ldg(&soa.fission_xs[low_idx]);
    double high_nu_fission = __ldg(&soa.nu_fission_xs[high_idx]);
    double low_nu_fission = __ldg(&soa.nu_fission_xs[low_idx]);

    xs[0] = high_total - f * (high_total - low_total);
    xs[1] = high_elastic - f * (high_elastic - low_elastic);
    xs[2] = high_absorbtion - f * (high_absorbtion - low_absorbtion);
    xs[3] = high_fission - f * (high_fission - low_fission);
    xs[4] = high_nu_fission - f * (high_nu_fission - low_nu_fission);
}

template<int GRID_TYPE>
__device__ __forceinline__ unsigned long warp_lookup_value_t(double p_energy, int mat, Inputs in, SimulationData GSD, NuclideGridSoA soa)
{
    const unsigned mask = 0xffffffffu;
    int lane = threadIdx.x & 31;
    long idx = -1;

    if constexpr (GRID_TYPE == UNIONIZED)
    {
        if (lane == 0)
        {
            idx = grid_search(in.n_isotopes * in.n_gridpoints, p_energy, GSD.unionized_energy_array);
        }
        unsigned long long idx_u = __shfl_sync(mask, (unsigned long long)idx, 0);
        idx = (long)idx_u;
    }
    else if constexpr (GRID_TYPE == HASH)
    {
        double du = 1.0 / in.hash_bins;
        idx = p_energy / du;
    }

    double macro[5];
    #pragma unroll
    for (int k = 0; k < 5; k++)
    {
        macro[k] = 0.0;
    }

    int n_material_nucs;
    int stride;
    if (GSD.max_num_nucs <= MAX_MAT_NUCS_CONST)
    {
        n_material_nucs = c_num_nucs[mat];
        stride = c_max_num_nucs;
    }
    else
    {
        n_material_nucs = __ldg(&GSD.num_nucs[mat]);
        stride = GSD.max_num_nucs;
    }

    for (int base = 0; base < n_material_nucs; base += 32)
    {
        int j = base + lane;
        double local[5];
        double conc = 0.0;

        #pragma unroll
        for (int k = 0; k < 5; k++)
        {
            local[k] = 0.0;
        }

        if (j < n_material_nucs)
        {
            int p_nuc;
            if (GSD.max_num_nucs <= MAX_MAT_NUCS_CONST)
            {
                p_nuc = c_mats[mat * stride + j];
                conc = c_concs[mat * stride + j];
            }
            else
            {
                p_nuc = __ldg(&GSD.mats[mat * stride + j]);
                conc = __ldg(&GSD.concs[mat * stride + j]);
            }
            micro_xs_device_t<GRID_TYPE>(p_energy, p_nuc, in.n_isotopes, in.n_gridpoints, GSD.index_grid, soa, idx, local, in.hash_bins);
        }

        #pragma unroll
        for (int src = 0; src < 32; src++)
        {
            int jj = base + src;
            double c = __shfl_sync(mask, conc, src);
            double x0 = __shfl_sync(mask, local[0], src);
            double x1 = __shfl_sync(mask, local[1], src);
            double x2 = __shfl_sync(mask, local[2], src);
            double x3 = __shfl_sync(mask, local[3], src);
            double x4 = __shfl_sync(mask, local[4], src);
            if (lane == 0 && jj < n_material_nucs)
            {
                macro[0] += x0 * c;
                macro[1] += x1 * c;
                macro[2] += x2 * c;
                macro[3] += x3 * c;
                macro[4] += x4 * c;
            }
        }
    }

    unsigned long ret = 0;
    if (lane == 0)
    {
        double max = -1.0;
        int max_idx = 0;
        #pragma unroll
        for (int j = 0; j < 5; j++)
        {
            if (macro[j] > max)
            {
                max = macro[j];
                max_idx = j;
            }
        }
        ret = (unsigned long)(max_idx + 1);
    }
    return ret;
}

template<int GRID_TYPE, int CHUNK>
__global__ __launch_bounds__(256, 2) void xs_lookup_kernel_warp_chunked(Inputs in, SimulationData GSD, NuclideGridSoA soa)
{
    __shared__ unsigned long warp_sums[8];

    int warp = threadIdx.x >> 5;
    int lane = threadIdx.x & 31;
    int warps_per_block = blockDim.x >> 5;
    int warp_lookup_group = blockIdx.x * warps_per_block + warp;
    int lookup_start = warp_lookup_group * CHUNK;

    unsigned long local_sum = 0;

    if (lookup_start < in.lookups)
    {
        uint64_t seed = STARTING_SEED;
        if (lane == 0)
        {
            seed = fast_forward_LCG(seed, 2ULL * (uint64_t)lookup_start);
        }

        #pragma unroll
        for (int c = 0; c < CHUNK; c++)
        {
            int lookup = lookup_start + c;
            double p_energy = 0.0;
            int mat = 0;
            if (lookup < in.lookups)
            {
                if (lane == 0)
                {
                    p_energy = LCG_random_double(&seed);
                    mat = pick_mat(&seed);
                }
                p_energy = __shfl_sync(0xffffffffu, p_energy, 0);
                mat = __shfl_sync(0xffffffffu, mat, 0);
                local_sum += warp_lookup_value_t<GRID_TYPE>(p_energy, mat, in, GSD, soa);
            }
        }
    }

    if (lane == 0)
    {
        warp_sums[warp] = local_sum;
    }
    __syncthreads();

    unsigned long sum = 0;
    if (threadIdx.x < 8)
    {
        sum = warp_sums[threadIdx.x];
    }
    if (threadIdx.x < 32)
    {
        #pragma unroll
        for (int offset = 16; offset > 0; offset >>= 1)
        {
            sum += __shfl_down_sync(0xffffffffu, sum, offset);
        }
    }
    if (threadIdx.x == 0)
    {
        atomicAdd((unsigned long long *)GSD.verification, (unsigned long long)sum);
    }
}

unsigned long long finish_verification_from_device(SimulationData GSD)
{
    unsigned long long verification_scalar = 0;
    gpuErrchk(cudaMemcpy(&verification_scalar, GSD.verification, sizeof(unsigned long long), cudaMemcpyDeviceToHost));
    return verification_scalar;
}

void launch_lookup_kernel(Inputs in, SimulationData GSD, int nblocks, int nthreads)
{
    if (in.grid_type == UNIONIZED)
    {
        xs_lookup_kernel_warp_chunked<UNIONIZED, XS_LOOKUPS_PER_WARP_CHUNK><<<nblocks, nthreads>>>(in, GSD, g_soa);
    }
    else if (in.grid_type == NUCLIDE)
    {
        xs_lookup_kernel_warp_chunked<NUCLIDE, XS_LOOKUPS_PER_WARP_CHUNK><<<nblocks, nthreads>>>(in, GSD, g_soa);
    }
    else
    {
        xs_lookup_kernel_warp_chunked<HASH, XS_LOOKUPS_PER_WARP_CHUNK><<<nblocks, nthreads>>>(in, GSD, g_soa);
    }
}

unsigned long long run_event_based_simulation_baseline(Inputs in, SimulationData GSD, int mype)
{
    if (mype == 0)
    {
        printf("Running baseline event-based simulation...\n");
    }
    int nthreads = XS_BLOCK_SIZE;
    int lookups_per_block = (nthreads / 32) * XS_LOOKUPS_PER_WARP_CHUNK;
    int nblocks = (in.lookups + lookups_per_block - 1) / lookups_per_block;
    gpuErrchk(cudaMemset(GSD.verification, 0, sizeof(unsigned long long)));
    launch_lookup_kernel(in, GSD, nblocks, nthreads);
    gpuErrchk(cudaPeekAtLastError());
    if (mype == 0)
    {
        printf("Reducing verification results...\n");
    }
    return finish_verification_from_device(GSD);
}

unsigned long long run_named_optimized(Inputs in, SimulationData GSD, int mype, const char *name)
{
    if (mype == 0)
    {
        printf("Simulation Kernel:\"%s\"\n", name);
        printf("Allocating additional device data required by kernel...\n");
        size_t total_sz = in.lookups * sizeof(double) + in.lookups * sizeof(int);
        printf("Allocated an additional %.0lf MB of data on GPU.\n", total_sz / 1024.0 / 1024.0);
        printf("Beginning optimized simulation...\n");
    }
    int nthreads = XS_BLOCK_SIZE;
    int lookups_per_block = (nthreads / 32) * XS_LOOKUPS_PER_WARP_CHUNK;
    int nblocks = (in.lookups + lookups_per_block - 1) / lookups_per_block;
    gpuErrchk(cudaMemset(GSD.verification, 0, sizeof(unsigned long long)));
    launch_lookup_kernel(in, GSD, nblocks, nthreads);
    gpuErrchk(cudaPeekAtLastError());
    if (mype == 0)
    {
        printf("Reducing verification results...\n");
    }
    return finish_verification_from_device(GSD);
}

__host__ __device__ double LCG_random_double(uint64_t *seed)
{
    const uint64_t mask = 9223372036854775807ULL;
    const uint64_t a = 2806196910506780709ULL;
    const uint64_t c = 1ULL;
    *seed = (a * (*seed) + c) & mask;
    return (double)(*seed) / 9223372036854775808.0;
}

__device__ uint64_t fast_forward_LCG(uint64_t seed, uint64_t n)
{
    const uint64_t mask = 9223372036854775807ULL;
    uint64_t a = 2806196910506780709ULL;
    uint64_t c = 1ULL;
    n = n & mask;
    uint64_t a_new = 1;
    uint64_t c_new = 0;
    while (n > 0)
    {
        if (n & 1)
        {
            a_new *= a;
            c_new = c_new * a + c;
        }
        c *= (a + 1);
        a *= a;
        n >>= 1;
    }
    return (a_new * seed + c_new) & mask;
}

int main(int argc, char *argv[])
{
    int version = 20;
    int mype = 0;
    int nprocs = 1;
    unsigned long long verification;
    Inputs in = read_CLI(argc, argv);

    gpuErrchk(cudaDeviceSetCacheConfig(cudaFuncCachePreferL1));
    gpuErrchk(cudaFuncSetCacheConfig(xs_lookup_kernel_warp_chunked<UNIONIZED, XS_LOOKUPS_PER_WARP_CHUNK>, cudaFuncCachePreferL1));
    gpuErrchk(cudaFuncSetCacheConfig(xs_lookup_kernel_warp_chunked<NUCLIDE, XS_LOOKUPS_PER_WARP_CHUNK>, cudaFuncCachePreferL1));
    gpuErrchk(cudaFuncSetCacheConfig(xs_lookup_kernel_warp_chunked<HASH, XS_LOOKUPS_PER_WARP_CHUNK>, cudaFuncCachePreferL1));

    if (mype == 0)
    {
        print_inputs(in, nprocs, version);
    }

    SimulationData SD;
    if (in.binary_mode == READ)
    {
        SD = binary_read(in);
    }
    else
    {
        SD = grid_init_do_not_profile(in, mype);
    }

    if (in.binary_mode == WRITE && mype == 0)
    {
        binary_write(in, SD);
    }

    SimulationData GSD = move_simulation_data_to_device(in, mype, SD);

    if (mype == 0)
    {
        printf("\n");
        border_print();
        center_print("SIMULATION", 79);
        border_print();
    }

    double omp_start = get_time();

    if (in.simulation_method == EVENT_BASED)
    {
        if (in.kernel_id == 0)
        {
            verification = run_event_based_simulation_baseline(in, GSD, mype);
        }
        else if (in.kernel_id == 1)
        {
            verification = run_event_based_simulation_optimization_1(in, GSD, mype);
        }
        else if (in.kernel_id == 2)
        {
            verification = run_event_based_simulation_optimization_2(in, GSD, mype);
        }
        else if (in.kernel_id == 3)
        {
            verification = run_event_based_simulation_optimization_3(in, GSD, mype);
        }
        else if (in.kernel_id == 4)
        {
            verification = run_event_based_simulation_optimization_4(in, GSD, mype);
        }
        else if (in.kernel_id == 5)
        {
            verification = run_event_based_simulation_optimization_5(in, GSD, mype);
        }
        else if (in.kernel_id == 6)
        {
            verification = run_event_based_simulation_optimization_6(in, GSD, mype);
        }
        else
        {
            printf("Error: No kernel ID %d found!\n", in.kernel_id);
            exit(1);
        }
    }
    else
    {
        printf("History-based simulation not implemented in CUDA code. Instead,\nuse the event-based method with \"-m event\" argument.\n");
        exit(1);
    }

    if (mype == 0)
    {
        printf("\n");
        printf("Simulation complete.\n");
    }

    double omp_end = get_time();
    release_device_memory(GSD);
    verification = verification % 999983;
    int is_invalid_result = print_results(in, mype, omp_end - omp_start, nprocs, verification);
    return is_invalid_result;
}
