static int parse_positive_int(const char *text, int fallback)
{
    char *end = nullptr;
    long value = std::strtol(text, &end, 10);

    if (end == text || *end != '\0') {
        return fallback;
    }

    if (value < 1 || value > std::numeric_limits<int>::max()) {
        return fallback;
    }

    return static_cast<int>(value);
}

Inputs read_CLI(int argc, char **argv)
{
    Inputs inputs;
    inputs.n = N;
    inputs.iterations = 10000;
    inputs.block_size = 16;
    inputs.help = false;

    for (int i = 1; i < argc; ++i) {
        if (std::strcmp(argv[i], "-h") == 0 || std::strcmp(argv[i], "--help") == 0) {
            inputs.help = true;
        } else if ((std::strcmp(argv[i], "-n") == 0 || std::strcmp(argv[i], "--size") == 0) && i + 1 < argc) {
            ++i;
            inputs.n = parse_positive_int(argv[i], inputs.n);
        } else if ((std::strcmp(argv[i], "-i") == 0 || std::strcmp(argv[i], "--iterations") == 0) && i + 1 < argc) {
            ++i;
            inputs.iterations = parse_positive_int(argv[i], inputs.iterations);
        } else if ((std::strcmp(argv[i], "-b") == 0 || std::strcmp(argv[i], "--block") == 0) && i + 1 < argc) {
            ++i;
            inputs.block_size = parse_positive_int(argv[i], inputs.block_size);
        }
    }

    if (inputs.block_size < 1) {
        inputs.block_size = 16;
    }

    if (inputs.block_size > 32) {
        inputs.block_size = 32;
    }

    return inputs;
}

void initialize_data(float *f)
{
    for (int j = 0; j < N; ++j) {
        for (int i = 0; i < N; ++i) {
            if (i == 0 || i == N - 1) {
                f[IDX(i, j)] = sinf(j * 2 * M_PI / (N - 1));
            } else if (j == 0 || j == N - 1) {
                f[IDX(i, j)] = sinf(i * 2 * M_PI / (N - 1));
            } else {
                f[IDX(i, j)] = 0.0f;
            }
        }
    }
}

void grid_init_do_not_profile(float *f, float *f_old)
{
    initialize_data(f);
    initialize_data(f_old);
}

__global__ void jacobi_step(float *__restrict__ f,
                            const float *__restrict__ f_old,
                            float *__restrict__ error)
{
    __shared__ float f_old_tile[18][18];

    int i = threadIdx.x + blockIdx.x * blockDim.x;
    int j = threadIdx.y + blockIdx.y * blockDim.y;

    f_old_tile[threadIdx.y + 1][threadIdx.x + 1] = f_old[IDX(i, j)];

    if (threadIdx.x == 0 && i >= 1) {
        f_old_tile[threadIdx.y + 1][threadIdx.x + 0] = f_old[IDX(i - 1, j)];
    }
    if (threadIdx.x == 15 && i <= N - 2) {
        f_old_tile[threadIdx.y + 1][threadIdx.x + 2] = f_old[IDX(i + 1, j)];
    }
    if (threadIdx.y == 0 && j >= 1) {
        f_old_tile[threadIdx.y + 0][threadIdx.x + 1] = f_old[IDX(i, j - 1)];
    }
    if (threadIdx.y == 15 && j <= N - 2) {
        f_old_tile[threadIdx.y + 2][threadIdx.x + 1] = f_old[IDX(i, j + 1)];
    }

    __syncthreads();

    float err = 0.0f;

    if (j >= 1 && j <= N - 2) {
        if (i >= 1 && i <= N - 2) {
            f[IDX(i, j)] = 0.25f * (f_old_tile[threadIdx.y + 1][threadIdx.x + 2] +
                                     f_old_tile[threadIdx.y + 1][threadIdx.x + 0] +
                                     f_old_tile[threadIdx.y + 2][threadIdx.x + 1] +
                                     f_old_tile[threadIdx.y + 0][threadIdx.x + 1]);
            float df = f[IDX(i, j)] - f_old_tile[threadIdx.y + 1][threadIdx.x + 1];
            err = df * df;
        }
    }

#pragma unroll
    for (int offset = 8; offset > 0; offset /= 2) {
        err += __shfl_down_sync(0xffffffff, err, offset);
    }

    __shared__ float reduction_array[16];

    if (threadIdx.x == 0) {
        reduction_array[threadIdx.y] = err;
    }

    __syncthreads();

    if (threadIdx.y == 0) {
        err = reduction_array[threadIdx.x];

#pragma unroll
        for (int offset = 8; offset > 0; offset /= 2) {
            err += __shfl_down_sync(0xffffffff, err, offset);
        }

        if (threadIdx.x == 0) {
            atomicAdd(error, err);
        }
    }
}

__global__ void swap_data(const float *__restrict__ f,
                          float *__restrict__ f_old)
{
    int i = threadIdx.x + blockIdx.x * blockDim.x;
    int j = threadIdx.y + blockIdx.y * blockDim.y;

    if (j >= 1 && j <= N - 2) {
        if (i >= 1 && i <= N - 2) {
            f_old[IDX(i, j)] = f[IDX(i, j)];
        }
    }
}

__host__ __device__ __forceinline__ unsigned long long checksum_mix(unsigned long long x)
{
    x ^= x >> 33;
    x *= 0xff51afd7ed558ccdULL;
    x ^= x >> 33;
    x *= 0xc4ceb9fe1a85ec53ULL;
    x ^= x >> 33;
    return x;
}

__global__ void checksum_kernel(const float *__restrict__ data,
                                unsigned long long *__restrict__ block_hashes,
                                size_t count)
{
    __shared__ unsigned long long shared_hash[256];

    const unsigned int tid = threadIdx.x;
    unsigned long long local = 0xcbf29ce484222325ULL;

    for (size_t idx = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
         idx < count;
         idx += static_cast<size_t>(blockDim.x) * gridDim.x) {
        unsigned int bits = __float_as_uint(data[idx]);
        unsigned long long value = (static_cast<unsigned long long>(bits) << 32) ^
                                   static_cast<unsigned long long>(idx);
        local ^= checksum_mix(value + 0x9e3779b97f4a7c15ULL);
    }

    shared_hash[tid] = local;
    __syncthreads();

#pragma unroll
    for (unsigned int stride = 128; stride > 0; stride >>= 1) {
        if (tid < stride) {
            shared_hash[tid] ^= shared_hash[tid + stride];
        }
        __syncthreads();
    }

    if (tid == 0) {
        block_hashes[blockIdx.x] = shared_hash[0];
    }
}

void launch_jacobi_step(float *d_f,
                        const float *d_f_old,
                        float *d_error,
                        dim3 grid,
                        dim3 block)
{
    jacobi_step<<<grid, block>>>(d_f, d_f_old, d_error);
    CUDA_CHECK(cudaGetLastError());
}

void launch_swap_data(const float *d_f,
                      float *d_f_old,
                      dim3 grid,
                      dim3 block)
{
    swap_data<<<grid, block>>>(d_f, d_f_old);
    CUDA_CHECK(cudaGetLastError());
}

unsigned long long compute_checksum(const float *d_f)
{
    const size_t count = static_cast<size_t>(N) * static_cast<size_t>(N);
    const int threads = 256;
    const int blocks = 4096;

    unsigned long long *d_block_hashes = nullptr;
    CUDA_CHECK(cudaMalloc(reinterpret_cast<void **>(&d_block_hashes),
                          static_cast<size_t>(blocks) * sizeof(unsigned long long)));

    checksum_kernel<<<blocks, threads>>>(d_f, d_block_hashes, count);
    CUDA_CHECK(cudaGetLastError());

    std::vector<unsigned long long> block_hashes(static_cast<size_t>(blocks));
    CUDA_CHECK(cudaMemcpy(block_hashes.data(),
                          d_block_hashes,
                          static_cast<size_t>(blocks) * sizeof(unsigned long long),
                          cudaMemcpyDeviceToHost));

    CUDA_CHECK(cudaFree(d_block_hashes));

    unsigned long long checksum = 0x84222325cbf29ce4ULL;
    for (int i = 0; i < blocks; ++i) {
        checksum ^= checksum_mix(block_hashes[static_cast<size_t>(i)] ^
                                 static_cast<unsigned long long>(i));
    }
    checksum ^= checksum_mix(static_cast<unsigned long long>(count));

    return checksum;
}

void print_results(float error,
                   int num_iters,
                   int max_iters,
                   long long time,
                   double duration)
{
    std::cout << "Average execution time per iteration: " << (time * 1e-9f) / num_iters << " (s)\n";

    if (error <= 1.e-5f && num_iters < max_iters) {
        std::cout << "PASS" << std::endl;
    } else {
        std::cout << "FAIL" << std::endl;
    }

    std::cout << "Total elapsed time: " << std::setprecision(4) << duration << " seconds" << std::endl;
}

int run_event_based_simulation_baseline()
{
    std::clock_t start_time = std::clock();

    float *d_f;
    float *d_f_old;
    float *d_error;

    float *f = static_cast<float *>(std::aligned_alloc(64, static_cast<size_t>(N) * N * sizeof(float)));
    float *f_old = static_cast<float *>(std::aligned_alloc(64, static_cast<size_t>(N) * N * sizeof(float)));

    if (f == nullptr || f_old == nullptr) {
        std::fprintf(stderr, "Host allocation failed\n");
        std::free(f);
        std::free(f_old);
        return -1;
    }

    grid_init_do_not_profile(f, f_old);

    CUDA_CHECK(cudaMalloc(reinterpret_cast<void **>(&d_f), static_cast<size_t>(N) * N * sizeof(float)));
    CUDA_CHECK(cudaMemcpy(d_f, f, static_cast<size_t>(N) * N * sizeof(float), cudaMemcpyHostToDevice));

    CUDA_CHECK(cudaMalloc(reinterpret_cast<void **>(&d_f_old), static_cast<size_t>(N) * N * sizeof(float)));
    CUDA_CHECK(cudaMemcpy(d_f_old, f_old, static_cast<size_t>(N) * N * sizeof(float), cudaMemcpyHostToDevice));

    CUDA_CHECK(cudaMalloc(reinterpret_cast<void **>(&d_error), sizeof(float)));

    float error = std::numeric_limits<float>::max();
    const float tolerance = 1.e-5f;

    const int max_iters = 10000;
    int num_iters = 0;

    dim3 grid(N / 16, N / 16);
    dim3 block(16, 16);

    CUDA_CHECK(cudaDeviceSynchronize());
    auto start = std::chrono::steady_clock::now();

    while (error > tolerance && num_iters < max_iters) {
        CUDA_CHECK(cudaMemset(d_error, 0, 4));

        launch_jacobi_step(d_f, d_f_old, d_error, grid, block);

        launch_swap_data(d_f, d_f_old, grid, block);

        CUDA_CHECK(cudaMemcpy(&error, d_error, sizeof(float), cudaMemcpyDeviceToHost));

        error = sqrtf(error / (N * N));

        if (num_iters % 1000 == 0) {
            std::cout << "Error after iteration " << num_iters << " = " << error << std::endl;
        }

        ++num_iters;
    }

    CUDA_CHECK(cudaDeviceSynchronize());
    auto end = std::chrono::steady_clock::now();
    auto time = std::chrono::duration_cast<std::chrono::nanoseconds>(end - start).count();

    unsigned long long checksum = compute_checksum(d_f);

    CUDA_CHECK(cudaFree(d_f));
    CUDA_CHECK(cudaFree(d_f_old));
    CUDA_CHECK(cudaFree(d_error));
    std::free(f);
    std::free(f_old);

    double duration = (std::clock() - start_time) / static_cast<double>(CLOCKS_PER_SEC);

    print_results(error, num_iters, max_iters, time, duration);
    std::cout << "Checksum: " << checksum << std::endl;

    if (error <= tolerance && num_iters < max_iters) {
        return 0;
    }

    return -1;
}

int main(int argc, char **argv)
{
    Inputs inputs = read_CLI(argc, argv);

    int num_nucs[N_MATERIALS];
    int mats[N_MATERIALS * MAX_NUCS];
    double concs[N_MATERIALS * MAX_NUCS];

    load_num_nucs(num_nucs);
    load_mats(mats);
    load_concs(concs);

    if (inputs.help) {
        std::printf("Usage: ./a.out [-n size] [-i iterations] [-b block]\n");
        return 0;
    }

    return run_event_based_simulation_baseline();
}
