inline double LCG_random_double(uint64_t *seed) {
    const uint64_t m = 9223372036854775808ULL;
    const uint64_t a = 2806196910506780709ULL;
    const uint64_t c = 1ULL;
    *seed = (a * (*seed) + c) % m;
    return static_cast<double>(*seed) / static_cast<double>(m);
}
inline uint64_t fast_forward_LCG(uint64_t seed, uint64_t n) {
    const uint64_t m = 9223372036854775808ULL;
    uint64_t a = 2806196910506780709ULL;
    uint64_t c = 1ULL;
    n %= m;
    uint64_t a_new = 1, c_new = 0;
    while (n) {
        if (n & 1) {
            a_new *= a;
            c_new = c_new * a + c;
        }
        c *= (a + 1);
        a *= a;
        n >>= 1;
    }
    return (a_new * seed + c_new) % m;
}
inline int pick_mat(uint64_t *seed) {
    const double dist[12] = {0.140,0.052,0.275,0.134,0.154,0.064,
                             0.066,0.055,0.008,0.015,0.025,0.013};
    double roll = LCG_random_double(seed);
    double running = 0.0;
    for (int i = 0; i < 12; ++i) {
        running += dist[i];
        if (roll < running) return i;
    }
    return 0;
}
inline long grid_search(long n, double quarry, const double *A) {
    long lo = 0, hi = n - 1;
    while (hi - lo > 1) {
        long mid = lo + (hi - lo) / 2;
        if (A[mid] > quarry) hi = mid;
        else lo = mid;
    }
    return lo;
}
inline long grid_search_nuclide(long n, double quarry,
                               const NuclideGridPoint *A,
                               long low, long high) {
    long lo = low, hi = high;
    while (hi - lo > 1) {
        long mid = lo + (hi - lo) / 2;
        if (A[mid].energy > quarry) hi = mid;
        else lo = mid;
    }
    return lo;
}

inline void calculate_micro_xs(double p_energy, int nuc,
                               long n_isotopes, long n_gridpoints,
                               const double *egrid, const int *index_data,
                               const NuclideGridPoint *nuclide_grids,
                               long idx, double *xs_vector,
                               int grid_type, int hash_bins) {
    const NuclideGridPoint *low, *high;
    if (grid_type == NUCLIDE) {
        idx = grid_search_nuclide(n_gridpoints, p_energy,
                                  nuclide_grids + nuc * n_gridpoints, 0,
                                  n_gridpoints - 1);
        low = (idx == n_gridpoints - 1) ?
              nuclide_grids + nuc * n_gridpoints + idx - 1 :
              nuclide_grids + nuc * n_gridpoints + idx;
    } else if (grid_type == UNIONIZED) {
        int id = index_data[idx * n_isotopes + nuc];
        low = (id == n_gridpoints - 1) ?
              nuclide_grids + nuc * n_gridpoints + id - 1 :
              nuclide_grids + nuc * n_gridpoints + id;
    } else { // HASH
        int u_low = index_data[idx * n_isotopes + nuc];
        int u_high = (idx == hash_bins - 1) ?
                     n_gridpoints - 1 :
                     index_data[(idx + 1) * n_isotopes + nuc] + 1;
        double e_low  = nuclide_grids[nuc * n_gridpoints + u_low].energy;
        double e_high = nuclide_grids[nuc * n_gridpoints + u_high].energy;
        int lower;
        if (p_energy <= e_low) lower = 0;
        else if (p_energy >= e_high) lower = n_gridpoints - 1;
        else lower = grid_search_nuclide(n_gridpoints, p_energy,
                                         nuclide_grids + nuc * n_gridpoints,
                                         u_low, u_high);
        low = (lower == n_gridpoints - 1) ?
              nuclide_grids + nuc * n_gridpoints + lower - 1 :
              nuclide_grids + nuc * n_gridpoints + lower;
    }
    high = low + 1;
    double f = (high->energy - p_energy) / (high->energy - low->energy);
    xs_vector[0] = high->total_xs - f * (high->total_xs - low->total_xs);
    xs_vector[1] = high->elastic_xs - f * (high->elastic_xs - low->elastic_xs);
    xs_vector[2] = high->absorbtion_xs - f * (high->absorbtion_xs - low->absorbtion_xs);
    xs_vector[3] = high->fission_xs - f * (high->fission_xs - low->fission_xs);
    xs_vector[4] = high->nu_fission_xs - f * (high->nu_fission_xs - low->nu_fission_xs);
}
inline void calculate_macro_xs(double p_energy, int mat,
                               long n_isotopes, long n_gridpoints,
                               const int *num_nucs, const double *concs,
                               const double *egrid, const int *index_data,
                               const NuclideGridPoint *nuclide_grids,
                               const int *mats, double *macro_xs_vector,
                               int grid_type, int hash_bins, int max_num_nucs) {
    for (int k = 0; k < 5; ++k) macro_xs_vector[k] = 0.0;
    long idx = -1;
    if (grid_type == UNIONIZED)
        idx = grid_search(n_isotopes * n_gridpoints, p_energy, egrid);
    else if (grid_type == HASH) {
        double du = 1.0 / hash_bins;
        idx = static_cast<long>(p_energy / du);
    }
    for (int j = 0; j < num_nucs[mat]; ++j) {
        double xs_vec[5];
        int p_nuc = mats[mat * max_num_nucs + j];
        double conc = concs[mat * max_num_nucs + j];
        calculate_micro_xs(p_energy, p_nuc, n_isotopes, n_gridpoints,
                           egrid, index_data, nuclide_grids,
                           idx, xs_vec, grid_type, hash_bins);
        for (int k = 0; k < 5; ++k) macro_xs_vector[k] += xs_vec[k] * conc;
    }
}

    const NuclideGridPoint *A = (const NuclideGridPoint *)a;
    const NuclideGridPoint *B = (const NuclideGridPoint *)b;

    return 0;
}

size_t estimate_mem_usage(Inputs in) {
    size_t single_nuclide_grid = in.n_gridpoints * sizeof(NuclideGridPoint);
    size_t all_nuclide_grids   = in.n_isotopes * single_nuclide_grid;
    size_t size_UEG            = in.n_isotopes * in.n_gridpoints * sizeof(double) +
                                 in.n_isotopes * in.n_gridpoints * in.n_isotopes * sizeof(int);
    size_t size_hash_grid      = in.hash_bins * in.n_isotopes * sizeof(int);
    size_t memtotal;
    if (in.grid_type == UNIONIZED) memtotal = all_nuclide_grids + size_UEG;
    else if (in.grid_type == NUCLIDE) memtotal = all_nuclide_grids;
    else memtotal = all_nuclide_grids + size_hash_grid;
    memtotal = static_cast<size_t>(std::ceil(memtotal / (1024.0 * 1024.0)));
    return memtotal;
}

    GSD.concs = sycl::malloc_device<double>(SD.length_concs, q);
    if (SD.length_concs)
        q.memcpy(GSD.concs, SD.concs,
                 SD.length_concs * sizeof(double)).wait();
    GSD.mats = sycl::malloc_device<int>(SD.length_mats, q);
    if (SD.length_mats)
        q.memcpy(GSD.mats, SD.mats,
                 SD.length_mats * sizeof(int)).wait();
    GSD.unionized_energy_array = sycl::malloc_device<double>(SD.length_unionized_energy_array, q);
    if (SD.length_unionized_energy_array)
        q.memcpy(GSD.unionized_energy_array, SD.unionized_energy_array,
                 SD.length_unionized_energy_array * sizeof(double)).wait();
    GSD.index_grid = sycl::malloc_device<int>(SD.length_index_grid, q);
    if (SD.length_index_grid)
        q.memcpy(GSD.index_grid, SD.index_grid,
                 SD.length_index_grid * sizeof(int)).wait();
    GSD.nuclide_grid = sycl::malloc_device<NuclideGridPoint>(SD.length_nuclide_grid, q);
    if (SD.length_nuclide_grid)
        q.memcpy(GSD.nuclide_grid, SD.nuclide_grid,
                 SD.length_nuclide_grid * sizeof(NuclideGridPoint)).wait();
    GSD.verification = sycl::malloc_device<unsigned long>(in.lookups, q);
    GSD.length_verification = in.lookups;
    q.wait();
    return GSD;
}

// ------------------------------------------------------------------
// Grid initialization (full implementation)
// ------------------------------------------------------------------
SimulationData grid_init_do_not_profile(const Inputs &in, int mype) {
    (void)mype;
    SimulationData SD{};
    size_t nbytes = 0;
    uint64_t seed = 42;
    // Nuclide grid
    SD.length_nuclide_grid = static_cast<int>(in.n_isotopes * in.n_gridpoints);
    SD.nuclide_grid = (NuclideGridPoint *)malloc(SD.length_nuclide_grid * sizeof(NuclideGridPoint));
    assert(SD.nuclide_grid);
    nbytes += SD.length_nuclide_grid * sizeof(NuclideGridPoint);
    for (int i = 0; i < SD.length_nuclide_grid; ++i) {
        SD.nuclide_grid[i].energy        = LCG_random_double(&seed);
        SD.nuclide_grid[i].total_xs      = LCG_random_double(&seed);
        SD.nuclide_grid[i].elastic_xs    = LCG_random_double(&seed);
        SD.nuclide_grid[i].absorbtion_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].fission_xs    = LCG_random_double(&seed);
        SD.nuclide_grid[i].nu_fission_xs = LCG_random_double(&seed);
    }
    for (int i = 0; i < in.n_isotopes; ++i)
        qsort(&SD.nuclide_grid[i * in.n_gridpoints], in.n_gridpoints,
              sizeof(NuclideGridPoint), NGP_compare);
    // Acceleration structures
    if (in.grid_type == UNIONIZED) {
        SD.length_unionized_energy_array = static_cast<int>(in.n_isotopes * in.n_gridpoints);
        SD.unionized_energy_array = (double *)malloc(SD.length_unionized_energy_array * sizeof(double));
        assert(SD.unionized_energy_array);
        nbytes += SD.length_unionized_energy_array * sizeof(double);
        for (int i = 0; i < SD.length_unionized_energy_array; ++i)
            SD.unionized_energy_array[i] = SD.nuclide_grid[i].energy;
        qsort(SD.unionized_energy_array, SD.length_unionized_energy_array,
              sizeof(double), double_compare);
        SD.length_index_grid = static_cast<long>(SD.length_unionized_energy_array * in.n_isotopes);
        SD.index_grid = (int *)malloc(SD.length_index_grid * sizeof(int));
        assert(SD.index_grid);
        nbytes += SD.length_index_grid * sizeof(int);
        int *idx_low = (int *)calloc(in.n_isotopes, sizeof(int));
        double *energy_high = (double *)malloc(in.n_isotopes * sizeof(double));
        for (int i = 0; i < in.n_isotopes; ++i)
            energy_high[i] = SD.nuclide_grid[i * in.n_gridpoints + 1].energy;
        for (long e = 0; e < SD.length_unionized_energy_array; ++e) {
            double ue = SD.unionized_energy_array[e];
            for (long i = 0; i < in.n_isotopes; ++i) {
                if (ue < energy_high[i])
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                else if (idx_low[i] == in.n_gridpoints - 2)
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                else {
                    idx_low[i]++;
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                    energy_high[i] = SD.nuclide_grid[i * in.n_gridpoints + idx_low[i] + 1].energy;
                }
            }
        }
        free(idx_low);
        free(energy_high);
    } else if (in.grid_type == HASH) {
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = static_cast<long>(in.hash_bins * in.n_isotopes);
        SD.index_grid = (int *)malloc(SD.length_index_grid * sizeof(int));
        assert(SD.index_grid);
        nbytes += SD.length_index_grid * sizeof(int);
        double du = 1.0 / in.hash_bins;
        for (long e = 0; e < in.hash_bins; ++e) {
            double energy = e * du;
            for (long i = 0; i < in.n_isotopes; ++i) {
                SD.index_grid[e * in.n_isotopes + i] =
                    grid_search_nuclide(in.n_gridpoints, energy,
                                        SD.nuclide_grid + i * in.n_gridpoints,
                                        0, in.n_gridpoints - 1);
            }
        }
    } else {
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = 0;
        SD.unionized_energy_array = nullptr;
        SD.index_grid = nullptr;
    }
    // Materials
    SD.num_nucs = load_num_nucs(in.n_isotopes);
    SD.length_num_nucs = 12;
    SD.mats = load_mats(SD.num_nucs, in.n_isotopes, &SD.max_num_nucs);
    SD.length_mats = SD.length_num_nucs * SD.max_num_nucs;
    SD.concs = load_concs(SD.num_nucs, SD.max_num_nucs);
    SD.length_concs = SD.length_mats;
    // Verification array (host side)
    SD.verification = nullptr;
    SD.length_verification = 0;
    SD.p_energy_samples = nullptr;
    SD.mat_samples = nullptr;
    return SD;
}

// ------------------------------------------------------------------
// Baseline simulation (event based)
// ------------------------------------------------------------------
unsigned long long run_event_based_simulation_baseline(sycl::queue &q,
                                                      const Inputs &in,
                                                      SimulationData &GSD,
                                                      int mype) {
    if (mype == 0) printf("Running baseline event‑based simulation...\n");
    const size_t global = static_cast<size_t>(in.lookups);
    q.submit([&](sycl::handler &cgh) {
        cgh.parallel_for<class baseline_kernel>(sycl::range<1>(global),
        [=](sycl::id<1> idx) {
            const int i = idx[0];
            uint64_t seed = STARTING_SEED;
            seed = fast_forward_LCG(seed, 2ULL * i);
            double p_energy = LCG_random_double(&seed);
            int mat = pick_mat(&seed);
            double macro_xs[5] = {0};
            calculate_macro_xs(p_energy, mat,
                               in.n_isotopes, in.n_gridpoints,
                               GSD.num_nucs, GSD.concs,
                               GSD.unionized_energy_array,
                               GSD.index_grid,
                               GSD.nuclide_grid,
                               GSD.mats,
                               macro_xs,
                               in.grid_type, in.hash_bins,
                               GSD.max_num_nucs);
            double max = -1.0;
            int max_idx = 0;
            for (int j = 0; j < 5; ++j)
                if (macro_xs[j] > max) { max = macro_xs[j]; max_idx = j; }
            GSD.verification[i] = static_cast<unsigned long>(max_idx + 1);
        });
    }).wait();
    unsigned long long checksum = 0;
    {
        std::vector<unsigned long> host_verif(in.lookups);
        q.memcpy(host_verif.data(), GSD.verification,
                 in.lookups * sizeof(unsigned long)).wait();
        for (auto v : host_verif) checksum += v;
    }
    return checksum;
}

// ------------------------------------------------------------------
// Optimization 4 – full material sort + per‑material kernels
// ------------------------------------------------------------------
unsigned long long run_event_based_simulation_optimization_4(sycl::queue &q,
                                                            const Inputs &in,
                                                            SimulationData &GSD,
                                                            int mype) {
    if (mype == 0) printf("Running Optimization 4 – material sort + kernels...\n");
    GSD.p_energy_samples = sycl::malloc_device<double>(in.lookups, q);
    GSD.mat_samples     = sycl::malloc_device<int>(in.lookups, q);
    GSD.length_p_energy_samples = in.lookups;
    GSD.length_mat_samples      = in.lookups;
    const size_t global = static_cast<size_t>(in.lookups);
    q.submit([&](sycl::handler &cgh){
        cgh.parallel_for<class sampling_opt4>(sycl::range<1>(global), [=](sycl::id<1> idx){
            const int i = idx[0];
            uint64_t seed = STARTING_SEED;
            seed = fast_forward_LCG(seed, 2ULL * i);
            GSD.p_energy_samples[i] = LCG_random_double(&seed);
            GSD.mat_samples[i] = pick_mat(&seed);
        });
    }).wait();
    int lookups_per_mat[12] = {0};
    {
        std::vector<int> host_mat(in.lookups);
        q.memcpy(host_mat.data(), GSD.mat_samples,
                 in.lookups * sizeof(int)).wait();
        for (int i = 0; i < in.lookups; ++i) {
            int m = host_mat[i];
            if (m >=0 && m < 12) lookups_per_mat[m]++;
        }
    }
    {
        std::vector<int> host_mat(in.lookups);
        std::vector<double> host_energy(in.lookups);
        q.memcpy(host_mat.data(), GSD.mat_samples,
                 in.lookups * sizeof(int)).wait();
        q.memcpy(host_energy.data(), GSD.p_energy_samples,
                 in.lookups * sizeof(double)).wait();
        std::stable_sort(host_mat.begin(), host_mat.end(),
                         [&](int a, int b){ return a < b; });
        // sort energies according to material order
        std::vector<std::pair<int,double>> pairs(in.lookups);
        for (size_t i = 0; i < in.lookups; ++i) pairs[i] = {host_mat[i], host_energy[i]};
        std::stable_sort(pairs.begin(), pairs.end(),
                         [](const auto &a, const auto &b){ return a.first < b.first; });
        for (size_t i = 0; i < in.lookups; ++i) {
            host_mat[i] = pairs[i].first;
            host_energy[i] = pairs[i].second;
        }
        q.memcpy(GSD.mat_samples, host_mat.data(),
                 in.lookups * sizeof(int)).wait();
        q.memcpy(GSD.p_energy_samples, host_energy.data(),
                 in.lookups * sizeof(double)).wait();
    }
    int offset = 0;
    for (int m = 0; m < 12; ++m) {
        int n = lookups_per_mat[m];
        if (n == 0) continue;
        q.submit([&, m, n, offset](sycl::handler &cgh){
            cgh.parallel_for<class mat_kernel_opt4>(sycl::range<1>(n), [=](sycl::id<1> idx){
                int i = idx[0] + offset;
                double macro_xs[5] = {0};
                calculate_macro_xs(GSD.p_energy_samples[i],
                                   GSD.mat_samples[i],
                                   in.n_isotopes, in.n_gridpoints,
                                   GSD.num_nucs, GSD.concs,
                                   GSD.unionized_energy_array,
                                   GSD.index_grid,
                                   GSD.nuclide_grid,
                                   GSD.mats,
                                   macro_xs,
                                   in.grid_type, in.hash_bins,
                                   GSD.max_num_nucs);
                double max = -1.0; int max_idx = 0;
                for (int j = 0; j < 5; ++j)
                    if (macro_xs[j] > max) { max = macro_xs[j]; max_idx = j; }
                GSD.verification[i] = static_cast<unsigned long>(max_idx + 1);
            });
        });
        offset += n;
    }
    q.wait();
    unsigned long long checksum = 0;
    {
        std::vector<unsigned long> host_verif(in.lookups);
        q.memcpy(host_verif.data(), GSD.verification,
                 in.lookups * sizeof(unsigned long)).wait();
        for (auto v : host_verif) checksum += v;
    }
    return checksum;
}

// ------------------------------------------------------------------
// Optimization 5 – fuel / non‑fuel partition
// ------------------------------------------------------------------
struct is_fuel {
    bool operator()(const int &v) const { return v == 0; }
};
unsigned long long run_event_based_simulation_optimization_5(sycl::queue &q,
                                                            const Inputs &in,
                                                            SimulationData &GSD,
                                                            int mype) {
    if (mype == 0) printf("Running Optimization 5 – fuel / non‑fuel partition...\n");
    GSD.p_energy_samples = sycl::malloc_device<double>(in.lookups, q);
    GSD.mat_samples     = sycl::malloc_device<int>(in.lookups, q);
    GSD.length_p_energy_samples = in.lookups;
    GSD.length_mat_samples      = in.lookups;
    const size_t global = static_cast<size_t>(in.lookups);
    q.submit([&](sycl::handler &cgh){
        cgh.parallel_for<class sampling_opt5>(sycl::range<1>(global), [=](sycl::id<1> idx){
            const int i = idx[0];
            uint64_t seed = STARTING_SEED;
            seed = fast_forward_LCG(seed, 2ULL * i);
            GSD.p_energy_samples[i] = LCG_random_double(&seed);
            GSD.mat_samples[i] = pick_mat(&seed);
        });
    }).wait();
    int n_fuel = 0;
    {
        std::vector<int> host_mat(in.lookups);
        q.memcpy(host_mat.data(), GSD.mat_samples,
                 in.lookups * sizeof(int)).wait();
        for (int i = 0; i < in.lookups; ++i)
            if (host_mat[i] == 0) n_fuel++;
    }
    {
        std::vector<int> host_mat(in.lookups);
        std::vector<double> host_energy(in.lookups);
        q.memcpy(host_mat.data(), GSD.mat_samples,
                 in.lookups * sizeof(int)).wait();
        q.memcpy(host_energy.data(), GSD.p_energy_samples,
                 in.lookups * sizeof(double)).wait();
        int f = 0, nf = in.lookups - 1;
        while (f < nf) {
            while (f < nf && host_mat[f] == 0) ++f;
            while (f < nf && host_mat[nf] != 0) --nf;
            if (f < nf) {
                std::swap(host_mat[f], host_mat[nf]);
                std::swap(host_energy[f], host_energy[nf]);
                ++f; --nf;
            }
        }
        q.memcpy(GSD.mat_samples, host_mat.data(),
                 in.lookups * sizeof(int)).wait();
        q.memcpy(GSD.p_energy_samples, host_energy.data(),
                 in.lookups * sizeof(double)).wait();
    }
    q.submit([&](sycl::handler &cgh){

            GSD.verification[i] = static_cast<unsigned long>(max_idx + 1);
        });
    }).wait();
    int n_nonfuel = in.lookups - n_fuel;
    q.submit([&](sycl::handler &cgh){

// Optimization 6 – material + energy double sort
// ------------------------------------------------------------------
unsigned long long run_event_based_simulation_optimization_6(sycl::queue &q,
                                                            const Inputs &in,
                                                            SimulationData &GSD,
                                                            int mype) {
    if (mype == 0) printf("Running Optimization 6 – material & energy double sort...\n");
    GSD.p_energy_samples = sycl::malloc_device<double>(in.lookups, q);
    GSD.mat_samples     = sycl::malloc_device<int>(in.lookups, q);
    GSD.length_p_energy_samples = in.lookups;
    GSD.length_mat_samples      = in.lookups;
    const size_t global = static_cast<size_t>(in.lookups);
    q.submit([&](sycl::handler &cgh){
        cgh.parallel_for<class sampling_opt6>(sycl::range<1>(global), [=](sycl::id<1> idx){
            const int i = idx[0];
            uint64_t seed = STARTING_SEED;
            seed = fast_forward_LCG(seed, 2ULL * i);
            GSD.p_energy_samples[i] = LCG_random_double(&seed);
            GSD.mat_samples[i] = pick_mat(&seed);
        });
    }).wait();
    int lookups_per_mat[12] = {0};
    {
        std::vector<int> host_mat(in.lookups);
        q.memcpy(host_mat.data(), GSD.mat_samples,
                 in.lookups * sizeof(int)).wait();
        for (int i = 0; i < in.lookups; ++i) {
            int m = host_mat[i];
            if (m >=0 && m < 12) lookups_per_mat[m]++;
        }
    }
    {
        std::vector<int> host_mat(in.lookups);
        std::vector<double> host_energy(in.lookups);
        q.memcpy(host_mat.data(), GSD.mat_samples,
                 in.lookups * sizeof(int)).wait();
        q.memcpy(host_energy.data(), GSD.p_energy_samples,
                 in.lookups * sizeof(double)).wait();
        std::vector<std::pair<int,double>> pairs(in.lookups);
        for (size_t i = 0; i < in.lookups; ++i) pairs[i] = {host_mat[i], host_energy[i]};
        std::stable_sort(pairs.begin(), pairs.end(),
                         [](const auto &a, const auto &b){ return a.first < b.first; });
        for (size_t i = 0; i < in.lookups; ++i) {
            host_mat[i] = pairs[i].first;
            host_energy[i] = pairs[i].second;
        }
        q.memcpy(GSD.mat_samples, host_mat.data(),
                 in.lookups * sizeof(int)).wait();
        q.memcpy(GSD.p_energy_samples, host_energy.data(),
                 in.lookups * sizeof(double)).wait();
    }
    int offset = 0;
    for (int m = 0; m < 12; ++m) {
        int n = lookups_per_mat[m];
        if (n == 0) continue;
        std::vector<double> sub_energy(n);
        std::vector<int> sub_mat(n);
        q.memcpy(sub_energy.data(),
                 GSD.p_energy_samples + offset,
                 n * sizeof(double)).wait();
        q.memcpy(sub_mat.data(),
                 GSD.mat_samples + offset,
                 n * sizeof(int)).wait();
        std::vector<std::pair<double,int>> sub_pairs(n);
        for (int i = 0; i < n; ++i) sub_pairs[i] = {sub_energy[i], sub_mat[i]};
        std::stable_sort(sub_pairs.begin(), sub_pairs.end(),
                         [](const auto &a, const auto &b){ return a.first < b.first; });
        for (int i = 0; i < n; ++i) {
            sub_energy[i] = sub_pairs[i].first;
            sub_mat[i]    = sub_pairs[i].second;
        }
        q.memcpy(GSD.p_energy_samples + offset, sub_energy.data(),
                 n * sizeof(double)).wait();
        q.memcpy(GSD.mat_samples + offset, sub_mat.data(),
                 n * sizeof(int)).wait();
        offset += n;
    }
    offset = 0;
    for (int m = 0; m < 12; ++m) {
        int n = lookups_per_mat[m];
        if (n == 0) continue;
        q.submit([&, m, n, offset](sycl::handler &cgh){
            cgh.parallel_for<class mat_kernel_opt6>(sycl::range<1>(n), [=](sycl::id<1> idx){
                int i = idx[0] + offset;
                double macro_xs[5] = {0};
                calculate_macro_xs(GSD.p_energy_samples[i],
                                   GSD.mat_samples[i],
                                   in.n_isotopes, in.n_gridpoints,
                                   GSD.num_nucs, GSD.concs,
                                   GSD.unionized_energy_array,
                                   GSD.index_grid,
                                   GSD.nuclide_grid,
                                   GSD.mats,
                                   macro_xs,
                                   in.grid_type, in.hash_bins,
                                   GSD.max_num_nucs);
                double max = -1.0; int max_idx = 0;
                for (int j = 0; j < 5; ++j)
                    if (macro_xs[j] > max) { max = macro_xs[j]; max_idx = j; }
                GSD.verification[i] = static_cast<unsigned long>(max_idx + 1);
            });
        });
        offset += n;
    }
    q.wait();
    unsigned long long checksum = 0;
    {
        std::vector<unsigned long> host_verif(in.lookups);
        q.memcpy(host_verif.data(), GSD.verification,
                 in.lookups * sizeof(unsigned long)).wait();
        for (auto v : host_verif) checksum += v;
    }
    return checksum;
}

int main(int argc, char *argv[]) {
    int version = 20;
    int mype = 0;
    double omp_start, omp_end;
    int nprocs = 1;
    unsigned long long verification = 0;

    Inputs in = read_CLI(argc, argv);
    if (mype == 0) print_inputs(in, nprocs, version);

    SimulationData SD;
    if (in.binary_mode == READ) SD = binary_read(in);
    else SD = grid_init_do_not_profile(in, mype);
    if (in.binary_mode == WRITE && mype == 0) binary_write(in, SD);

    sycl::queue q{sycl::gpu_selector_v, sycl::property::queue::in_order{}};

    SimulationData GSD = move_simulation_data_to_device(q, in, SD);

    if (mype == 0) {
        printf("\n");
        border_print();
        center_print("SIMULATION", 79);
        border_print();
    }

    omp_start = get_time();

    if (in.simulation_method == EVENT_BASED) {
        switch (in.kernel_id) {
            case 0: verification = run_event_based_simulation_baseline(q, in, GSD, mype); break;
            case 1: verification = run_event_based_simulation_optimization_1(q, in, GSD, mype); break;
            case 2: verification = run_event_based_simulation_optimization_2(q, in, GSD, mype); break;
            case 3: verification = run_event_based_simulation_optimization_3(q, in, GSD, mype); break;
            case 4: verification = run_event_based_simulation_optimization_4(q, in, GSD, mype); break;
            case 5: verification = run_event_based_simulation_optimization_5(q, in, GSD, mype); break;
            case 6: verification = run_event_based_simulation_optimization_6(q, in, GSD, mype); break;
            default:
                printf("Error: No kernel ID %d found!\n", in.kernel_id);
                std::exit(1);
        }
    } else {
        printf("History‑based simulation not implemented. Use event‑based method.\n");
        std::exit(1);
    }

    if (mype == 0) {
        printf("\nSimulation complete.\n");
    }

    omp_end = get_time();

    release_device_memory(q, GSD);
    verification %= 999983ULL;

    int is_invalid = print_results(in, mype, omp_end - omp_start,
                                   nprocs, verification);
    return is_invalid;
}
