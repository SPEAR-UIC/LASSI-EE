int NGP_compare(const void* a, const void* b) {
    const NuclideGridPoint* A = (const NuclideGridPoint*)a;
    const NuclideGridPoint* B = (const NuclideGridPoint*)b;
    return (A->energy > B->energy) - (A->energy < B->energy);
}
int double_compare(const void* a, const void* b) {
    double A = *(const double*)a;
    double B = *(const double*)b;
    return (A > B) - (A < B);
}
double get_time() {
    auto now = std::chrono::high_resolution_clock::now();
    auto us = std::chrono::duration_cast<std::chrono::microseconds>(now.time_since_epoch()).count();
    return us * 1e-6;
}

inline double LCG_random_double(uint64_t* seed) {
    const uint64_t m = 9223372036854775808ULL;
    const uint64_t a = 2806196910506780709ULL;
    const uint64_t c = 1ULL;
    *seed = (a * (*seed) + c) % m;
    return static_cast<double>(*seed) / static_cast<double>(m);
}
inline uint64_t fast_forward_LCG(uint64_t seed, uint64_t n) {
    const uint64_t m = 9223372036854775808ULL;
    uint64_t a = 2806196910506780709ULL;
    uint64_t c = 1ULL;
    n %= m;
    uint64_t a_new = 1;
    uint64_t c_new = 0;
    while (n > 0) {
        if (n & 1) {
            a_new = a_new * a;
            c_new = c_new * a + c;
        }
        c *= (a + 1);
        a *= a;
        n >>= 1;
    }
    return (a_new * seed + c_new) % m;
}

inline long grid_search(long n, double quarry, const double* A) {
    long lower = 0, upper = n - 1;
    while (upper - lower > 1) {
        long mid = lower + (upper - lower) / 2;
        if (A[mid] > quarry) upper = mid;
        else lower = mid;
    }
    return lower;
}
inline long grid_search_nuclide(long n, double quarry,
                               const NuclideGridPoint* A,
                               long low, long high) {
    long lower = low, upper = high;
    while (upper - lower > 1) {
        long mid = lower + (upper - lower) / 2;
        if (A[mid].energy > quarry) upper = mid;
        else lower = mid;
    }
    return lower;
}
inline int pick_mat(uint64_t* seed) {
    const double dist[12] = {0.140,0.052,0.275,0.134,0.154,0.064,
                             0.066,0.055,0.008,0.015,0.025,0.013};
    double roll = LCG_random_double(seed);
    double cum = 0.0;
    for (int i = 0; i < 12; ++i) {
        cum += dist[i];
        if (roll < cum) return i;
    }
    return 0;
}
inline void calculate_micro_xs(double p_energy, int nuc, long n_isotopes,
                               long n_gridpoints,
                               const double* egrid, const int* index_data,
                               const NuclideGridPoint* nuclide_grids,
                               long idx, double* xs_vector,
                               int grid_type, int hash_bins) {
    const NuclideGridPoint* low;
    const NuclideGridPoint* high;
    if (grid_type == NUCLIDE) {
        idx = grid_search_nuclide(n_gridpoints, p_energy,
                                  nuclide_grids + nuc * n_gridpoints, 0, n_gridpoints - 1);
        low = (idx == n_gridpoints - 1) ?
              &nuclide_grids[nuc * n_gridpoints + idx - 1] :
              &nuclide_grids[nuc * n_gridpoints + idx];
    } else if (grid_type == UNIONIZED) {
        int ind = index_data[idx * n_isotopes + nuc];
        low = (ind == n_gridpoints - 1) ?
              &nuclide_grids[nuc * n_gridpoints + ind - 1] :
              &nuclide_grids[nuc * n_gridpoints + ind];
    } else { // HASH
        int u_low = index_data[idx * n_isotopes + nuc];
        int u_high = (idx == hash_bins - 1) ?
                     n_gridpoints - 1 :
                     index_data[(idx + 1) * n_isotopes + nuc] + 1;
        double e_low = nuclide_grids[nuc * n_gridpoints + u_low].energy;
        double e_high = nuclide_grids[nuc * n_gridpoints + u_high].energy;
        int lower;
        if (p_energy <= e_low) lower = 0;
        else if (p_energy >= e_high) lower = n_gridpoints - 1;
        else lower = grid_search_nuclide(n_gridpoints, p_energy,
                                         nuclide_grids + nuc * n_gridpoints,
                                         u_low, u_high);
        low = (lower == n_gridpoints - 1) ?
              &nuclide_grids[nuc * n_gridpoints + lower - 1] :
              &nuclide_grids[nuc * n_gridpoints + lower];
    }
    high = low + 1;
    double f = (high->energy - p_energy) / (high->energy - low->energy);
    xs_vector[0] = high->total_xs - f * (high->total_xs - low->total_xs);
    xs_vector[1] = high->elastic_xs - f * (high->elastic_xs - low->elastic_xs);
    xs_vector[2] = high->absorbtion_xs - f * (high->absorbtion_xs - low->absorbtion_xs);
    xs_vector[3] = high->fission_xs - f * (high->fission_xs - low->fission_xs);
    xs_vector[4] = high->nu_fission_xs - f * (high->nu_fission_xs - low->nu_fission_xs);
}
inline void calculate_macro_xs(double p_energy, int mat,
                               long n_isotopes, long n_gridpoints,
                               const int* num_nucs, const double* concs,
                               const double* egrid, const int* index_data,
                               const NuclideGridPoint* nuclide_grids,
                               const int* mats, double* macro_xs_vector,
                               int grid_type, int hash_bins, int max_num_nucs) {
    for (int k = 0; k < 5; ++k) macro_xs_vector[k] = 0.0;
    long idx = -1;
    if (grid_type == UNIONIZED) idx = grid_search(n_isotopes * n_gridpoints, p_energy, egrid);
    else if (grid_type == HASH) {
        double du = 1.0 / hash_bins;
        idx = static_cast<long>(p_energy / du);
    }
    for (int j = 0; j < num_nucs[mat]; ++j) {
        double xs_vec[5];
        int p_nuc = mats[mat * max_num_nucs + j];
        double conc = concs[mat * max_num_nucs + j];
        calculate_micro_xs(p_energy, p_nuc, n_isotopes, n_gridpoints,
                           egrid, index_data, nuclide_grids,
                           idx, xs_vec, grid_type, hash_bins);
        for (int k = 0; k < 5; ++k) macro_xs_vector[k] += xs_vec[k] * conc;
    }
}

double* load_concs(int* num_nucs, int max_num_nucs) {
    uint64_t seed = STARTING_SEED * STARTING_SEED;
    double* concs = (double*)malloc(12 * max_num_nucs * sizeof(double));
    for (int i=0;i<12;i++)
        for (int j=0;j<num_nucs[i];j++)
            concs[i*max_num_nucs + j] = LCG_random_double(&seed);
    return concs;
}

SimulationData grid_init_do_not_profile(const Inputs& in) {
    SimulationData SD{};
    uint64_t seed = 42ULL;
    // Nuclide grid
    SD.length_nuclide_grid = in.n_isotopes * in.n_gridpoints;
    SD.nuclide_grid = (NuclideGridPoint*)malloc(SD.length_nuclide_grid * sizeof(NuclideGridPoint));
    for (long i=0;i<SD.length_nuclide_grid;i++) {
        SD.nuclide_grid[i].energy        = LCG_random_double(&seed);
        SD.nuclide_grid[i].total_xs      = LCG_random_double(&seed);
        SD.nuclide_grid[i].elastic_xs    = LCG_random_double(&seed);
        SD.nuclide_grid[i].absorbtion_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].fission_xs    = LCG_random_double(&seed);
        SD.nuclide_grid[i].nu_fission_xs = LCG_random_double(&seed);
    }
    for (long iso=0; iso<in.n_isotopes; ++iso)
        qsort(&SD.nuclide_grid[iso*in.n_gridpoints], in.n_gridpoints,
              sizeof(NuclideGridPoint), NGP_compare);
    // Acceleration structures
    if (in.grid_type == UNIONIZED) {
        SD.length_unionized_energy_array = in.n_isotopes * in.n_gridpoints;
        SD.unionized_energy_array = (double*)malloc(SD.length_unionized_energy_array * sizeof(double));
        for (long i=0;i<SD.length_unionized_energy_array;i++)
            SD.unionized_energy_array[i] = SD.nuclide_grid[i].energy;
        qsort(SD.unionized_energy_array, SD.length_unionized_energy_array,
              sizeof(double), double_compare);
        SD.length_index_grid = SD.length_unionized_energy_array * in.n_isotopes;
        SD.index_grid = (int*)malloc(SD.length_index_grid * sizeof(int));
        int* idx_low = (int*)calloc(in.n_isotopes, sizeof(int));
        double* energy_high = (double*)malloc(in.n_isotopes * sizeof(double));
        for (int i=0;i<in.n_isotopes;i++)
            energy_high[i] = SD.nuclide_grid[i*in.n_gridpoints + 1].energy;
        for (long e=0;e<SD.length_unionized_energy_array;e++) {
            double ue = SD.unionized_energy_array[e];
            for (long i=0;i<in.n_isotopes;i++) {
                if (ue < energy_high[i]) SD.index_grid[e*in.n_isotopes + i] = idx_low[i];
                else if (idx_low[i] == in.n_gridpoints-2) SD.index_grid[e*in.n_isotopes + i] = idx_low[i];
                else {
                    idx_low[i]++;
                    SD.index_grid[e*in.n_isotopes + i] = idx_low[i];
                    energy_high[i] = SD.nuclide_grid[i*in.n_gridpoints + idx_low[i] + 1].energy;
                }
            }
        }
        free(idx_low); free(energy_high);
    } else if (in.grid_type == HASH) {
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = (long)in.hash_bins * in.n_isotopes;
        SD.index_grid = (int*)malloc(SD.length_index_grid * sizeof(int));
        double du = 1.0 / in.hash_bins;
        for (long e=0;e<in.hash_bins;e++) {
            double energy = e * du;
            for (long i=0;i<in.n_isotopes;i++) {
                SD.index_grid[e*in.n_isotopes + i] =
                    grid_search_nuclide(in.n_gridpoints, energy,
                                        SD.nuclide_grid + i*in.n_gridpoints, 0, in.n_gridpoints-1);
            }
        }
    } else {
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = 0;
    }
    // Materials
    SD.num_nucs = load_num_nucs(in.n_isotopes);
    SD.length_num_nucs = 12;
    SD.mats = load_mats(SD.num_nucs, in.n_isotopes, &SD.max_num_nucs);
    SD.length_mats = SD.length_num_nucs * SD.max_num_nucs;
    SD.concs = load_concs(SD.num_nucs, SD.max_num_nucs);
    SD.length_concs = SD.length_mats;
    return SD;
}

// Transfer to device

SimulationData move_simulation_data_to_device(const Inputs& in, const SimulationData& SD, queue& q) {
    SimulationData GSD = SD; // shallow copy, pointers will be replaced
    // Allocate & copy
    GSD.num_nucs = (int*)malloc_device(SD.length_num_nucs * sizeof(int), q);
    q.memcpy(GSD.num_nucs, SD.num_nucs, SD.length_num_nucs * sizeof(int)).wait();
    GSD.concs = (double*)malloc_device(SD.length_concs * sizeof(double), q);
    q.memcpy(GSD.concs, SD.concs, SD.length_concs * sizeof(double)).wait();
    GSD.mats = (int*)malloc_device(SD.length_mats * sizeof(int), q);
    q.memcpy(GSD.mats, SD.mats, SD.length_mats * sizeof(int)).wait();
    GSD.unionized_energy_array = nullptr;
    if (SD.length_unionized_energy_array) {
        GSD.unionized_energy_array = (double*)malloc_device(SD.length_unionized_energy_array * sizeof(double), q);
        q.memcpy(GSD.unionized_energy_array, SD.unionized_energy_array,
                 SD.length_unionized_energy_array * sizeof(double)).wait();
    }
    GSD.index_grid = nullptr;
    if (SD.length_index_grid) {
        GSD.index_grid = (int*)malloc_device(SD.length_index_grid * sizeof(int), q);
        q.memcpy(GSD.index_grid, SD.index_grid,
                 SD.length_index_grid * sizeof(int)).wait();
    }
    GSD.nuclide_grid = (NuclideGridPoint*)malloc_device(SD.length_nuclide_grid * sizeof(NuclideGridPoint), q);
    q.memcpy(GSD.nuclide_grid, SD.nuclide_grid,
             SD.length_nuclide_grid * sizeof(NuclideGridPoint)).wait();
    GSD.verification = (unsigned long*)malloc_device(in.lookups * sizeof(unsigned long), q);
    GSD.length_verification = in.lookups;
    return GSD;
}
void release_device_memory(SimulationData& GSD, queue& q) {
    free(GSD.num_nucs, q);
    free(GSD.concs, q);
    free(GSD.mats, q);
    if (GSD.unionized_energy_array) free(GSD.unionized_energy_array, q);
    if (GSD.index_grid) free(GSD.index_grid, q);
    free(GSD.nuclide_grid, q);
    free(GSD.verification, q);
    if (GSD.p_energy_samples) free(GSD.p_energy_samples, q);
    if (GSD.mat_samples) free(GSD.mat_samples, q);
}

// Baseline simulation
// ------------------------------------------------------------------
unsigned long long run_event_based_simulation_baseline(const Inputs& in, SimulationData& GSD, queue& q) {
    const int nthreads = 256;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;
    q.submit([&](handler& h){
        h.parallel_for(nd_range<1>(range<1>(nblocks * nthreads),
                                   range<1>(nthreads)),
                       [=](nd_item<1> it){
            int i = it.get_global_id(0);
            if (i >= in.lookups) return;
            uint64_t seed = STARTING_SEED;
            seed = fast_forward_LCG(seed, 2ULL * i);
            double p_energy = LCG_random_double(&seed);
            int mat = pick_mat(&seed);
            double macro_xs[5] = {0};
            calculate_macro_xs(p_energy, mat,
                               in.n_isotopes, in.n_gridpoints,
                               GSD.num_nucs, GSD.concs,
                               GSD.unionized_energy_array,
                               GSD.index_grid,
                               GSD.nuclide_grid,
                               GSD.mats,
                               macro_xs,
                               in.grid_type, in.hash_bins,
                               GSD.max_num_nucs);
            double maxv = -1.0; int max_idx = 0;
            for (int j=0;j<5;j++) if (macro_xs[j] > maxv){maxv=macro_xs[j]; max_idx=j;}
            GSD.verification[i] = max_idx + 1;
        });
    }).wait();
    std::vector<unsigned long> host_verif(in.lookups);
    q.memcpy(host_verif.data(), GSD.verification, in.lookups * sizeof(unsigned long)).wait();
    unsigned long result = std::accumulate(host_verif.begin(), host_verif.end(), 0UL);
    return result;
}

// Optimization 1 (sampling + lookup)
// ------------------------------------------------------------------
unsigned long long run_event_based_simulation_optimization_1(const Inputs& in, SimulationData& GSD, queue& q) {
    const int nthreads = 32;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;
    // allocate samples
    GSD.p_energy_samples = (double*)malloc_device(in.lookups * sizeof(double), q);
    GSD.mat_samples = (int*)malloc_device(in.lookups * sizeof(int), q);
    // sampling kernel
    q.submit([&](handler& h){
        h.parallel_for(nd_range<1>(range<1>(nblocks*nthreads), range<1>(nthreads)),
                       [=](nd_item<1> it){
            int i = it.get_global_id(0);
            if (i >= in.lookups) return;
            uint64_t seed = STARTING_SEED;
            seed = fast_forward_LCG(seed, 2ULL * i);
            GSD.p_energy_samples[i] = LCG_random_double(&seed);
            GSD.mat_samples[i] = pick_mat(&seed);
        });
    }).wait();
    // lookup kernel
    q.submit([&](handler& h){
        h.parallel_for(nd_range<1>(range<1>(nblocks*nthreads), range<1>(nthreads)),
                       [=](nd_item<1> it){
            int i = it.get_global_id(0);
            if (i >= in.lookups) return;
            double macro_xs[5] = {0};
            calculate_macro_xs(GSD.p_energy_samples[i],
                               GSD.mat_samples[i],
                               in.n_isotopes, in.n_gridpoints,
                               GSD.num_nucs, GSD.concs,
                               GSD.unionized_energy_array,
                               GSD.index_grid,
                               GSD.nuclide_grid,
                               GSD.mats,
                               macro_xs,
                               in.grid_type, in.hash_bins,
                               GSD.max_num_nucs);
            double maxv = -1.0; int max_idx = 0;
            for (int j=0;j<5;j++) if (macro_xs[j] > maxv){maxv=macro_xs[j]; max_idx=j;}
            GSD.verification[i] = max_idx + 1;
        });
    }).wait();
    std::vector<unsigned long> host_verif(in.lookups);
    q.memcpy(host_verif.data(), GSD.verification, in.lookups * sizeof(unsigned long)).wait();
    unsigned long result = std::accumulate(host_verif.begin(), host_verif.end(), 0UL);
    return result;
}

unsigned long long run_event_based_simulation_optimization_2(const Inputs& in, SimulationData& GSD, queue& q) {

unsigned long long run_event_based_simulation_optimization_4(const Inputs& in, SimulationData& GSD, queue& q) {
    const int nthreads = 32;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;
    GSD.p_energy_samples = (double*)malloc_device(in.lookups * sizeof(double), q);
    GSD.mat_samples = (int*)malloc_device(in.lookups * sizeof(int), q);
    // sampling
    q.submit([&](handler& h){
        h.parallel_for(nd_range<1>(range<1>(nblocks*nthreads), range<1>(nthreads)),

    q.memcpy(host_energy.data(), GSD.p_energy_samples, in.lookups * sizeof(double)).wait();
    // count per material
    int lookups_per_mat[12] = {0};
    for (int i=0;i<in.lookups;i++) lookups_per_mat[host_mat[i]]++;
    // sort by material (stable)
    std::vector<std::pair<int,double>> pairs(in.lookups);

                     [](const auto& a, const auto& b){ return a.first < b.first; });
    for (int i=0;i<in.lookups;i++) {

        host_energy[i] = pairs[i].second;
    }
    // copy back sorted data
    q.memcpy(GSD.mat_samples, host_mat.data(), in.lookups * sizeof(int)).wait();
    q.memcpy(GSD.p_energy_samples, host_energy.data(), in.lookups * sizeof(double)).wait();
    // launch per‑material kernels
    int offset = 0;
    for (int m=0; m<12; ++m) {
        int nlook = lookups_per_mat[m];
        if (nlook == 0) continue;
        int nb = (nlook + nthreads - 1) / nthreads;
        q.submit([&, m, nlook, offset](handler& h){
            h.parallel_for(nd_range<1>(range<1>(nb*nthreads), range<1>(nthreads)),
                           [=](nd_item<1> it){
                int i = it.get_global_id(0);
                if (i >= nlook) return;
                int idx = i + offset;
                double macro_xs[5] = {0};
                calculate_macro_xs(GSD.p_energy_samples[idx],
                                   m,
                                   in.n_isotopes, in.n_gridpoints,
                                   GSD.num_nucs, GSD.concs,
                                   GSD.unionized_energy_array,
                                   GSD.index_grid,
                                   GSD.nuclide_grid,
                                   GSD.mats,
                                   macro_xs,
                                   in.grid_type, in.hash_bins,
                                   GSD.max_num_nucs);
                double maxv = -1.0; int max_idx = 0;
                for (int j=0;j<5;j++) if (macro_xs[j] > maxv){maxv=macro_xs[j]; max_idx=j;}
                GSD.verification[idx] = max_idx + 1;
            });
        });
        offset += nlook;
    }
    q.wait();
    std::vector<unsigned long> host_verif(in.lookups);
    q.memcpy(host_verif.data(), GSD.verification, in.lookups * sizeof(unsigned long)).wait();
    unsigned long result = std::accumulate(host_verif.begin(), host_verif.end(), 0UL);

struct is_fuel {
    bool operator()(const int& v) const { return v == 0; }
};
unsigned long long run_event_based_simulation_optimization_5(const Inputs& in, SimulationData& GSD, queue& q) {
    const int nthreads = 32;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;
    GSD.p_energy_samples = (double*)malloc_device(in.lookups * sizeof(double), q);
    GSD.mat_samples = (int*)malloc_device(in.lookups * sizeof(int), q);
    // sampling
    q.submit([&](handler& h){
        h.parallel_for(nd_range<1>(range<1>(nblocks*nthreads), range<1>(nthreads)),
                       [=](nd_item<1> it){
            int i = it.get_global_id(0);
            if (i >= in.lookups) return;
            uint64_t seed = STARTING_SEED;
            seed = fast_forward_LCG(seed, 2ULL * i);
            GSD.p_energy_samples[i] = LCG_random_double(&seed);
            GSD.mat_samples[i] = pick_mat(&seed);
        });
    }).wait();
    // copy to host for partitioning

    std::vector<double> host_energy(in.lookups);
    q.memcpy(host_mat.data(), GSD.mat_samples, in.lookups * sizeof(int)).wait();
    q.memcpy(host_energy.data(), GSD.p_energy_samples, in.lookups * sizeof(double)).wait();
    // count fuel
    int n_fuel = 0;
    for (int v : host_mat) if (v == 0) n_fuel++;
    // stable partition fuel to front
    std::stable_partition(host_mat.begin(), host_mat.end(),
                          [&](int v){ return v == 0; });
    std::stable_partition(host_energy.begin(), host_energy.end(),
                          [&](int idx){ return host_mat[idx] == 0; });
    // copy back partitioned data
    q.memcpy(GSD.mat_samples, host_mat.data(), in.lookups * sizeof(int)).wait();
    q.memcpy(GSD.p_energy_samples, host_energy.data(), in.lookups * sizeof(double)).wait();
    // fuel kernel
    int nb_fuel = (n_fuel + nthreads - 1) / nthreads;
    q.submit([&](handler& h){
        h.parallel_for(nd_range<1>(range<1>(nb_fuel*nthreads), range<1>(nthreads)),
                       [=](nd_item<1> it){
            int i = it.get_global_id(0);
            if (i >= n_fuel) return;
            double macro_xs[5] = {0};
            calculate_macro_xs(GSD.p_energy_samples[i],

                               in.n_isotopes, in.n_gridpoints,
                               GSD.num_nucs, GSD.concs,
                               GSD.unionized_energy_array,
                               GSD.index_grid,
                               GSD.nuclide_grid,
                               GSD.mats,
                               macro_xs,
                               in.grid_type, in.hash_bins,
                               GSD.max_num_nucs);
            double maxv = -1.0; int max_idx = 0;
            for (int j=0;j<5;j++) if (macro_xs[j] > maxv){maxv=macro_xs[j]; max_idx=j;}
            GSD.verification[i] = max_idx + 1;
        });
    });
    // non‑fuel kernel
    int n_non = in.lookups - n_fuel;
    int nb_non = (n_non + nthreads - 1) / nthreads;
    q.submit([&](handler& h){
        h.parallel_for(nd_range<1>(range<1>(nb_non*nthreads), range<1>(nthreads)),
                       [=](nd_item<1> it){
            int i = it.get_global_id(0);
            if (i >= n_non) return;
            int idx = i + n_fuel;
            int mat = GSD.mat_samples[idx];
            double macro_xs[5] = {0};
            calculate_macro_xs(GSD.p_energy_samples[idx],
                               mat,
                               in.n_isotopes, in.n_gridpoints,
                               GSD.num_nucs, GSD.concs,
                               GSD.unionized_energy_array,
                               GSD.index_grid,
                               GSD.nuclide_grid,
                               GSD.mats,
                               macro_xs,
                               in.grid_type, in.hash_bins,
                               GSD.max_num_nucs);
            double maxv = -1.0; int max_idx = 0;
            for (int j=0;j<5;j++) if (macro_xs[j] > maxv){maxv=macro_xs[j]; max_idx=j;}
            GSD.verification[idx] = max_idx + 1;
        });
    }).wait();
    std::vector<unsigned long> host_verif(in.lookups);
    q.memcpy(host_verif.data(), GSD.verification, in.lookups * sizeof(unsigned long)).wait();
    unsigned long result = std::accumulate(host_verif.begin(), host_verif.end(), 0UL);
    return result;
}

// Simple CLI parser (mirrors original behavior)

void print_CLI_error() {
    printf("Usage: ./XSBench <options>\n");
    exit(4);
}
Inputs read_CLI(int argc, char* argv[]) {
    Inputs input{};
    input.simulation_method = HISTORY_BASED;
    input.nthreads = 1;
    input.n_isotopes = 355;
    input.n_gridpoints = 11303;
    input.particles = 500000;
    input.lookups = 34;
    input.grid_type = UNIONIZED;
    input.hash_bins = 10000;
    input.binary_mode = NONE;
    input.kernel_id = 0;
    input.HM = (char*)malloc(6);
    strcpy(input.HM, "large");
    int user_g = 0;
    int default_lookups = 1, default_particles = 1;
    for (int i=1;i<argc;i++) {
        if (strcmp(argv[i], "-g")==0) {
            if (++i<argc){ user_g=1; input.n_gridpoints = atol(argv[i]); }
            else print_CLI_error();
        } else if (strcmp(argv[i], "-m")==0) {
            if (++i<argc){
                if (strcmp(argv[i],"history")==0) input.simulation_method = HISTORY_BASED;
                else if (strcmp(argv[i],"event")==0){
                    input.simulation_method = EVENT_BASED;
                    if (default_lookups && default_particles){
                        input.lookups = input.lookups * input.particles;
                        input.particles = 0;
                    }
                } else print_CLI_error();
            } else print_CLI_error();
        } else if (strcmp(argv[i], "-l")==0) {
            if (++i<argc){ input.lookups = atoi(argv[i]); default_lookups=0; }
            else print_CLI_error();
        } else if (strcmp(argv[i], "-h")==0) {
            if (++i<argc){ input.hash_bins = atoi(argv[i]); }
            else print_CLI_error();
        } else if (strcmp(argv[i], "-p")==0) {
            if (++i<argc){ input.particles = atoi(argv[i]); default_particles=0; }
            else print_CLI_error();
        } else if (strcmp(argv[i], "-s")==0) {
            if (++i<argc){ input.HM = argv[i]; }
            else print_CLI_error();
        } else if (strcmp(argv[i], "-G")==0) {
            if (++i<argc){
                if (strcmp(argv[i],"unionized")==0) input.grid_type = UNIONIZED;
                else if (strcmp(argv[i],"nuclide")==0) input.grid_type = NUCLIDE;
                else if (strcmp(argv[i],"hash")==0) input.grid_type = HASH;
                else print_CLI_error();
            } else print_CLI_error();
        } else if (strcmp(argv[i], "-b")==0) {
            if (++i<argc){
                if (strcmp(argv[i],"read")==0) input.binary_mode = READ;
                else if (strcmp(argv[i],"write")==0) input.binary_mode = WRITE;
                else print_CLI_error();
            } else print_CLI_error();
        } else if (strcmp(argv[i], "-k")==0) {
            if (++i<argc){ input.kernel_id = atoi(argv[i]); }
            else print_CLI_error();
        } else print_CLI_error();
    }
    // Validate
    if (input.nthreads < 1) print_CLI_error();
    if (input.n_isotopes < 1) print_CLI_error();
    if (input.n_gridpoints < 1) print_CLI_error();
    if (input.lookups < 1) print_CLI_error();
    if (input.hash_bins < 1) print_CLI_error();
    if (strcasecmp(input.HM,"small") && strcasecmp(input.HM,"large") &&
        strcasecmp(input.HM,"XL") && strcasecmp(input.HM,"XXL"))
        print_CLI_error();
    if (strcasecmp(input.HM,"small")==0) input.n_isotopes = 68;
    else if (strcasecmp(input.HM,"XL")==0 && !user_g) input.n_gridpoints = 238847;
    else if (strcasecmp(input.HM,"XXL")==0 && !user_g) input.n_gridpoints = static_cast<long>(238847 * 2.1);
    return input;
}

// ------------------------------------------------------------------
// Main
// ------------------------------------------------------------------
int main(int argc, char* argv[]) {
    try {
        // Device selection (default selector, out‑of‑order queue)
        queue q{default_selector_v};
        // Parse CLI
        Inputs in = read_CLI(argc, argv);
        // Data init
        SimulationData SD = grid_init_do_not_profile(in);
        // Transfer to device
        SimulationData GSD = move_simulation_data_to_device(in, SD, q);
        // Run selected kernel
        unsigned long long verification = 0;
        double start = get_time();
        if (in.simulation_method == EVENT_BASED) {
            switch (in.kernel_id) {
                case 0: verification = run_event_based_simulation_baseline(in, GSD, q); break;
                case 1: verification = run_event_based_simulation_optimization_1(in, GSD, q); break;
                case 2: verification = run_event_based_simulation_optimization_2(in, GSD, q); break;
                case 3: verification = run_event_based_simulation_optimization_3(in, GSD, q); break;
                case 4: verification = run_event_based_simulation_optimization_4(in, GSD, q); break;
                case 5: verification = run_event_based_simulation_optimization_5(in, GSD, q); break;
                case 6: verification = run_event_based_simulation_optimization_6(in, GSD, q); break;
                default:
                    printf("Error: No kernel ID %d found!\n", in.kernel_id);
                    return 1;
            }
        } else {
            printf("History‑based simulation not implemented. Use event‑based.\n");
            return 1;
        }
        double end = get_time();
        // Final checksum
        verification %= 999983ULL;
        // Simple result print
        printf("\nRuntime: %.3f s\nLookups: %d\nLookups/s: %d\nChecksum: %llu\n",
               end - start,
               in.lookups,
               static_cast<int>(in.lookups / (end - start)),
               verification);
        // Cleanup
        release_device_memory(GSD, q);
        return 0;
    } catch (exception const& e) {
        std::cerr << "SYCL exception: " << e.what() << std::endl;
        return 1;
    }
}
