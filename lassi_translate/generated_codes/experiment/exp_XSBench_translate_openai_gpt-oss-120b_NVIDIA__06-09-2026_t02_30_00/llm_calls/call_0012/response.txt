constexpr uint64_t STARTING_SEED = 1070ULL;

inline uint64_t fast_forward_LCG(uint64_t seed, uint64_t n) {
    const uint64_t m = 9223372036854775808ULL; // 2^63
    const uint64_t a = 2806196910506780709ULL;
    const uint64_t c = 1ULL;

    n = n % m;
    uint64_t a_new = 1;
    uint64_t c_new = 0;
    uint64_t a_tmp = a;
    uint64_t c_tmp = c;

    while (n > 0) {
        if (n & 1) {
            a_new = a_new * a_tmp;
            c_new = c_new * a_tmp + c_tmp;
        }
        c_tmp = c_tmp * (a_tmp + 1);
        a_tmp = a_tmp * a_tmp;
        n >>= 1;
    }
    return (a_new * seed + c_new) % m;
}

inline double LCG_random_double(uint64_t *seed) {
    const uint64_t m = 9223372036854775808ULL; // 2^63
    const uint64_t a = 2806196910506780709ULL;
    const uint64_t c = 1ULL;
    *seed = (a * (*seed) + c) % m;
    return static_cast<double>(*seed) / static_cast<double>(m);
}

// Utility RNG for material selection (same as CUDA version)

inline int pick_mat(uint64_t *seed) {
    double dist[12] = {0.140, 0.052, 0.275, 0.134, 0.154,
                       0.064, 0.066, 0.055, 0.008, 0.015,
                       0.025, 0.013};
    double roll = LCG_random_double(seed);
    double cum = 0.0;
    for (int i = 0; i < 12; ++i) {
        cum += dist[i];
        if (roll < cum) return i;
    }
    return 0;
}

//--------------------------------------------------------------------------
// Grid search helpers (device callable)
//--------------------------------------------------------------------------
inline long grid_search(long n, double quarry, const double *A) {
    long low = 0, high = n - 1;
    while (high - low > 1) {
        long mid = low + (high - low) / 2;
        if (A[mid] > quarry)
            high = mid;
        else
            low = mid;
    }
    return low;
}

inline long grid_search_nuclide(long n, double quarry,
                                const NuclideGridPoint *A, long low,
                                long high) {
    while (high - low > 1) {
        long mid = low + (high - low) / 2;
        if (A[mid].energy > quarry)
            high = mid;
        else
            low = mid;
    }
    return low;
}

// Micro & Macro XS calculations (device callable)
//--------------------------------------------------------------------------
inline void calculate_micro_xs(double p_energy, int nuc, long n_isotopes,
                               long n_gridpoints, const double *egrid,
                               const int *index_data,
                               const NuclideGridPoint *nuclide_grids, long idx,
                               double *xs_vector, int grid_type,
                               int hash_bins) {
    const NuclideGridPoint *low;
    const NuclideGridPoint *high;
    if (grid_type == 1) { // NUCLIDE
        idx = grid_search_nuclide(n_gridpoints, p_energy,
                                  nuclide_grids + nuc * n_gridpoints, 0,
                                  n_gridpoints - 1);
        low = (idx == n_gridpoints - 1)
                  ? &nuclide_grids[nuc * n_gridpoints + idx - 1]
                  : &nuclide_grids[nuc * n_gridpoints + idx];
    } else if (grid_type == 0) { // UNIONIZED
        int ind = index_data[idx * n_isotopes + nuc];
        low = (ind == n_gridpoints - 1)
                  ? &nuclide_grids[nuc * n_gridpoints + ind - 1]
                  : &nuclide_grids[nuc * n_gridpoints + ind];
    } else { // HASH
        int u_low = index_data[idx * n_isotopes + nuc];
        int u_high = (idx == hash_bins - 1)
                         ? n_gridpoints - 1
                         : index_data[(idx + 1) * n_isotopes + nuc] + 1;
        double e_low = nuclide_grids[nuc * n_gridpoints + u_low].energy;
        double e_high = nuclide_grids[nuc * n_gridpoints + u_high].energy;
        int lower;
        if (p_energy <= e_low)
            lower = 0;
        else if (p_energy >= e_high)
            lower = n_gridpoints - 1;
        else
            lower = grid_search_nuclide(n_gridpoints, p_energy,
                                        nuclide_grids + nuc * n_gridpoints,
                                        u_low, u_high);
        low = (lower == n_gridpoints - 1)
                  ? &nuclide_grids[nuc * n_gridpoints + lower - 1]
                  : &nuclide_grids[nuc * n_gridpoints + lower];
    }
    high = low + 1;
    double f = (high->energy - p_energy) / (high->energy - low->energy);
    xs_vector[0] = high->total_xs - f * (high->total_xs - low->total_xs);
    xs_vector[1] = high->elastic_xs - f * (high->elastic_xs - low->elastic_xs);
    xs_vector[2] = high->absorbtion_xs - f * (high->absorbtion_xs - low->absorbtion_xs);
    xs_vector[3] = high->fission_xs - f * (high->fission_xs - low->fission_xs);
    xs_vector[4] = high->nu_fission_xs - f * (high->nu_fission_xs - low->nu_fission_xs);
}

inline void calculate_macro_xs(double p_energy, int mat, long n_isotopes,
                               long n_gridpoints, const int *num_nucs,
                               const double *concs, const double *egrid,
                               const int *index_data,
                               const NuclideGridPoint *nuclide_grids,
                               const int *mats, double *macro_xs_vector,
                               int grid_type, int hash_bins,
                               int max_num_nucs) {
    for (int k = 0; k < 5; ++k) macro_xs_vector[k] = 0.0;
    long idx = -1;
    if (grid_type == 0) // UNIONIZED
        idx = grid_search(n_isotopes * n_gridpoints, p_energy, egrid);
    else if (grid_type == 2) // HASH
        idx = static_cast<long>(p_energy * hash_bins);
    for (int j = 0; j < num_nucs[mat]; ++j) {
        double xs_vec[5];
        int p_nuc = mats[mat * max_num_nucs + j];
        double conc = concs[mat * max_num_nucs + j];
        calculate_micro_xs(p_energy, p_nuc, n_isotopes, n_gridpoints, egrid,
                           index_data, nuclide_grids, idx, xs_vec, grid_type,
                           hash_bins);
        for (int k = 0; k < 5; ++k) macro_xs_vector[k] += xs_vec[k] * conc;
    }
}

//--------------------------------------------------------------------------
// Baseline simulation (SYCL)
//--------------------------------------------------------------------------
unsigned long long run_event_based_simulation_baseline(const Inputs &in,
                                                       SimulationData &GSD,
                                                       int mype, queue &q) {
    const int nthreads = 256;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;

    q.submit([&](handler &h) {
        h.parallel_for(nd_range<1>(range<1>(nblocks * nthreads),
                                   range<1>(nthreads)),
                       [=](nd_item<1> it) {
                           int i = it.get_global_id(0);
                           if (i >= in.lookups) return;
                           uint64_t seed = STARTING_SEED;
                           seed = fast_forward_LCG(seed, 2ULL * i);
                           double p_energy = LCG_random_double(&seed);
                           int mat = pick_mat(&seed);
                           double macro_xs[5] = {0};
                           calculate_macro_xs(p_energy, mat, in.n_isotopes,
                                              in.n_gridpoints,
                                              GSD.d_num_nucs, GSD.d_concs,
                                              GSD.d_unionized_energy_array,
                                              GSD.d_index_grid,
                                              GSD.d_nuclide_grid, GSD.d_mats,
                                              macro_xs, in.grid_type,
                                              in.hash_bins, GSD.max_num_nucs);
                           double maxv = -1.0;
                           int max_idx = 0;
                           for (int j = 0; j < 5; ++j)
                               if (macro_xs[j] > maxv) {
                                   maxv = macro_xs[j];
                                   max_idx = j;
                               }
                           GSD.d_verification[i] = static_cast<unsigned long>(max_idx + 1);
                       });
    }).wait();

    // Reduce verification on host using std::accumulate
    std::vector<unsigned long> host_verif(in.lookups);
    q.memcpy(host_verif.data(), GSD.d_verification,
             in.lookups * sizeof(unsigned long))
        .wait();
    unsigned long long result = std::accumulate(host_verif.begin(),
                                                host_verif.end(), 0ULL);
    return result;
}

// Optimization 5 (fuel / non‑fuel partition) – SYCL version
//--------------------------------------------------------------------------
struct is_fuel {
    bool operator()(const int &v) const { return v == 0; }
};

unsigned long long run_event_based_simulation_optimization_5(const Inputs &in,
                                                             SimulationData &GSD,
                                                             int mype, queue &q) {
    const int nthreads = 32;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;

    // allocate samples on device
    GSD.d_p_energy_samples = malloc_device<double>(in.lookups, q);
    GSD.d_mat_samples = malloc_device<int>(in.lookups, q);
    GSD.d_verification = malloc_device<unsigned long>(in.lookups, q);
    GSD.length_p_energy_samples = in.lookups;
    GSD.length_mat_samples = in.lookups;
    GSD.length_verification = in.lookups;

    // sampling kernel
    q.submit([&](handler &h) {
        h.parallel_for(nd_range<1>(range<1>(nblocks * nthreads),
                                   range<1>(nthreads)),
                       [=](nd_item<1> it) {
                           int i = it.get_global_id(0);
                           if (i >= in.lookups) return;
                           uint64_t seed = STARTING_SEED;
                           seed = fast_forward_LCG(seed, 2ULL * i);
                           GSD.d_p_energy_samples[i] = LCG_random_double(&seed);
                           GSD.d_mat_samples[i] = pick_mat(&seed);
                       });
    }).wait();

    // copy to host for partition
    std::vector<int> host_mat(in.lookups);
    std::vector<double> host_energy(in.lookups);
    q.memcpy(host_mat.data(), GSD.d_mat_samples,
             in.lookups * sizeof(int))
        .wait();
    q.memcpy(host_energy.data(), GSD.d_p_energy_samples,
             in.lookups * sizeof(double))
        .wait();

    // partition fuel vs non‑fuel on host
    std::vector<std::pair<int, double>> pairs(in.lookups);
    for (int i = 0; i < in.lookups; ++i)
        pairs[i] = {host_mat[i], host_energy[i]};
    auto fuel_end = std::stable_partition(
        pairs.begin(), pairs.end(),
        [](const auto &p) { return p.first == 0; });
    int n_fuel = static_cast<int>(std::distance(pairs.begin(), fuel_end));

    for (int i = 0; i < in.lookups; ++i) {
        host_mat[i] = pairs[i].first;
        host_energy[i] = pairs[i].second;
    }

    q.memcpy(GSD.d_mat_samples, host_mat.data(),
             in.lookups * sizeof(int))
        .wait();
    q.memcpy(GSD.d_p_energy_samples, host_energy.data(),
             in.lookups * sizeof(double))
        .wait();

    // fuel kernel
    int nb_fuel = (n_fuel + nthreads - 1) / nthreads;
    q.submit([&](handler &h) {
        h.parallel_for(nd_range<1>(range<1>(nb_fuel * nthreads),
                                   range<1>(nthreads)),
                       [=](nd_item<1> it) {
                           int i = it.get_global_id(0);
                           if (i >= n_fuel) return;
                           double macro_xs[5] = {0};
                           calculate_macro_xs(
                               GSD.d_p_energy_samples[i], 0, in.n_isotopes,
                               in.n_gridpoints, GSD.d_num_nucs, GSD.d_concs,
                               GSD.d_unionized_energy_array,
                               GSD.d_index_grid, GSD.d_nuclide_grid,
                               GSD.d_mats, macro_xs, in.grid_type,
                               in.hash_bins, GSD.max_num_nucs);
                           double maxv = -1.0;
                           int max_idx = 0;
                           for (int j = 0; j < 5; ++j)
                               if (macro_xs[j] > maxv) {
                                   maxv = macro_xs[j];
                                   max_idx = j;
                               }
                           GSD.d_verification[i] =
                               static_cast<unsigned long>(max_idx + 1);
                       });
    });

    // non‑fuel kernel
    int n_non = in.lookups - n_fuel;
    int nb_non = (n_non + nthreads - 1) / nthreads;
    q.submit([&](handler &h) {
        h.parallel_for(nd_range<1>(range<1>(nb_non * nthreads),
                                   range<1>(nthreads)),
                       [=](nd_item<1> it) {
                           int i = it.get_global_id(0);
                           if (i >= n_non) return;
                           int idx = i + n_fuel;
                           int mat = GSD.d_mat_samples[idx];
                           double macro_xs[5] = {0};
                           calculate_macro_xs(
                               GSD.d_p_energy_samples[idx], mat,
                               in.n_isotopes, in.n_gridpoints,
                               GSD.d_num_nucs, GSD.d_concs,
                               GSD.d_unionized_energy_array,
                               GSD.d_index_grid, GSD.d_nuclide_grid,
                               GSD.d_mats, macro_xs, in.grid_type,
                               in.hash_bins, GSD.max_num_nucs);
                           double maxv = -1.0;
                           int max_idx = 0;
                           for (int j = 0; j < 5; ++j)
                               if (macro_xs[j] > maxv) {
                                   maxv = macro_xs[j];
                                   max_idx = j;
                               }
                           GSD.d_verification[idx] =
                               static_cast<unsigned long>(max_idx + 1);
                       });
    }).wait();

    // reduction
    std::vector<unsigned long> host_verif(in.lookups);
    q.memcpy(host_verif.data(), GSD.d_verification,
             in.lookups * sizeof(unsigned long))
        .wait();
    unsigned long long result = std::accumulate(host_verif.begin(),
                                                host_verif.end(), 0ULL);
    return result;
}

//--------------------------------------------------------------------------
// Grid initialization (host only, then moved to device)
//--------------------------------------------------------------------------
SimulationData grid_init_do_not_profile(const Inputs &in, int mype) {
    SimulationData SD;
    uint64_t seed = 42ULL;

    // Nuclide grid
    SD.length_nuclide_grid = in.n_isotopes * in.n_gridpoints;
    SD.nuclide_grid = (NuclideGridPoint *)malloc(
        SD.length_nuclide_grid * sizeof(NuclideGridPoint));
    for (long i = 0; i < SD.length_nuclide_grid; ++i) {
        SD.nuclide_grid[i].energy = LCG_random_double(&seed);
        SD.nuclide_grid[i].total_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].elastic_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].absorbtion_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].fission_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].nu_fission_xs = LCG_random_double(&seed);
    }
    for (long iso = 0; iso < in.n_isotopes; ++iso) {
        qsort(&SD.nuclide_grid[iso * in.n_gridpoints],
              in.n_gridpoints, sizeof(NuclideGridPoint), NGP_compare);
    }

    // Unionized or hash grid
    if (in.grid_type == 0) { // UNIONIZED
        SD.length_unionized_energy_array = in.n_isotopes * in.n_gridpoints;
        SD.unionized_energy_array = (double *)malloc(
            SD.length_unionized_energy_array * sizeof(double));
        for (long i = 0; i < SD.length_unionized_energy_array; ++i)
            SD.unionized_energy_array[i] = SD.nuclide_grid[i].energy;
        qsort(SD.unionized_energy_array,
              SD.length_unionized_energy_array, sizeof(double),
              double_compare);
        SD.length_index_grid = SD.length_unionized_energy_array * in.n_isotopes;
        SD.index_grid = (int *)malloc(SD.length_index_grid * sizeof(int));
        int *idx_low = (int *)calloc(in.n_isotopes, sizeof(int));
        double *energy_high = (double *)malloc(in.n_isotopes * sizeof(double));
        for (int i = 0; i < in.n_isotopes; ++i)
            energy_high[i] = SD.nuclide_grid[i * in.n_gridpoints + 1].energy;
        for (long e = 0; e < SD.length_unionized_energy_array; ++e) {
            double ue = SD.unionized_energy_array[e];
            for (long i = 0; i < in.n_isotopes; ++i) {
                if (ue < energy_high[i])
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                else if (idx_low[i] == in.n_gridpoints - 2)
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                else {
                    ++idx_low[i];
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                    energy_high[i] = SD.nuclide_grid[i * in.n_gridpoints +
                                                    idx_low[i] + 1]
                                         .energy;
                }
            }
        }
        free(idx_low);
        free(energy_high);
    } else if (in.grid_type == 2) { // HASH
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = in.hash_bins * in.n_isotopes;
        SD.index_grid = (int *)malloc(SD.length_index_grid * sizeof(int));
        double du = 1.0 / in.hash_bins;
        for (long e = 0; e < in.hash_bins; ++e) {
            double energy = e * du;
            for (long i = 0; i < in.n_isotopes; ++i) {
                SD.index_grid[e * in.n_isotopes + i] =
                    grid_search_nuclide(in.n_gridpoints, energy,
                                        SD.nuclide_grid + i * in.n_gridpoints,
                                        0, in.n_gridpoints - 1);
            }
        }
    } else { // NUCLIDE
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = 0;
    }

    // Materials
    SD.num_nucs = load_num_nucs(in.n_isotopes);
    SD.length_num_nucs = 12;
    SD.mats = load_mats(SD.num_nucs, in.n_isotopes, &SD.max_num_nucs);
    SD.length_mats = SD.length_num_nucs * SD.max_num_nucs;
    SD.concs = load_concs(SD.num_nucs, SD.max_num_nucs);
    SD.length_concs = SD.length_mats;

    return SD;
}

//--------------------------------------------------------------------------
// Move data to device (SYCL)
//--------------------------------------------------------------------------
SimulationData move_simulation_data_to_device(const Inputs &in, int mype,
                                             const SimulationData &SD,
                                             queue &q) {
    SimulationData GSD = SD;
    // allocate and copy each array
    GSD.d_num_nucs = malloc_device<int>(SD.length_num_nucs, q);
    q.memcpy(GSD.d_num_nucs, SD.num_nucs,
             SD.length_num_nucs * sizeof(int))
        .wait();

    GSD.d_concs = malloc_device<double>(SD.length_concs, q);
    q.memcpy(GSD.d_concs, SD.concs,
             SD.length_concs * sizeof(double))
        .wait();

    GSD.d_mats = malloc_device<int>(SD.length_mats, q);
    q.memcpy(GSD.d_mats, SD.mats,
             SD.length_mats * sizeof(int))
        .wait();

    GSD.d_unionized_energy_array =
        malloc_device<double>(SD.length_unionized_energy_array, q);
    q.memcpy(GSD.d_unionized_energy_array, SD.unionized_energy_array,
             SD.length_unionized_energy_array * sizeof(double))
        .wait();

    GSD.d_index_grid = malloc_device<int>(SD.length_index_grid, q);
    q.memcpy(GSD.d_index_grid, SD.index_grid,
             SD.length_index_grid * sizeof(int))
        .wait();

    GSD.d_nuclide_grid =
        malloc_device<NuclideGridPoint>(SD.length_nuclide_grid, q);
    q.memcpy(GSD.d_nuclide_grid, SD.nuclide_grid,
             SD.length_nuclide_grid * sizeof(NuclideGridPoint))
        .wait();

    GSD.d_verification = malloc_device<unsigned long>(in.lookups, q);
    GSD.length_verification = in.lookups;

    return GSD;
}

//--------------------------------------------------------------------------
// Release device memory
//--------------------------------------------------------------------------
void release_device_memory(SimulationData &GSD, queue &q) {
    free(GSD.d_num_nucs, q);
    free(GSD.d_concs, q);
    free(GSD.d_mats, q);
    free(GSD.d_unionized_energy_array, q);
    free(GSD.d_index_grid, q);
    free(GSD.d_nuclide_grid, q);
    free(GSD.d_verification, q);
    if (GSD.d_p_energy_samples) free(GSD.d_p_energy_samples, q);

//--------------------------------------------------------------------------

int NGP_compare(const void *a, const void *b) {
    const NuclideGridPoint *A = (const NuclideGridPoint *)a;
    const NuclideGridPoint *B = (const NuclideGridPoint *)b;
    if (A->energy > B->energy) return 1;
    if (A->energy < B->energy) return -1;
    return 0;
}
int double_compare(const void *a, const void *b) {
    double A = *(const double *)a;
    double B = *(const double *)b;
    if (A > B) return 1;
    if (A < B) return -1;
    return 0;
}

//--------------------------------------------------------------------------
// Memory usage estimate (unused in SYCL version but kept for completeness)
//--------------------------------------------------------------------------
size_t estimate_mem_usage(const Inputs &in) {
    size_t single_nuclide_grid = in.n_gridpoints * sizeof(NuclideGridPoint);
    size_t all_nuclide_grids = in.n_isotopes * single_nuclide_grid;
    size_t size_UEG = in.n_isotopes * in.n_gridpoints * sizeof(double) +
                     in.n_isotopes * in.n_gridpoints * in.n_isotopes *
                         sizeof(int);
    size_t size_hash_grid = in.hash_bins * in.n_isotopes * sizeof(int);
    size_t memtotal = 0;
    if (in.grid_type == 0)
        memtotal = all_nuclide_grids + size_UEG;
    else if (in.grid_type == 1)
        memtotal = all_nuclide_grids;
    else
        memtotal = all_nuclide_grids + size_hash_grid;
    return static_cast<size_t>(std::ceil(memtotal / (1024.0 * 1024.0)));
}

void center_print(const char *s, int width) {
    int length = strlen(s);
    for (int i = 0; i <= (width - length) / 2; ++i) fputs(" ", stdout);
    fputs(s, stdout);
    fputs("\n", stdout);
}
void fancy_int(long a) {
    if (a < 1000)
        printf("%ld\n", a);
    else if (a < 1000000)
        printf("%ld,%03ld\n", a / 1000, a % 1000);
    else if (a < 1000000000)
        printf("%ld,%03ld,%03ld\n", a / 1000000,
               (a % 1000000) / 1000, a % 1000);
    else
        printf("%ld,%03ld,%03ld,%03ld\n",
               a / 1000000000,
               (a % 1000000000) / 1000000,
               (a % 1000000) / 1000,
               a % 1000);
}
void print_results(const Inputs &in, int mype, double runtime, int nprocs,
                   unsigned long long vhash) {
    int lookups = (in.simulation_method == 1) ? in.lookups * in.particles
                                              : in.lookups;
    int lookups_per_sec = (int)((double)lookups / runtime);
    int is_invalid_result = 1;
    if (mype == 0) {
        border_print();
        center_print("RESULTS", 79);
        border_print();
        printf("NOTE: Timings are estimated -- use nvprof/nsys/iprof/rocprof for formal analysis\n");
        printf("Runtime: %.3lf seconds\n", runtime);
        printf("Lookups: "); fancy_int(lookups);
        printf("Lookups/s: "); fancy_int(lookups_per_sec);
    }
    unsigned long long large = 952131, small = 945990;
    if (strcmp(in.HM, "large") == 0) {
        if (vhash == large) is_invalid_result = 0;
    } else if (strcmp(in.HM, "small") == 0) {
        if (vhash == small) is_invalid_result = 0;
    }
    if (mype == 0) {
        if (is_invalid_result)
            printf("Verification checksum: %llu (WARNING - INVALID CHECKSUM!)\n",
                   vhash);
        else
            printf("Verification checksum: %llu (Valid)\n", vhash);
        border_print();
    }
}

int main(int argc, char *argv[]) {
    queue q{default_selector{}};

    Inputs in = read_CLI(argc, argv);
    SimulationData SD = grid_init_do_not_profile(in, 0);
    SimulationData GSD = move_simulation_data_to_device(in, 0, SD, q);

    double start = get_time();
    unsigned long long verification = 0;
    if (in.kernel_id == 0)
        verification = run_event_based_simulation_baseline(in, GSD, 0, q);
    else if (in.kernel_id == 5)
        verification = run_event_based_simulation_optimization_5(in, GSD, 0, q);
    else {
        printf("Unsupported kernel_id %d in SYCL port\n", in.kernel_id);
        return 1;
    }
    double end = get_time();

    release_device_memory(GSD, q);
    verification = verification % 999983;
    print_results(in, 0, end - start, 1, verification);
    return 0;
}
