// ---------- RNG ----------
inline double LCG_random_double(uint64_t *seed) {
    const uint64_t m = (1ULL << 63);
    const uint64_t a = 2806196910506780709ULL;
    const uint64_t c = 1ULL;
    *seed = (a * (*seed) + c) % m;
    return static_cast<double>(*seed) / static_cast<double>(m);
}
inline uint64_t fast_forward_LCG(uint64_t seed, uint64_t n) {
    const uint64_t m = (1ULL << 63);
    uint64_t a = 2806196910506780709ULL;
    uint64_t c = 1ULL;
    n %= m;
    uint64_t a_new = 1;
    uint64_t c_new = 0;
    while (n > 0) {
        if (n & 1) {
            a_new *= a;
            c_new = c_new * a + c;
        }
        c *= (a + 1);
        a *= a;
        n >>= 1;
    }
    return (a_new * seed + c_new) % m;
}
inline int pick_mat(uint64_t *seed) {
    const double dist[12] = {0.140,0.052,0.275,0.134,0.154,0.064,0.066,0.055,0.008,0.015,0.025,0.013};
    double roll = LCG_random_double(seed);
    double running = 0.0;
    for (int i = 0; i < 12; ++i) {
        running += dist[i];
        if (roll < running) return i;
    }
    return 0;
}

// ---------- Search ----------
inline long grid_search(long n, double quarry, const double *A) {
    long low = 0, high = n - 1;
    while (high - low > 1) {
        long mid = low + (high - low) / 2;
        if (A[mid] > quarry) high = mid;
        else low = mid;
    }
    return low;
}
inline long grid_search_nuclide(long n, double quarry,
                               const NuclideGridPoint *A,
                               long low, long high) {
    while (high - low > 1) {
        long mid = low + (high - low) / 2;
        if (A[mid].energy > quarry) high = mid;
        else low = mid;
    }
    return low;
}

// ---------- XS Calculations ----------
inline void calculate_micro_xs(double p_energy, int nuc, long n_isotopes,
                               long n_gridpoints,
                               const double *egrid, const int *index_data,
                               const NuclideGridPoint *nuclide_grids,
                               long idx, double *xs_vector,
                               int grid_type, int hash_bins) {
    const NuclideGridPoint *low;
    const NuclideGridPoint *high;
    if (grid_type == NUCLIDE) {
        idx = grid_search_nuclide(n_gridpoints, p_energy,
                                  &nuclide_grids[nuc * n_gridpoints], 0, n_gridpoints - 1);
        low = (idx == n_gridpoints - 1) ?
              &nuclide_grids[nuc * n_gridpoints + idx - 1] :
              &nuclide_grids[nuc * n_gridpoints + idx];
    } else if (grid_type == UNIONIZED) {
        int ind = index_data[idx * n_isotopes + nuc];
        low = (ind == n_gridpoints - 1) ?
              &nuclide_grids[nuc * n_gridpoints + ind - 1] :
              &nuclide_grids[nuc * n_gridpoints + ind];
    } else { // HASH
        int u_low = index_data[idx * n_isotopes + nuc];
        int u_high = (idx == hash_bins - 1) ?
                     n_gridpoints - 1 :
                     index_data[(idx + 1) * n_isotopes + nuc] + 1;
        double e_low = nuclide_grids[nuc * n_gridpoints + u_low].energy;
        double e_high = nuclide_grids[nuc * n_gridpoints + u_high].energy;
        int lower;
        if (p_energy <= e_low) lower = 0;
        else if (p_energy >= e_high) lower = n_gridpoints - 1;
        else lower = grid_search_nuclide(n_gridpoints, p_energy,
                                         &nuclide_grids[nuc * n_gridpoints],
                                         u_low, u_high);
        low = (lower == n_gridpoints - 1) ?
              &nuclide_grids[nuc * n_gridpoints + lower - 1] :
              &nuclide_grids[nuc * n_gridpoints + lower];
    }
    high = low + 1;
    double f = (high->energy - p_energy) / (high->energy - low->energy);
    xs_vector[0] = high->total_xs - f * (high->total_xs - low->total_xs);
    xs_vector[1] = high->elastic_xs - f * (high->elastic_xs - low->elastic_xs);
    xs_vector[2] = high->absorbtion_xs - f * (high->absorbtion_xs - low->absorbtion_xs);
    xs_vector[3] = high->fission_xs - f * (high->fission_xs - low->fission_xs);
    xs_vector[4] = high->nu_fission_xs - f * (high->nu_fission_xs - low->nu_fission_xs);
}
inline void calculate_macro_xs(double p_energy, int mat,
                               long n_isotopes, long n_gridpoints,
                               const int *num_nucs, const double *concs,
                               const double *egrid, const int *index_data,
                               const NuclideGridPoint *nuclide_grids,
                               const int *mats, double *macro_xs_vector,
                               int grid_type, int hash_bins, int max_num_nucs) {
    for (int k = 0; k < 5; ++k) macro_xs_vector[k] = 0.0;
    long idx = -1;
    if (grid_type == UNIONIZED) idx = grid_search(n_isotopes * n_gridpoints, p_energy, egrid);
    else if (grid_type == HASH) {
        double du = 1.0 / hash_bins;
        idx = static_cast<long>(p_energy / du);
    }
    for (int j = 0; j < num_nucs[mat]; ++j) {
        double xs_vec[5];
        int p_nuc = mats[mat * max_num_nucs + j];
        double conc = concs[mat * max_num_nucs + j];
        calculate_micro_xs(p_energy, p_nuc, n_isotopes, n_gridpoints,
                           egrid, index_data, nuclide_grids,
                           idx, xs_vec, grid_type, hash_bins);
        for (int k = 0; k < 5; ++k) macro_xs_vector[k] += xs_vec[k] * conc;
    }
}

// ---------- Comparison ----------
int NGP_compare(const void *a, const void *b) {
    const NuclideGridPoint *A = static_cast<const NuclideGridPoint *>(a);
    const NuclideGridPoint *B = static_cast<const NuclideGridPoint *>(b);
    return (A->energy > B->energy) - (A->energy < B->energy);
}
int double_compare(const void *a, const void *b) {
    double A = *static_cast<const double *>(a);
    double B = *static_cast<const double *>(b);
    return (A > B) - (A < B);
}

double *load_concs(int *num_nucs, int max_num_nucs) {
    uint64_t seed = STARTING_SEED * STARTING_SEED;
    double *concs = (double *)malloc(12 * max_num_nucs * sizeof(double));
    for (int i = 0; i < 12; ++i)
        for (int j = 0; j < num_nucs[i]; ++j)
            concs[i * max_num_nucs + j] = LCG_random_double(&seed);
    return concs;
}

// ---------- Grid Init ----------
SimulationData grid_init_do_not_profile(const Inputs &in) {
    SimulationData SD{};
    size_t nbytes = 0;
    uint64_t seed = 42;
    // Nuclide grid
    SD.length_nuclide_grid = static_cast<int>(in.n_isotopes * in.n_gridpoints);
    SD.nuclide_grid = (NuclideGridPoint *)malloc(SD.length_nuclide_grid * sizeof(NuclideGridPoint));
    for (int i = 0; i < SD.length_nuclide_grid; ++i) {
        SD.nuclide_grid[i].energy = LCG_random_double(&seed);
        SD.nuclide_grid[i].total_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].elastic_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].absorbtion_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].fission_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].nu_fission_xs = LCG_random_double(&seed);
    }
    for (int i = 0; i < in.n_isotopes; ++i)
        qsort(&SD.nuclide_grid[i * in.n_gridpoints], in.n_gridpoints,
              sizeof(NuclideGridPoint), NGP_compare);
    // Acceleration structures
    if (in.grid_type == UNIONIZED) {
        SD.length_unionized_energy_array = static_cast<int>(in.n_isotopes * in.n_gridpoints);
        SD.unionized_energy_array = (double *)malloc(SD.length_unionized_energy_array * sizeof(double));
        for (int i = 0; i < SD.length_unionized_energy_array; ++i)
            SD.unionized_energy_array[i] = SD.nuclide_grid[i].energy;
        qsort(SD.unionized_energy_array, SD.length_unionized_energy_array,
              sizeof(double), double_compare);
        SD.length_index_grid = SD.length_unionized_energy_array * in.n_isotopes;
        SD.index_grid = (int *)malloc(SD.length_index_grid * sizeof(int));
        int *idx_low = (int *)calloc(in.n_isotopes, sizeof(int));
        double *energy_high = (double *)malloc(in.n_isotopes * sizeof(double));
        for (int i = 0; i < in.n_isotopes; ++i)
            energy_high[i] = SD.nuclide_grid[i * in.n_gridpoints + 1].energy;
        for (long e = 0; e < SD.length_unionized_energy_array; ++e) {
            double ue = SD.unionized_energy_array[e];
            for (long i = 0; i < in.n_isotopes; ++i) {
                if (ue < energy_high[i])
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                else if (idx_low[i] == in.n_gridpoints - 2)
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                else {
                    idx_low[i]++;
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                    energy_high[i] = SD.nuclide_grid[i * in.n_gridpoints + idx_low[i] + 1].energy;
                }
            }
        }
        free(idx_low);
        free(energy_high);
    } else if (in.grid_type == HASH) {
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = static_cast<long>(in.hash_bins * in.n_isotopes);
        SD.index_grid = (int *)malloc(SD.length_index_grid * sizeof(int));
        double du = 1.0 / in.hash_bins;
        for (long e = 0; e < in.hash_bins; ++e) {
            double energy = e * du;
            for (long i = 0; i < in.n_isotopes; ++i) {
                SD.index_grid[e * in.n_isotopes + i] =
                    grid_search_nuclide(in.n_gridpoints, energy,
                                        SD.nuclide_grid + i * in.n_gridpoints, 0,
                                        in.n_gridpoints - 1);
            }
        }
    } else {
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = 0;
    }
    // Materials
    SD.num_nucs = load_num_nucs(in.n_isotopes);
    SD.length_num_nucs = 12;
    SD.mats = load_mats(SD.num_nucs, in.n_isotopes, &SD.max_num_nucs);
    SD.length_mats = SD.length_num_nucs * SD.max_num_nucs;
    SD.concs = load_concs(SD.num_nucs, SD.max_num_nucs);
    SD.length_concs = SD.length_mats;
    return SD;
}

// ---------- Device Transfer ----------
SimulationData move_simulation_data_to_device(sycl::queue &q, const Inputs &in,
                                             const SimulationData &SD) {
    SimulationData GSD = SD;
    GSD.num_nucs = sycl::malloc_device<int>(SD.length_num_nucs, q);
    q.memcpy(GSD.num_nucs, SD.num_nucs, SD.length_num_nucs * sizeof(int)).wait();
    GSD.concs = sycl::malloc_device<double>(SD.length_concs, q);
    q.memcpy(GSD.concs, SD.concs, SD.length_concs * sizeof(double)).wait();
    GSD.mats = sycl::malloc_device<int>(SD.length_mats, q);
    q.memcpy(GSD.mats, SD.mats, SD.length_mats * sizeof(int)).wait();
    GSD.unionized_energy_array = sycl::malloc_device<double>(SD.length_unionized_energy_array, q);
    q.memcpy(GSD.unionized_energy_array, SD.unionized_energy_array,
             SD.length_unionized_energy_array * sizeof(double)).wait();
    GSD.index_grid = sycl::malloc_device<int>(SD.length_index_grid, q);
    q.memcpy(GSD.index_grid, SD.index_grid,
             SD.length_index_grid * sizeof(int)).wait();
    GSD.nuclide_grid = sycl::malloc_device<NuclideGridPoint>(SD.length_nuclide_grid, q);
    q.memcpy(GSD.nuclide_grid, SD.nuclide_grid,
             SD.length_nuclide_grid * sizeof(NuclideGridPoint)).wait();
    GSD.verification = sycl::malloc_device<unsigned long>(in.lookups, q);
    GSD.length_verification = in.lookups;
    return GSD;
}
void release_device_memory(sycl::queue &q, SimulationData &GSD) {
    q.free(GSD.num_nucs);
    q.free(GSD.concs);
    q.free(GSD.mats);
    q.free(GSD.unionized_energy_array);
    q.free(GSD.index_grid);
    q.free(GSD.nuclide_grid);
    q.free(GSD.verification);
    if (GSD.p_energy_samples) q.free(GSD.p_energy_samples);
    if (GSD.mat_samples) q.free(GSD.mat_samples);
}

// ---------- Baseline Simulation ----------
unsigned long long run_event_based_simulation_baseline(sycl::queue &q,
                                                      const Inputs &in,
                                                      SimulationData &GSD) {
    const int nthreads = 256;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;
    q.submit([&](sycl::handler &cgh) {
        cgh.parallel_for<class baseline_kernel>(sycl::nd_range<1>(sycl::range<1>(nblocks * nthreads),
                                                                sycl::range<1>(nthreads)),
            [=](sycl::nd_item<1> item) {
                int i = item.get_global_id(0);
                if (i >= in.lookups) return;
                uint64_t seed = STARTING_SEED;
                seed = fast_forward_LCG(seed, 2ULL * i);
                double p_energy = LCG_random_double(&seed);
                int mat = pick_mat(&seed);
                double macro_xs[5] = {0};
                calculate_macro_xs(p_energy, mat, in.n_isotopes, in.n_gridpoints,
                                   GSD.num_nucs, GSD.concs,
                                   GSD.unionized_energy_array, GSD.index_grid,
                                   GSD.nuclide_grid, GSD.mats,
                                   macro_xs, in.grid_type, in.hash_bins,
                                   GSD.max_num_nucs);
                double max = -1.0;
                int max_idx = 0;
                for (int j = 0; j < 5; ++j)
                    if (macro_xs[j] > max) { max = macro_xs[j]; max_idx = j; }
                GSD.verification[i] = static_cast<unsigned long>(max_idx + 1);
            });
    }).wait();
    // reduction
    auto exec = oneapi::dpl::execution::make_device_policy(q);
    unsigned long result = oneapi::dpl::reduce(exec,
        GSD.verification, GSD.verification + in.lookups, 0ul);
    return result;
}

// ---------- Optimization 1 ----------
unsigned long long run_event_based_simulation_optimization_1(sycl::queue &q,
                                                            const Inputs &in,
                                                            SimulationData &GSD) {
    // allocate extra buffers
    GSD.p_energy_samples = sycl::malloc_device<double>(in.lookups, q);
    GSD.mat_samples = sycl::malloc_device<int>(in.lookups, q);
    const int nthreads = 32;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;
    // sampling kernel
    q.submit([&](sycl::handler &cgh) {
        cgh.parallel_for<class sample1>(sycl::nd_range<1>(sycl::range<1>(nblocks * nthreads),
                                                         sycl::range<1>(nthreads)),

// ---------- Optimization 4 ----------
unsigned long long run_event_based_simulation_optimization_4(sycl::queue &q,
                                                            const Inputs &in,
                                                            SimulationData &GSD) {
    GSD.p_energy_samples = sycl::malloc_device<double>(in.lookups, q);
    GSD.mat_samples = sycl::malloc_device<int>(in.lookups, q);
    const int nthreads = 32;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;
    // sampling
    q.submit([&](sycl::handler &cgh) {
        cgh.parallel_for<class sample4>(sycl::nd_range<1>(sycl::range<1>(nblocks * nthreads),
                                                         sycl::range<1>(nthreads)),
            [=](sycl::nd_item<1> item) {
                int i = item.get_global_id(0);
                if (i >= in.lookups) return;
                uint64_t seed = STARTING_SEED;
                seed = fast_forward_LCG(seed, 2ULL * i);
                GSD.p_energy_samples[i] = LCG_random_double(&seed);
                GSD.mat_samples[i] = pick_mat(&seed);
            });
    }).wait();
    // count per material
    int lookups_per_mat[12];
    for (int m = 0; m < 12; ++m) {
        lookups_per_mat[m] = oneapi::dpl::count(oneapi::dpl::execution::make_device_policy(q),
                                                GSD.mat_samples,
                                                GSD.mat_samples + in.lookups, m);
    }
    // sort by material (key = mat, value = energy)
    oneapi::dpl::sort_by_key(oneapi::dpl::execution::make_device_policy(q),
                             GSD.mat_samples,
                             GSD.mat_samples + in.lookups,
                             GSD.p_energy_samples);
    // launch per‑material kernels
    int offset = 0;
    for (int m = 0; m < 12; ++m) {
        int nlook = lookups_per_mat[m];
        if (nlook == 0) continue;
        int nb = (nlook + nthreads - 1) / nthreads;
        q.submit([&, m, nlook, offset](sycl::handler &cgh) {
            cgh.parallel_for<class lookup4>(sycl::nd_range<1>(sycl::range<1>(nb * nthreads),
                                                             sycl::range<1>(nthreads)),
                [=](sycl::nd_item<1> item) {
                    int i = item.get_global_id(0);
                    if (i >= nlook) return;
                    i += offset;
                    int mat = GSD.mat_samples[i];
                    if (mat != m) return;
                    double macro_xs[5] = {0};
                    calculate_macro_xs(GSD.p_energy_samples[i], mat,
                                       in.n_isotopes, in.n_gridpoints,
                                       GSD.num_nucs, GSD.concs,
                                       GSD.unionized_energy_array, GSD.index_grid,
                                       GSD.nuclide_grid, GSD.mats,
                                       macro_xs, in.grid_type, in.hash_bins,
                                       GSD.max_num_nucs);
                    double max = -1.0;
                    int max_idx = 0;
                    for (int j = 0; j < 5; ++j)
                        if (macro_xs[j] > max) { max = macro_xs[j]; max_idx = j; }
                    GSD.verification[i] = static_cast<unsigned long>(max_idx + 1);
                });
        }).wait();
        offset += nlook;
    }
    auto exec = oneapi::dpl::execution::make_device_policy(q);
    unsigned long result = oneapi::dpl::reduce(exec,
        GSD.verification, GSD.verification + in.lookups, 0ul);
    return result;
}

struct is_fuel {
    bool operator()(const int &a) const { return a == 0; }
};
unsigned long long run_event_based_simulation_optimization_5(sycl::queue &q,
                                                            const Inputs &in,
                                                            SimulationData &GSD) {
    GSD.p_energy_samples = sycl::malloc_device<double>(in.lookups, q);
    GSD.mat_samples = sycl::malloc_device<int>(in.lookups, q);
    const int nthreads = 32;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;

    q.submit([&](sycl::handler &cgh) {

    oneapi::dpl::partition(oneapi::dpl::execution::make_device_policy(q),
                           GSD.mat_samples,
                           GSD.mat_samples + in.lookups,

                                                             sycl::range<1>(nthreads)),

                for (int j = 0; j < 5; ++j)
                    if (macro_xs[j] > max) { max = macro_xs[j]; max_idx = j; }
                GSD.verification[i] = static_cast<unsigned long>(max_idx + 1);
            });
    }).wait();
    // non‑fuel kernel
    int n_nonfuel = in.lookups - n_fuel;
    int nb_non = (n_nonfuel + nthreads - 1) / nthreads;
    q.submit([&](sycl::handler &cgh) {
        cgh.parallel_for<class lookup5_nonfuel>(sycl::nd_range<1>(sycl::range<1>(nb_non * nthreads),
                                                                sycl::range<1>(nthreads)),
            [=](sycl::nd_item<1> item) {
                int i = item.get_global_id(0);
                if (i >= n_nonfuel) return;
                i += n_fuel;
                double macro_xs[5] = {0};
                calculate_macro_xs(GSD.p_energy_samples[i], GSD.mat_samples[i],
                                   in.n_isotopes, in.n_gridpoints,
                                   GSD.num_nucs, GSD.concs,
                                   GSD.unionized_energy_array, GSD.index_grid,
                                   GSD.nuclide_grid, GSD.mats,
                                   macro_xs, in.grid_type, in.hash_bins,
                                   GSD.max_num_nucs);
                double max = -1.0;
                int max_idx = 0;
                for (int j = 0; j < 5; ++j)

                GSD.verification[i] = static_cast<unsigned long>(max_idx + 1);
            });
    }).wait();
    auto exec = oneapi::dpl::execution::make_device_policy(q);
    unsigned long result = oneapi::dpl::reduce(exec,
        GSD.verification, GSD.verification + in.lookups, 0ul);

}

void fancy_int(long a) {
    if (a < 1000) printf("%ld\n", a);
    else if (a < 1000000) printf("%ld,%03ld\n", a / 1000, a % 1000);
    else if (a < 1000000000) printf("%ld,%03ld,%03ld\n",
                                    a / 1000000, (a % 1000000) / 1000, a % 1000);
    else printf("%ld,%03ld,%03ld,%03ld\n",
                a / 1000000000,
                (a % 1000000000) / 1000000,
                (a % 1000000) / 1000,
                a % 1000);
}

Inputs read_CLI(int argc, char *argv[]) {
    Inputs input{};
    input.simulation_method = HISTORY_BASED;
    input.nthreads = 1;
    input.n_isotopes = 355;
    input.n_gridpoints = 11303;
    input.particles = 500000;
    input.lookups = 34;
    input.grid_type = UNIONIZED;
    input.hash_bins = 10000;
    input.binary_mode = NONE;
    input.kernel_id = 0;
    input.HM = (char *)malloc(6);
    strcpy(input.HM, "large");
    int user_g = 0;
    int default_lookups = 1, default_particles = 1;
    for (int i = 1; i < argc; ++i) {
        const char *arg = argv[i];
        if (!strcmp(arg, "-g")) {
            if (++i < argc) { user_g = 1; input.n_gridpoints = atol(argv[i]); }
            else print_CLI_error();
        } else if (!strcmp(arg, "-m")) {
            if (++i < argc) {
                if (!strcmp(argv[i], "history")) input.simulation_method = HISTORY_BASED;
                else if (!strcmp(argv[i], "event")) {
                    input.simulation_method = EVENT_BASED;
                    if (default_lookups && default_particles) {
                        input.lookups = input.lookups * input.particles;
                        input.particles = 0;
                    }
                } else print_CLI_error();
            } else print_CLI_error();
        } else if (!strcmp(arg, "-l")) {
            if (++i < argc) { input.lookups = atoi(argv[i]); default_lookups = 0; }
            else print_CLI_error();
        } else if (!strcmp(arg, "-h")) {
            if (++i < argc) input.hash_bins = atoi(argv[i]); else print_CLI_error();
        } else if (!strcmp(arg, "-p")) {
            if (++i < argc) { input.particles = atoi(argv[i]); default_particles = 0; }
            else print_CLI_error();
        } else if (!strcmp(arg, "-s")) {
            if (++i < argc) input.HM = argv[i]; else print_CLI_error();
        } else if (!strcmp(arg, "-G")) {
            if (++i < argc) {
                if (!strcmp(argv[i], "unionized")) input.grid_type = UNIONIZED;
                else if (!strcmp(argv[i], "nuclide")) input.grid_type = NUCLIDE;
                else if (!strcmp(argv[i], "hash")) input.grid_type = HASH;
                else print_CLI_error();
            } else print_CLI_error();
        } else if (!strcmp(arg, "-b")) {
            if (++i < argc) {
                if (!strcmp(argv[i], "read")) input.binary_mode = READ;
                else if (!strcmp(argv[i], "write")) input.binary_mode = WRITE;
                else print_CLI_error();
            } else print_CLI_error();
        } else if (!strcmp(arg, "-k")) {
            if (++i < argc) input.kernel_id = atoi(argv[i]); else print_CLI_error();
        } else print_CLI_error();
    }
    if (input.nthreads < 1) print_CLI_error();
    if (input.n_isotopes < 1) print_CLI_error();
    if (input.n_gridpoints < 1) print_CLI_error();
    if (input.lookups < 1) print_CLI_error();
    if (input.hash_bins < 1) print_CLI_error();
    if (strcasecmp(input.HM, "small") && strcasecmp(input.HM, "large") &&
        strcasecmp(input.HM, "XL") && strcasecmp(input.HM, "XXL"))
        print_CLI_error();
    if (!strcasecmp(input.HM, "small")) input.n_isotopes = 68;
    else if (!strcasecmp(input.HM, "XL") && !user_g) input.n_gridpoints = 238847;
    else if (!strcasecmp(input.HM, "XXL") && !user_g) input.n_gridpoints = static_cast<long>(238847 * 2.1);
    return input;
}
int print_results(const Inputs &in, int mype, double runtime, int nprocs,
                  unsigned long long vhash) {
    int lookups = (in.simulation_method == HISTORY_BASED) ?
                  in.lookups * in.particles : in.lookups;
    int lookups_per_sec = static_cast<int>(lookups / runtime);
    int is_invalid = 1;
    if (mype == 0) {
        border_print();
        center_print("RESULTS", 79);
        border_print();
        printf("Runtime: %.3lf seconds\n", runtime);
        printf("Lookups: "); fancy_int(lookups);
        printf("Lookups/s: "); fancy_int(lookups_per_sec);
        unsigned long long small = 945990, large = 952131;
        if (!strcmp(in.HM, "large")) {
            if (vhash == large) is_invalid = 0;
        } else if (!strcmp(in.HM, "small")) {
            if (vhash == small) is_invalid = 0;
        }
        if (is_invalid)
            printf("Verification checksum: %llu (WARNING - INVALID CHECKSUM!)\n", vhash);
        else
            printf("Verification checksum: %llu (Valid)\n", vhash);
        border_print();
    }
    return is_invalid;
}
void print_inputs(const Inputs &in, int nprocs, int version) {
    logo(version);
    center_print("INPUT SUMMARY", 79);
    border_print();
    printf("Programming Model:            SYCL\n");
    printf("Simulation Method:            %s\n",
           in.simulation_method == EVENT_BASED ? "Event Based" : "History Based");
    if (in.grid_type == NUCLIDE) printf("Grid Type:                    Nuclide Grid\n");
    else if (in.grid_type == UNIONIZED) printf("Grid Type:                    Unionized Grid\n");
    else printf("Grid Type:                    Hash\n");
    printf("Materials:                    12\n");
    printf("H-M Benchmark Size:           %s\n", in.HM);
    printf("Total Nuclides:               %ld\n", in.n_isotopes);
    printf("Gridpoints (per Nuclide):     "); fancy_int(in.n_gridpoints);
    if (in.grid_type == HASH) { printf("Hash Bins:                    "); fancy_int(in.hash_bins); }
    if (in.grid_type == UNIONIZED) {
        printf("Unionized Energy Gridpoints:  "); fancy_int(in.n_isotopes * in.n_gridpoints);
    }
    if (in.simulation_method == HISTORY_BASED) {
        printf("Particle Histories:           "); fancy_int(in.particles);
        printf("XS Lookups per Particle:      "); fancy_int(in.lookups);
    }

    printf("Binary File Mode:             ");
    if (in.binary_mode == NONE) printf("Off\n");
    else if (in.binary_mode == READ) printf("Read\n");
    else printf("Write\n");
    border_print();
    center_print("INITIALIZATION - DO NOT PROFILE", 79);
    border_print();
}

// ---------- Main ----------
int main(int argc, char *argv[]) {
    int version = 20;
    int mype = 0;
    int nprocs = 1;
    double start, end;
    unsigned long long verification = 0;

    Inputs in = read_CLI(argc, argv);
    print_inputs(in, nprocs, version);

    SimulationData SD;
    if (in.binary_mode == READ) SD = binary_read(in);
    else SD = grid_init_do_not_profile(in);

    if (in.binary_mode == WRITE && mype == 0) binary_write(in, SD);

    sycl::queue q(sycl::default_selector{},
                  sycl::property::queue::in_order{});
    SimulationData GSD = move_simulation_data_to_device(q, in, SD);

    if (mype == 0) {
        printf("\n");
        border_print();
        center_print("SIMULATION", 79);
        border_print();
    }

    start = get_time();

    if (in.simulation_method == EVENT_BASED) {
        switch (in.kernel_id) {
            case 0: verification = run_event_based_simulation_baseline(q, in, GSD); break;
            case 1: verification = run_event_based_simulation_optimization_1(q, in, GSD); break;
            case 2: verification = run_event_based_simulation_optimization_2(q, in, GSD); break;
            case 3: verification = run_event_based_simulation_optimization_3(q, in, GSD); break;
            case 4: verification = run_event_based_simulation_optimization_4(q, in, GSD); break;
            case 5: verification = run_event_based_simulation_optimization_5(q, in, GSD); break;
            case 6: verification = run_event_based_simulation_optimization_6(q, in, GSD); break;
            default:
                printf("Error: No kernel ID %d found!\n", in.kernel_id);
                exit(1);
        }
    } else {
        printf("History-based simulation not implemented. Use event-based method.\n");
        exit(1);
    }

    if (mype == 0) {
        printf("\nSimulation complete.\n");
    }

    end = get_time();

    release_device_memory(q, GSD);
    verification %= 999983ULL;
    int invalid = print_results(in, mype, end - start, nprocs, verification);
    return invalid;
}
