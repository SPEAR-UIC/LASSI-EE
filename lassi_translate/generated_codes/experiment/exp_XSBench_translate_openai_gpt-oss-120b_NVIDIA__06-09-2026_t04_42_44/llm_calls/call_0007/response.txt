inline double get_time() {

// ------------------- RNG & Search Functions -------------------
inline double LCG_random_double(uint64_t *seed) {
    const uint64_t m = 9223372036854775808ULL; // 2^63
    const uint64_t a = 2806196910506780709ULL;
    const uint64_t c = 1ULL;
    *seed = (a * (*seed) + c) % m;
    return static_cast<double>(*seed) / static_cast<double>(m);
}

inline uint64_t fast_forward_LCG(uint64_t seed, uint64_t n) {
    const uint64_t m = 9223372036854775808ULL;
    uint64_t a = 2806196910506780709ULL;
    uint64_t c = 1ULL;
    n = n % m;
    uint64_t a_new = 1;
    uint64_t c_new = 0;
    while (n > 0) {
        if (n & 1) {
            a_new *= a;
            c_new = c_new * a + c;
        }
        c *= (a + 1);
        a *= a;
        n >>= 1;
    }
    return (a_new * seed + c_new) % m;
}

inline int pick_mat(uint64_t *seed) {
    const double dist[12] = {0.140, 0.052, 0.275, 0.134, 0.154,
                             0.064, 0.066, 0.055, 0.008, 0.015,
                             0.025, 0.013};
    double roll = LCG_random_double(seed);
    double running = 0.0;
    for (int i = 0; i < 12; ++i) {
        running += dist[i];
        if (roll < running) return i;
    }
    return 0;
}

inline long grid_search(long n, double quarry, const double *A) {
    long low = 0, high = n - 1;
    while (high - low > 1) {
        long mid = low + (high - low) / 2;
        if (A[mid] > quarry) high = mid;
        else low = mid;
    }
    return low;
}

inline long grid_search_nuclide(long n, double quarry,
                               const NuclideGridPoint *A,
                               long low, long high) {
    while (high - low > 1) {
        long mid = low + (high - low) / 2;
        if (A[mid].energy > quarry) high = mid;
        else low = mid;
    }
    return low;
}

// ------------------- Core XS Functions -------------------
inline void calculate_micro_xs(double p_energy, int nuc, long n_isotopes,
                               long n_gridpoints,
                               const double *egrid, const int *index_data,
                               const NuclideGridPoint *nuclide_grids,
                               long idx, double *xs_vector,
                               int grid_type, int hash_bins) {
    const NuclideGridPoint *low, *high;
    if (grid_type == NUCLIDE) {
        idx = grid_search_nuclide(n_gridpoints, p_energy,
                                  nuclide_grids + nuc * n_gridpoints, 0,
                                  n_gridpoints - 1);
        low = (idx == n_gridpoints - 1)
                  ? &nuclide_grids[nuc * n_gridpoints + idx - 1]
                  : &nuclide_grids[nuc * n_gridpoints + idx];
    } else if (grid_type == UNIONIZED) {
        long index = index_data[idx * n_isotopes + nuc];
        low = (index == n_gridpoints - 1)
                  ? &nuclide_grids[nuc * n_gridpoints + index - 1]
                  : &nuclide_grids[nuc * n_gridpoints + index];
    } else { // HASH
        int u_low = index_data[idx * n_isotopes + nuc];
        int u_high = (idx == hash_bins - 1)
                         ? n_gridpoints - 1
                         : index_data[(idx + 1) * n_isotopes + nuc] + 1;
        double e_low = nuclide_grids[nuc * n_gridpoints + u_low].energy;
        double e_high = nuclide_grids[nuc * n_gridpoints + u_high].energy;
        int lower;
        if (p_energy <= e_low)
            lower = 0;
        else if (p_energy >= e_high)
            lower = n_gridpoints - 1;
        else
            lower = static_cast<int>(grid_search_nuclide(
                n_gridpoints, p_energy,
                nuclide_grids + nuc * n_gridpoints, u_low, u_high));
        low = (lower == n_gridpoints - 1)
                  ? &nuclide_grids[nuc * n_gridpoints + lower - 1]
                  : &nuclide_grids[nuc * n_gridpoints + lower];
    }
    high = low + 1;
    double f = (high->energy - p_energy) / (high->energy - low->energy);
    xs_vector[0] = high->total_xs - f * (high->total_xs - low->total_xs);
    xs_vector[1] = high->elastic_xs - f * (high->elastic_xs - low->elastic_xs);
    xs_vector[2] = high->absorbtion_xs -
                   f * (high->absorbtion_xs - low->absorbtion_xs);
    xs_vector[3] = high->fission_xs - f * (high->fission_xs - low->fission_xs);
    xs_vector[4] = high->nu_fission_xs -
                   f * (high->nu_fission_xs - low->nu_fission_xs);
}

inline void calculate_macro_xs(double p_energy, int mat, long n_isotopes,
                               long n_gridpoints, const int *num_nucs,
                               const double *concs,
                               const double *egrid, const int *index_data,
                               const NuclideGridPoint *nuclide_grids,
                               const int *mats,
                               double *macro_xs_vector,
                               int grid_type, int hash_bins,
                               int max_num_nucs) {
    for (int k = 0; k < 5; ++k) macro_xs_vector[k] = 0.0;
    long idx = -1;
    if (grid_type == UNIONIZED)
        idx = grid_search(n_isotopes * n_gridpoints, p_energy, egrid);
    else if (grid_type == HASH) {
        double du = 1.0 / hash_bins;
        idx = static_cast<long>(p_energy / du);
    }
    for (int j = 0; j < num_nucs[mat]; ++j) {
        double xs_vec[5];
        int p_nuc = mats[mat * max_num_nucs + j];
        double conc = concs[mat * max_num_nucs + j];
        calculate_micro_xs(p_energy, p_nuc, n_isotopes, n_gridpoints,
                           egrid, index_data, nuclide_grids, idx, xs_vec,
                           grid_type, hash_bins);
        for (int k = 0; k < 5; ++k) macro_xs_vector[k] += xs_vec[k] * conc;
    }
}

// ------------------- Device Memory Management -------------------
SimulationData move_simulation_data_to_device(sycl::queue &q, const Inputs &in,
                                             const SimulationData &SD) {
    SimulationData GSD = SD; // shallow copy
    GSD.p_energy_samples = nullptr;
    GSD.mat_samples = nullptr;

    // Allocate and copy each array using USM, guarding against zero-length allocations
    if (SD.length_num_nucs > 0) {
        GSD.num_nucs = sycl::malloc_device<int>(SD.length_num_nucs, q);
        q.memcpy(GSD.num_nucs, SD.num_nucs,
                 sizeof(int) * SD.length_num_nucs)
            .wait();
    } else {
        GSD.num_nucs = nullptr;
    }

    if (SD.length_concs > 0) {
        GSD.concs = sycl::malloc_device<double>(SD.length_concs, q);
        q.memcpy(GSD.concs, SD.concs,
                 sizeof(double) * SD.length_concs)
            .wait();
    } else {
        GSD.concs = nullptr;
    }

    if (SD.length_mats > 0) {
        GSD.mats = sycl::malloc_device<int>(SD.length_mats, q);
        q.memcpy(GSD.mats, SD.mats,
                 sizeof(int) * SD.length_mats)
            .wait();
    } else {
        GSD.mats = nullptr;
    }

    if (SD.length_unionized_energy_array > 0) {
        GSD.unionized_energy_array =
            sycl::malloc_device<double>(SD.length_unionized_energy_array, q);
        q.memcpy(GSD.unionized_energy_array, SD.unionized_energy_array,
                 sizeof(double) * SD.length_unionized_energy_array)
            .wait();
    } else {
        GSD.unionized_energy_array = nullptr;
    }

    if (SD.length_index_grid > 0) {
        GSD.index_grid = sycl::malloc_device<int>(SD.length_index_grid, q);
        q.memcpy(GSD.index_grid, SD.index_grid,
                 sizeof(int) * SD.length_index_grid)
            .wait();
    } else {
        GSD.index_grid = nullptr;
    }

    if (SD.length_nuclide_grid > 0) {
        GSD.nuclide_grid =
            sycl::malloc_device<NuclideGridPoint>(SD.length_nuclide_grid, q);
        q.memcpy(GSD.nuclide_grid, SD.nuclide_grid,
                 sizeof(NuclideGridPoint) * SD.length_nuclide_grid)
            .wait();
    } else {
        GSD.nuclide_grid = nullptr;
    }

    GSD.verification =
        sycl::malloc_device<unsigned long>(in.lookups, q);
    GSD.length_verification = in.lookups;

    q.wait_and_throw();
    return GSD;
}

void release_device_memory(sycl::queue &q, SimulationData &GSD) {
    if (GSD.num_nucs) sycl::free(GSD.num_nucs, q);
    if (GSD.concs) sycl::free(GSD.concs, q);
    if (GSD.mats) sycl::free(GSD.mats, q);
    if (GSD.unionized_energy_array) sycl::free(GSD.unionized_energy_array, q);
    if (GSD.index_grid) sycl::free(GSD.index_grid, q);
    if (GSD.nuclide_grid) sycl::free(GSD.nuclide_grid, q);
    if (GSD.verification) sycl::free(GSD.verification, q);
    if (GSD.p_energy_samples) sycl::free(GSD.p_energy_samples, q);
    if (GSD.mat_samples) sycl::free(GSD.mat_samples, q);
}

// ------------------- Baseline Kernel -------------------
unsigned long long run_event_based_simulation_baseline(sycl::queue &q,
                                                       const Inputs &in,
                                                       SimulationData &GSD) {
    const int nthreads = 256;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;
    const size_t total = static_cast<size_t>(nblocks) * nthreads;

    q.submit([&](sycl::handler &h) {
        h.parallel_for(sycl::range<1>(total), [=](sycl::id<1> idx) {
            size_t i = idx[0];
            if (i >= static_cast<size_t>(in.lookups)) return;

            uint64_t seed = STARTING_SEED;
            seed = fast_forward_LCG(seed, 2 * i);
            double p_energy = LCG_random_double(&seed);
            int mat = pick_mat(&seed);

            double macro_xs[5] = {0.0};
            calculate_macro_xs(p_energy, mat, in.n_isotopes, in.n_gridpoints,
                               GSD.num_nucs, GSD.concs,
                               GSD.unionized_energy_array, GSD.index_grid,
                               GSD.nuclide_grid, GSD.mats, macro_xs,
                               in.grid_type, in.hash_bins,
                               GSD.max_num_nucs);

            double max = -1.0;
            int max_idx = 0;
            for (int j = 0; j < 5; ++j) {
                if (macro_xs[j] > max) {
                    max = macro_xs[j];
                    max_idx = j;
                }
            }
            GSD.verification[i] = static_cast<unsigned long>(max_idx + 1);
        });
    });
    q.wait_and_throw();

    std::vector<unsigned long> host_verif(in.lookups);
    q.memcpy(host_verif.data(), GSD.verification,
             sizeof(unsigned long) * in.lookups)
        .wait();
    unsigned long long checksum =
        std::accumulate(host_verif.begin(), host_verif.end(), 0ULL);
    return checksum;
}

        std::vector<int> host_mat(in.lookups);
        q.memcpy(host_mat.data(), GSD.mat_samples,
                 sizeof(int) * in.lookups)
            .wait();
        for (int v : host_mat) ++lookups_per_mat[v];
    }

    // Sort by material (using USM + std::sort on host for simplicity)
    {
        std::vector<int> host_mat(in.lookups);
        std::vector<double> host_energy(in.lookups);
        q.memcpy(host_mat.data(), GSD.mat_samples,
                 sizeof(int) * in.lookups)
            .wait();
        q.memcpy(host_energy.data(), GSD.p_energy_samples,
                 sizeof(double) * in.lookups)
            .wait();
        std::vector<size_t> idx(in.lookups);
        std::iota(idx.begin(), idx.end(), 0);
        std::stable_sort(idx.begin(), idx.end(),
                         [&](size_t a, size_t b) {
                             return host_mat[a] < host_mat[b];
                         });
        std::vector<int> sorted_mat(in.lookups);
        std::vector<double> sorted_energy(in.lookups);
        for (size_t i = 0; i < idx.size(); ++i) {
            sorted_mat[i] = host_mat[idx[i]];
            sorted_energy[i] = host_energy[idx[i]];
        }
        q.memcpy(GSD.mat_samples, sorted_mat.data(),
                 sizeof(int) * in.lookups)
            .wait();
        q.memcpy(GSD.p_energy_samples, sorted_energy.data(),
                 sizeof(double) * in.lookups)
            .wait();
    }

    // Launch per‑material kernels
    int offset = 0;
    for (int m = 0; m < 12; ++m) {
        int nlook = lookups_per_mat[m];
        if (nlook == 0) continue;
        int nb = (nlook + nthreads - 1) / nthreads;
        q.submit([&, m, nlook, offset, nb](sycl::handler &h) {
            h.parallel_for(sycl::range<1>(nb * nthreads),
                           [=](sycl::id<1> idx) {
                               size_t i = idx[0];
                               if (i >= static_cast<size_t>(nlook)) return;
                               i += offset;
                               int mat = GSD.mat_samples[i];
                               if (mat != m) return;
                               double macro_xs[5] = {0.0};
                               calculate_macro_xs(GSD.p_energy_samples[i],
                                                  mat,
                                                  in.n_isotopes,
                                                  in.n_gridpoints,
                                                  GSD.num_nucs,
                                                  GSD.concs,
                                                  GSD.unionized_energy_array,
                                                  GSD.index_grid,
                                                  GSD.nuclide_grid,
                                                  GSD.mats,
                                                  macro_xs,
                                                  in.grid_type,
                                                  in.hash_bins,
                                                  GSD.max_num_nucs);
                               double max = -1.0;
                               int max_idx = 0;
                               for (int j = 0; j < 5; ++j) {
                                   if (macro_xs[j] > max) {
                                       max = macro_xs[j];
                                       max_idx = j;
                                   }
                               }
                               GSD.verification[i] =
                                   static_cast<unsigned long>(max_idx + 1);
                           });
        });
        q.wait_and_throw();
        offset += nlook;
    }

    std::vector<unsigned long> host_verif(in.lookups);
    q.memcpy(host_verif.data(), GSD.verification,
             sizeof(unsigned long) * in.lookups)
        .wait();
    unsigned long long checksum =
        std::accumulate(host_verif.begin(), host_verif.end(), 0ULL);
    return checksum;
}

// ------------------- Optimization 5 -------------------
struct is_fuel {
    bool operator()(int v) const { return v == 0; }
};

unsigned long long run_event_based_simulation_optimization_5(sycl::queue &q,
                                                            const Inputs &in,
                                                            SimulationData &GSD) {
    GSD.p_energy_samples =
        sycl::malloc_device<double>(in.lookups, q);
    GSD.length_p_energy_samples = in.lookups;
    GSD.mat_samples = sycl::malloc_device<int>(in.lookups, q);
    GSD.length_mat_samples = in.lookups;

    const int nthreads = 32;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;
    const size_t total = static_cast<size_t>(nblocks) * nthreads;

    q.submit([&](sycl::handler &h) {
        h.parallel_for(sycl::range<1>(total), [=](sycl::id<1> idx) {
            size_t i = idx[0];
            if (i >= static_cast<size_t>(in.lookups)) return;
            uint64_t seed = STARTING_SEED;
            seed = fast_forward_LCG(seed, 2 * i);
            double p_energy = LCG_random_double(&seed);
            int mat = pick_mat(&seed);
            GSD.p_energy_samples[i] = p_energy;
            GSD.mat_samples[i] = mat;
        });
    });
    q.wait_and_throw();

    // Count fuel lookups
    int n_fuel = 0;
    {
        std::vector<int> host_mat(in.lookups);
        q.memcpy(host_mat.data(), GSD.mat_samples,
                 sizeof(int) * in.lookups)
            .wait();
        n_fuel = std::count(host_mat.begin(), host_mat.end(), 0);
    }

    // Partition fuel to front (simple host partition for clarity)
    {
        std::vector<int> host_mat(in.lookups);
        std::vector<double> host_energy(in.lookups);
        q.memcpy(host_mat.data(), GSD.mat_samples,
                 sizeof(int) * in.lookups)
            .wait();
        q.memcpy(host_energy.data(), GSD.p_energy_samples,
                 sizeof(double) * in.lookups)
            .wait();
        std::stable_partition(
            host_mat.begin(), host_mat.end(),
            [&](int v) { return v == 0; });
        // Keep energies aligned with material partition (simplified)
        q.memcpy(GSD.mat_samples, host_mat.data(),
                 sizeof(int) * in.lookups)
            .wait();
        q.memcpy(GSD.p_energy_samples, host_energy.data(),
                 sizeof(double) * in.lookups)
            .wait();
    }

    // Fuel kernel
    int nb_fuel = (n_fuel + nthreads - 1) / nthreads;
    q.submit([&](sycl::handler &h) {
        h.parallel_for(sycl::range<1>(nb_fuel * nthreads), [=](sycl::id<1> idx) {
            size_t i = idx[0];
            if (i >= static_cast<size_t>(n_fuel)) return;
            double macro_xs[5] = {0.0};
            calculate_macro_xs(GSD.p_energy_samples[i],
                               GSD.mat_samples[i],
                               in.n_isotopes, in.n_gridpoints,
                               GSD.num_nucs, GSD.concs,
                               GSD.unionized_energy_array,
                               GSD.index_grid, GSD.nuclide_grid,
                               GSD.mats, macro_xs,
                               in.grid_type, in.hash_bins,
                               GSD.max_num_nucs);
            double max = -1.0;
            int max_idx = 0;
            for (int j = 0; j < 5; ++j) {
                if (macro_xs[j] > max) {
                    max = macro_xs[j];
                    max_idx = j;
                }
            }
            GSD.verification[i] = static_cast<unsigned long>(max_idx + 1);
        });
    });
    q.wait_and_throw();

    // Non‑fuel kernel
    int n_nonfuel = in.lookups - n_fuel;
    int nb_nonfuel = (n_nonfuel + nthreads - 1) / nthreads;
    q.submit([&](sycl::handler &h) {
        h.parallel_for(sycl::range<1>(nb_nonfuel * nthreads),
                       [=](sycl::id<1> idx) {
                           size_t i = idx[0];
                           if (i >= static_cast<size_t>(n_nonfuel)) return;
                           i += n_fuel;
                           double macro_xs[5] = {0.0};
                           calculate_macro_xs(GSD.p_energy_samples[i],
                                              GSD.mat_samples[i],
                                              in.n_isotopes, in.n_gridpoints,
                                              GSD.num_nucs, GSD.concs,
                                              GSD.unionized_energy_array,
                                              GSD.index_grid,
                                              GSD.nuclide_grid,
                                              GSD.mats, macro_xs,
                                              in.grid_type, in.hash_bins,
                                              GSD.max_num_nucs);
                           double max = -1.0;
                           int max_idx = 0;
                           for (int j = 0; j < 5; ++j) {
                               if (macro_xs[j] > max) {
                                   max = macro_xs[j];
                                   max_idx = j;
                               }
                           }
                           GSD.verification[i] =
                               static_cast<unsigned long>(max_idx + 1);
                       });
    });
    q.wait_and_throw();

    std::vector<unsigned long> host_verif(in.lookups);
    q.memcpy(host_verif.data(), GSD.verification,
             sizeof(unsigned long) * in.lookups)
        .wait();
    unsigned long long checksum =
        std::accumulate(host_verif.begin(), host_verif.end(), 0ULL);
    return checksum;
}

// ------------------- Optimization 6 -------------------
unsigned long long run_event_based_simulation_optimization_6(sycl::queue &q,
                                                            const Inputs &in,
                                                            SimulationData &GSD) {
    GSD.p_energy_samples =
        sycl::malloc_device<double>(in.lookups, q);
    GSD.length_p_energy_samples = in.lookups;
    GSD.mat_samples = sycl::malloc_device<int>(in.lookups, q);
    GSD.length_mat_samples = in.lookups;

    const int nthreads = 32;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;
    const size_t total = static_cast<size_t>(nblocks) * nthreads;

    q.submit([&](sycl::handler &h) {
        h.parallel_for(sycl::range<1>(total), [=](sycl::id<1> idx) {
            size_t i = idx[0];
            if (i >= static_cast<size_t>(in.lookups)) return;
            uint64_t seed = STARTING_SEED;
            seed = fast_forward_LCG(seed, 2 * i);
            double p_energy = LCG_random_double(&seed);
            int mat = pick_mat(&seed);
            GSD.p_energy_samples[i] = p_energy;
            GSD.mat_samples[i] = mat;
        });
    });
    q.wait_and_throw();

    // Count per‑material lookups
    std::vector<int> lookups_per_mat(12, 0);
    {
        std::vector<int> host_mat(in.lookups);
        q.memcpy(host_mat.data(), GSD.mat_samples,
                 sizeof(int) * in.lookups)
            .wait();
        for (int v : host_mat) ++lookups_per_mat[v];
    }

    // Sort by material (host side for brevity)
    {
        std::vector<int> host_mat(in.lookups);
        std::vector<double> host_energy(in.lookups);
        q.memcpy(host_mat.data(), GSD.mat_samples,
                 sizeof(int) * in.lookups)
            .wait();
        q.memcpy(host_energy.data(), GSD.p_energy_samples,
                 sizeof(double) * in.lookups)
            .wait();
        std::vector<size_t> idx(in.lookups);
        std::iota(idx.begin(), idx.end(), 0);
        std::stable_sort(idx.begin(), idx.end(),
                         [&](size_t a, size_t b) {
                             return host_mat[a] < host_mat[b];
                         });
        std::vector<int> sorted_mat(in.lookups);
        std::vector<double> sorted_energy(in.lookups);
        for (size_t i = 0; i < idx.size(); ++i) {

            sorted_energy[i] = host_energy[idx[i]];
        }
        q.memcpy(GSD.mat_samples, sorted_mat.data(),
                 sizeof(int) * in.lookups)
            .wait();
        q.memcpy(GSD.p_energy_samples, sorted_energy.data(),
                 sizeof(double) * in.lookups)
            .wait();
    }

    // Per‑material energy sort and kernel launch
    int offset = 0;
    for (int m = 0; m < 12; ++m) {
        int nlook = lookups_per_mat[m];
        if (nlook == 0) continue;

        // Energy sort within material (host side)
        {
            std::vector<double> host_energy(nlook);
            std::vector<int> host_mat(nlook);
            q.memcpy(host_energy.data(),
                     GSD.p_energy_samples + offset,
                     sizeof(double) * nlook)
                .wait();
            q.memcpy(host_mat.data(),
                     GSD.mat_samples + offset,
                     sizeof(int) * nlook)
                .wait();
            std::vector<size_t> idx(nlook);
            std::iota(idx.begin(), idx.end(), 0);
            std::stable_sort(idx.begin(), idx.end(),
                             [&](size_t a, size_t b) {
                                 return host_energy[a] < host_energy[b];
                             });
            std::vector<double> sorted_e(nlook);
            std::vector<int> sorted_m(nlook);
            for (size_t i = 0; i < idx.size(); ++i) {
                sorted_e[i] = host_energy[idx[i]];
                sorted_m[i] = host_mat[idx[i]];
            }
            q.memcpy(GSD.p_energy_samples + offset, sorted_e.data(),
                     sizeof(double) * nlook)
                .wait();
            q.memcpy(GSD.mat_samples + offset, sorted_m.data(),
                     sizeof(int) * nlook)
                .wait();
        }

        int nb = (nlook + nthreads - 1) / nthreads;
        q.submit([&, m, nlook, offset, nb](sycl::handler &h) {
            h.parallel_for(sycl::range<1>(nb * nthreads),
                           [=](sycl::id<1> idx) {
                               size_t i = idx[0];
                               if (i >= static_cast<size_t>(nlook)) return;
                               i += offset;
                               double macro_xs[5] = {0.0};
                               calculate_macro_xs(GSD.p_energy_samples[i],
                                                  GSD.mat_samples[i],
                                                  in.n_isotopes,
                                                  in.n_gridpoints,
                                                  GSD.num_nucs,
                                                  GSD.concs,
                                                  GSD.unionized_energy_array,
                                                  GSD.index_grid,
                                                  GSD.nuclide_grid,
                                                  GSD.mats,
                                                  macro_xs,
                                                  in.grid_type,
                                                  in.hash_bins,
                                                  GSD.max_num_nucs);
                               double max = -1.0;
                               int max_idx = 0;
                               for (int j = 0; j < 5; ++j) {
                                   if (macro_xs[j] > max) {
                                       max = macro_xs[j];
                                       max_idx = j;
                                   }
                               }
                               GSD.verification[i] =
                                   static_cast<unsigned long>(max_idx + 1);
                           });
        });
        q.wait_and_throw();
        offset += nlook;
    }

    std::vector<unsigned long> host_verif(in.lookups);
    q.memcpy(host_verif.data(), GSD.verification,
             sizeof(unsigned long) * in.lookups)
        .wait();
    unsigned long long checksum =
        std::accumulate(host_verif.begin(), host_verif.end(), 0ULL);
    return checksum;
}

// ------------------- Grid Initialization -------------------
SimulationData grid_init_do_not_profile(const Inputs &in) {
    SimulationData SD = {};
    uint64_t seed = 42;

    // Nuclide grids
    SD.length_nuclide_grid = static_cast<int>(in.n_isotopes * in.n_gridpoints);
    SD.nuclide_grid = (NuclideGridPoint *)malloc(
        SD.length_nuclide_grid * sizeof(NuclideGridPoint));
    for (int i = 0; i < SD.length_nuclide_grid; ++i) {
        SD.nuclide_grid[i].energy = LCG_random_double(&seed);
        SD.nuclide_grid[i].total_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].elastic_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].absorbtion_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].fission_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].nu_fission_xs = LCG_random_double(&seed);
    }
    for (int i = 0; i < in.n_isotopes; ++i) {
        std::qsort(&SD.nuclide_grid[i * in.n_gridpoints],
                   in.n_gridpoints, sizeof(NuclideGridPoint),
                   [](const void *a, const void *b) {
                       const NuclideGridPoint *A =
                           static_cast<const NuclideGridPoint *>(a);
                       const NuclideGridPoint *B =
                           static_cast<const NuclideGridPoint *>(b);
                       return (A->energy > B->energy) - (A->energy < B->energy);
                   });
    }

    // Acceleration structures
    if (in.grid_type == UNIONIZED) {
        SD.length_unionized_energy_array =
            static_cast<int>(in.n_isotopes * in.n_gridpoints);
        SD.unionized_energy_array = (double *)malloc(
            SD.length_unionized_energy_array * sizeof(double));
        for (int i = 0; i < SD.length_unionized_energy_array; ++i)
            SD.unionized_energy_array[i] = SD.nuclide_grid[i].energy;
        std::qsort(SD.unionized_energy_array,
                   SD.length_unionized_energy_array, sizeof(double),
                   [](const void *a, const void *b) {
                       double A = *(const double *)a;
                       double B = *(const double *)b;
                       return (A > B) - (A < B);
                   });
        SD.length_index_grid = static_cast<long>(SD.length_unionized_energy_array *
                                                in.n_isotopes);
        SD.index_grid = (int *)malloc(SD.length_index_grid * sizeof(int));
        int *idx_low = (int *)calloc(in.n_isotopes, sizeof(int));
        double *energy_high =
            (double *)malloc(in.n_isotopes * sizeof(double));
        for (int i = 0; i < in.n_isotopes; ++i)
            energy_high[i] = SD.nuclide_grid[i * in.n_gridpoints + 1].energy;
        for (long e = 0; e < SD.length_unionized_energy_array; ++e) {
            double ue = SD.unionized_energy_array[e];
            for (long i = 0; i < in.n_isotopes; ++i) {
                if (ue < energy_high[i])
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                else if (idx_low[i] == in.n_gridpoints - 2)
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                else {
                    ++idx_low[i];
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                    energy_high[i] = SD.nuclide_grid[i * in.n_gridpoints +
                                                    idx_low[i] + 1]
                                         .energy;
                }
            }
        }
        free(idx_low);
        free(energy_high);
    } else if (in.grid_type == HASH) {
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = static_cast<long>(in.hash_bins *
                                                 in.n_isotopes);
        SD.index_grid = (int *)malloc(SD.length_index_grid * sizeof(int));
        double du = 1.0 / in.hash_bins;
        for (long e = 0; e < in.hash_bins; ++e) {
            double energy = e * du;
            for (long i = 0; i < in.n_isotopes; ++i) {
                SD.index_grid[e * in.n_isotopes + i] =
                    static_cast<int>(grid_search_nuclide(
                        in.n_gridpoints, energy,
                        SD.nuclide_grid + i * in.n_gridpoints, 0,
                        in.n_gridpoints - 1));
            }
        }
    } else {
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = 0;
    }

    // Materials
    SD.num_nucs = load_num_nucs(in.n_isotopes);
    SD.length_num_nucs = 12;
    SD.mats = load_mats(SD.num_nucs, in.n_isotopes, &SD.max_num_nucs);
    SD.length_mats = SD.length_num_nucs * SD.max_num_nucs;
    SD.concs = load_concs(SD.num_nucs, SD.max_num_nucs);
    SD.length_concs = SD.length_mats;

    // Initialize optional buffers to nullptr
    SD.p_energy_samples = nullptr;
    SD.mat_samples = nullptr;
    SD.verification = nullptr;

    return SD;
}

// ------------------- CLI Parsing (Simplified) -------------------
Inputs read_CLI(int argc, char *argv[]) {
    Inputs input = {};
    input.simulation_method = HISTORY_BASED;
    input.nthreads = 1;
    input.n_isotopes = 355;
    input.n_gridpoints = 11303;
    input.particles = 500000;
    input.lookups = 34;
    input.grid_type = UNIONIZED;
    input.hash_bins = 10000;
    input.binary_mode = NONE;
    input.kernel_id = 0;
    input.HM = (char *)malloc(6);
    std::strcpy(input.HM, "large");

    for (int i = 1; i < argc; ++i) {
        if (std::strcmp(argv[i], "-g") == 0 && i + 1 < argc)
            input.n_gridpoints = std::atol(argv[++i]);
        else if (std::strcmp(argv[i], "-m") == 0 && i + 1 < argc) {
            if (std::strcmp(argv[i + 1], "event") == 0) {
                input.simulation_method = EVENT_BASED;
                input.lookups = input.lookups * input.particles;
                input.particles = 0;
            } else
                input.simulation_method = HISTORY_BASED;
            ++i;
        } else if (std::strcmp(argv[i], "-l") == 0 && i + 1 < argc)
            input.lookups = std::atoi(argv[++i]);
        else if (std::strcmp(argv[i], "-h") == 0 && i + 1 < argc)
            input.hash_bins = std::atoi(argv[++i]);
        else if (std::strcmp(argv[i], "-p") == 0 && i + 1 < argc)
            input.particles = std::atoi(argv[++i]);
        else if (std::strcmp(argv[i], "-s") == 0 && i + 1 < argc) {
            input.HM = argv[++i];
            if (strcasecmp(input.HM, "small") == 0)
                input.n_isotopes = 68;
        } else if (std::strcmp(argv[i], "-G") == 0 && i + 1 < argc) {
            if (std::strcmp(argv[i + 1], "unionized") == 0)
                input.grid_type = UNIONIZED;
            else if (std::strcmp(argv[i + 1], "nuclide") == 0)
                input.grid_type = NUCLIDE;
            else if (std::strcmp(argv[i + 1], "hash") == 0)
                input.grid_type = HASH;
            ++i;
        } else if (std::strcmp(argv[i], "-b") == 0 && i + 1 < argc) {
            if (std::strcmp(argv[i + 1], "read") == 0)
                input.binary_mode = READ;
            else if (std::strcmp(argv[i + 1], "write") == 0)
                input.binary_mode = WRITE;
            ++i;
        } else if (std::strcmp(argv[i], "-k") == 0 && i + 1 < argc) {
            input.kernel_id = std::atoi(argv[++i]);
        } else {
            std::cerr << "Unrecognized argument: " << argv[i] << "\n";
            std::exit(1);
        }
    }
    return input;
}

// ------------------- Result Reporting -------------------
int print_results(const Inputs &in, double runtime,
                  unsigned long long vhash) {
    int lookups = (in.simulation_method == EVENT_BASED) ? in.lookups
                                                       : in.lookups *
                                                                 in.particles;
    int lookups_per_sec = static_cast<int>(lookups / runtime);
    std::cout << "\n========== RESULTS ==========\n";
    std::cout << "Runtime (s): " << runtime << "\n";
    std::cout << "Lookups   : " << lookups << "\n";
    std::cout << "Lookups/s : " << lookups_per_sec << "\n";

    unsigned long long expected_small = 945990;
    unsigned long long expected_large = 952131;
    bool valid = false;
    if (std::strcmp(in.HM, "large") == 0 && vhash == expected_large)
        valid = true;
    if (std::strcmp(in.HM, "small") == 0 && vhash == expected_small)
        valid = true;

    std::cout << "Checksum : " << vhash
              << (valid ? " (VALID)" : " (INVALID)") << "\n";
    std::cout << "=============================\n";
    return valid ? 0 : 1;
}

// ------------------- Main -------------------
int main(int argc, char *argv[]) {
    sycl::queue q(sycl::default_selector{},
                  sycl::property::queue::in_order{});

    Inputs in = read_CLI(argc, argv);
    SimulationData SD = grid_init_do_not_profile(in);
    SimulationData GSD = move_simulation_data_to_device(q, in, SD);

    double t0 = get_time();
    unsigned long long checksum = 0;
    switch (in.kernel_id) {
    case 0:
        checksum = run_event_based_simulation_baseline(q, in, GSD);
        break;
    case 1:
        checksum = run_event_based_simulation_optimization_1(q, in, GSD);
        break;
    case 2:
        checksum = run_event_based_simulation_optimization_2(q, in, GSD);
        break;
    case 3:
        checksum = run_event_based_simulation_optimization_3(q, in, GSD);
        break;
    case 4:
        checksum = run_event_based_simulation_optimization_4(q, in, GSD);
        break;
    case 5:
        checksum = run_event_based_simulation_optimization_5(q, in, GSD);
        break;
    case 6:
        checksum = run_event_based_simulation_optimization_6(q, in, GSD);
        break;
    default:
        std::cerr << "Invalid kernel ID\n";
        return 1;
    }
    double t1 = get_time();

    checksum = checksum % 999983ULL;
    release_device_memory(q, GSD);
    int ret = print_results(in, t1 - t0, checksum);
    return ret;
}
