inline double LCG_random_double(uint64_t *seed) {
    const uint64_t m = 9223372036854775808ULL;
    const uint64_t a = 2806196910506780709ULL;
    const uint64_t c = 1ULL;
    *seed = (a * (*seed) + c) % m;
    return static_cast<double>(*seed) / static_cast<double>(m);
}
inline uint64_t fast_forward_LCG(uint64_t seed, uint64_t n) {
    const uint64_t m = 9223372036854775808ULL;
    uint64_t a = 2806196910506780709ULL;
    uint64_t c = 1ULL;
    n %= m;
    uint64_t a_new = 1, c_new = 0;
    while (n) {
        if (n & 1) {
            a_new *= a;
            c_new = c_new * a + c;
        }
        c *= (a + 1);
        a *= a;
        n >>= 1;
    }
    return (a_new * seed + c_new) % m;
}

inline long grid_search(long n, double quarry, const double *A) {
    long low = 0, high = n - 1;
    while (high - low > 1) {
        long mid = low + (high - low) / 2;
        if (A[mid] > quarry) high = mid;
        else low = mid;
    }
    return low;
}
inline long grid_search_nuclide(long n, double quarry,
                               const NuclideGridPoint *A,
                               long low, long high) {
    while (high - low > 1) {
        long mid = low + (high - low) / 2;
        if (A[mid].energy > quarry) high = mid;
        else low = mid;
    }
    return low;
}
inline int pick_mat(uint64_t *seed) {
    const double dist[12] = {0.140,0.052,0.275,0.134,0.154,
                             0.064,0.066,0.055,0.008,0.015,
                             0.025,0.013};
    double roll = LCG_random_double(seed);
    double cum = 0.0;
    for (int i = 0; i < 12; ++i) {
        cum += dist[i];
        if (roll < cum) return i;
    }
    return 0;
}
inline void calculate_micro_xs(double p_energy, int nuc,
                               long n_isotopes, long n_gridpoints,
                               const double *egrid, const int *index_data,
                               const NuclideGridPoint *nuclide_grids,
                               long idx, double *xs_vector,
                               int grid_type, int hash_bins) {
    const NuclideGridPoint *low;
    if (grid_type == NUCLIDE) {
        idx = grid_search_nuclide(n_gridpoints, p_energy,
                                  nuclide_grids + nuc * n_gridpoints,
                                  0, n_gridpoints - 1);
        low = (idx == n_gridpoints - 1) ?
              &nuclide_grids[nuc * n_gridpoints + idx - 1] :
              &nuclide_grids[nuc * n_gridpoints + idx];
    } else if (grid_type == UNIONIZED) {
        int ind = index_data[idx * n_isotopes + nuc];
        low = (ind == n_gridpoints - 1) ?
              &nuclide_grids[nuc * n_gridpoints + ind - 1] :
              &nuclide_grids[nuc * n_gridpoints + ind];
    } else { // HASH
        int u_low = index_data[idx * n_isotopes + nuc];
        int u_high = (idx == hash_bins - 1) ?
                     n_gridpoints - 1 :
                     index_data[(idx + 1) * n_isotopes + nuc] + 1;
        double e_low  = nuclide_grids[nuc * n_gridpoints + u_low].energy;
        double e_high = nuclide_grids[nuc * n_gridpoints + u_high].energy;
        int lower;
        if (p_energy <= e_low) lower = 0;
        else if (p_energy >= e_high) lower = n_gridpoints - 1;
        else lower = grid_search_nuclide(n_gridpoints, p_energy,
                                         nuclide_grids + nuc * n_gridpoints,
                                         u_low, u_high);
        low = (lower == n_gridpoints - 1) ?
              &nuclide_grids[nuc * n_gridpoints + lower - 1] :
              &nuclide_grids[nuc * n_gridpoints + lower];
    }
    const NuclideGridPoint *high = low + 1;
    double f = (high->energy - p_energy) / (high->energy - low->energy);
    xs_vector[0] = high->total_xs - f * (high->total_xs - low->total_xs);
    xs_vector[1] = high->elastic_xs - f * (high->elastic_xs - low->elastic_xs);
    xs_vector[2] = high->absorbtion_xs - f * (high->absorbtion_xs - low->absorbtion_xs);
    xs_vector[3] = high->fission_xs - f * (high->fission_xs - low->fission_xs);
    xs_vector[4] = high->nu_fission_xs - f * (high->nu_fission_xs - low->nu_fission_xs);
}
inline void calculate_macro_xs(double p_energy, int mat,
                               long n_isotopes, long n_gridpoints,
                               const int *num_nucs, const double *concs,
                               const double *egrid, const int *index_data,
                               const NuclideGridPoint *nuclide_grids,
                               const int *mats, double *macro_xs_vector,
                               int grid_type, int hash_bins, int max_num_nucs) {
    for (int k = 0; k < 5; ++k) macro_xs_vector[k] = 0.0;
    long idx = -1;
    if (grid_type == UNIONIZED) idx = grid_search(n_isotopes * n_gridpoints, p_energy, egrid);
    else if (grid_type == HASH) {
        double du = 1.0 / hash_bins;
        idx = static_cast<long>(p_energy / du);
    }
    for (int j = 0; j < num_nucs[mat]; ++j) {
        double xs_vec[5];
        int p_nuc = mats[mat * max_num_nucs + j];
        double conc = concs[mat * max_num_nucs + j];
        calculate_micro_xs(p_energy, p_nuc, n_isotopes, n_gridpoints,
                           egrid, index_data, nuclide_grids,
                           idx, xs_vec, grid_type, hash_bins);
        for (int k = 0; k < 5; ++k) macro_xs_vector[k] += xs_vec[k] * conc;
    }
}

int double_compare(const void *a, const void *b) {
    double A = *(const double *)a;
    double B = *(const double *)b;
    return (A > B) - (A < B);
}
int NGP_compare(const void *a, const void *b) {
    const NuclideGridPoint *A = (const NuclideGridPoint *)a;
    const NuclideGridPoint *B = (const NuclideGridPoint *)b;
    return (A->energy > B->energy) - (A->energy < B->energy);
}

double *load_concs(int *num_nucs, int max_num_nucs) {
    uint64_t seed = STARTING_SEED * STARTING_SEED;
    double *concs = (double *)malloc(12 * max_num_nucs * sizeof(double));
    for (int i = 0; i < 12; ++i)
        for (int j = 0; j < num_nucs[i]; ++j)
            concs[i * max_num_nucs + j] = LCG_random_double(&seed);
    return concs;
}

SimulationData grid_init_do_not_profile(const Inputs &in) {
    SimulationData SD{};
    uint64_t seed = 42;
    SD.length_nuclide_grid = static_cast<int>(in.n_isotopes * in.n_gridpoints);
    SD.nuclide_grid = (NuclideGridPoint*)malloc(SD.length_nuclide_grid * sizeof(NuclideGridPoint));
    for (int i = 0; i < SD.length_nuclide_grid; ++i) {
        SD.nuclide_grid[i].energy        = LCG_random_double(&seed);
        SD.nuclide_grid[i].total_xs      = LCG_random_double(&seed);
        SD.nuclide_grid[i].elastic_xs    = LCG_random_double(&seed);
        SD.nuclide_grid[i].absorbtion_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].fission_xs    = LCG_random_double(&seed);
        SD.nuclide_grid[i].nu_fission_xs = LCG_random_double(&seed);
    }
    for (int i = 0; i < in.n_isotopes; ++i)
        qsort(&SD.nuclide_grid[i * in.n_gridpoints],
              in.n_gridpoints, sizeof(NuclideGridPoint), NGP_compare);
    if (in.grid_type == UNIONIZED) {
        SD.length_unionized_energy_array = static_cast<int>(in.n_isotopes * in.n_gridpoints);
        SD.unionized_energy_array = (double*)malloc(SD.length_unionized_energy_array * sizeof(double));
        for (int i = 0; i < SD.length_unionized_energy_array; ++i)
            SD.unionized_energy_array[i] = SD.nuclide_grid[i].energy;
        qsort(SD.unionized_energy_array, SD.length_unionized_energy_array,
              sizeof(double), double_compare);
        SD.length_index_grid = static_cast<long>(SD.length_unionized_energy_array * in.n_isotopes);
        SD.index_grid = (int*)malloc(SD.length_index_grid * sizeof(int));
        int *idx_low = (int*)calloc(in.n_isotopes, sizeof(int));
        double *energy_high = (double*)malloc(in.n_isotopes * sizeof(double));
        for (int i = 0; i < in.n_isotopes; ++i)
            energy_high[i] = SD.nuclide_grid[i * in.n_gridpoints + 1].energy;
        for (long e = 0; e < SD.length_unionized_energy_array; ++e) {
            double ue = SD.unionized_energy_array[e];
            for (long i = 0; i < in.n_isotopes; ++i) {
                if (ue < energy_high[i]) SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                else if (idx_low[i] == in.n_gridpoints - 2) SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                else {
                    ++idx_low[i];
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                    energy_high[i] = SD.nuclide_grid[i * in.n_gridpoints + idx_low[i] + 1].energy;
                }
            }
        }
        free(idx_low); free(energy_high);
    } else if (in.grid_type == HASH) {
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = static_cast<long>(in.hash_bins * in.n_isotopes);
        SD.index_grid = (int*)malloc(SD.length_index_grid * sizeof(int));
        double du = 1.0 / in.hash_bins;
        for (long e = 0; e < in.hash_bins; ++e) {
            double energy = e * du;
            for (long i = 0; i < in.n_isotopes; ++i) {
                SD.index_grid[e * in.n_isotopes + i] = static_cast<int>(
                    grid_search_nuclide(in.n_gridpoints, energy,
                                        SD.nuclide_grid + i * in.n_gridpoints, 0,
                                        in.n_gridpoints - 1));
            }
        }
    } else {
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = 0;
    }
    SD.num_nucs = load_num_nucs(in.n_isotopes);
    SD.length_num_nucs = 12;
    SD.mats = load_mats(SD.num_nucs, in.n_isotopes, &SD.max_num_nucs);
    SD.length_mats = SD.length_num_nucs * SD.max_num_nucs;
    SD.concs = load_concs(SD.num_nucs, SD.max_num_nucs);
    SD.length_concs = SD.length_mats;
    return SD;
}

SimulationData move_to_device(sycl::queue &q, const Inputs &in, const SimulationData &hostSD) {
    SimulationData devSD = hostSD;
    devSD.num_nucs = sycl::malloc_device<int>(hostSD.length_num_nucs, q);
    q.memcpy(devSD.num_nucs, hostSD.num_nucs,
             hostSD.length_num_nucs * sizeof(int)).wait();
    devSD.concs = sycl::malloc_device<double>(hostSD.length_concs, q);
    q.memcpy(devSD.concs, hostSD.concs,
             hostSD.length_concs * sizeof(double)).wait();
    devSD.mats = sycl::malloc_device<int>(hostSD.length_mats, q);
    q.memcpy(devSD.mats, hostSD.mats,
             hostSD.length_mats * sizeof(int)).wait();
    devSD.unionized_energy_array = nullptr;
    if (hostSD.length_unionized_energy_array) {
        devSD.unionized_energy_array = sycl::malloc_device<double>(hostSD.length_unionized_energy_array, q);
        q.memcpy(devSD.unionized_energy_array, hostSD.unionized_energy_array,
                 hostSD.length_unionized_energy_array * sizeof(double)).wait();
    }
    devSD.index_grid = nullptr;
    if (hostSD.length_index_grid) {
        devSD.index_grid = sycl::malloc_device<int>(hostSD.length_index_grid, q);
        q.memcpy(devSD.index_grid, hostSD.index_grid,
                 hostSD.length_index_grid * sizeof(int)).wait();
    }
    devSD.nuclide_grid = sycl::malloc_device<NuclideGridPoint>(hostSD.length_nuclide_grid, q);
    q.memcpy(devSD.nuclide_grid, hostSD.nuclide_grid,
             hostSD.length_nuclide_grid * sizeof(NuclideGridPoint)).wait();
    devSD.verification = sycl::malloc_device<unsigned long>(in.lookups, q);
    devSD.length_verification = in.lookups;
    devSD.p_energy_samples = nullptr;
    devSD.mat_samples = nullptr;
    return devSD;
}
void release_device_memory(sycl::queue &q, SimulationData &devSD) {
    sycl::free(devSD.num_nucs, q);
    sycl::free(devSD.concs, q);
    sycl::free(devSD.mats, q);
    sycl::free(devSD.unionized_energy_array, q);
    sycl::free(devSD.index_grid, q);
    sycl::free(devSD.nuclide_grid, q);
    sycl::free(devSD.verification, q);
    sycl::free(devSD.p_energy_samples, q);
    sycl::free(devSD.mat_samples, q);
}

// ------------------------------------------------------------
// Baseline kernel launch
// ------------------------------------------------------------
unsigned long long run_baseline(sycl::queue &q, const Inputs &in, SimulationData &devSD) {
    const int nthreads = 256;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;
    q.submit([&](sycl::handler &cgh) {
        cgh.parallel_for<class baseline_kernel>(sycl::nd_range<1>(sycl::range<1>(nblocks * nthreads),
                                                                sycl::range<1>(nthreads)),
            [=](sycl::nd_item<1> it) {
                int i = it.get_global_id(0);
                if (i >= in.lookups) return;
                uint64_t seed = STARTING_SEED;
                seed = fast_forward_LCG(seed, 2ULL * i);
                double p_energy = LCG_random_double(&seed);
                int mat = pick_mat(&seed);
                double macro_xs[5] = {0};
                calculate_macro_xs(p_energy, mat,
                                   in.n_isotopes, in.n_gridpoints,
                                   devSD.num_nucs, devSD.concs,
                                   devSD.unionized_energy_array,
                                   devSD.index_grid,
                                   devSD.nuclide_grid,
                                   devSD.mats,
                                   macro_xs,
                                   in.grid_type, in.hash_bins,
                                   devSD.max_num_nucs);
                double maxv = -1.0;
                int max_idx = 0;
                for (int j = 0; j < 5; ++j)
                    if (macro_xs[j] > maxv) { maxv = macro_xs[j]; max_idx = j; }
                devSD.verification[i] = static_cast<unsigned long>(max_idx + 1);
            });
    }).wait();
    std::vector<unsigned long> host_verif(in.lookups);
    q.memcpy(host_verif.data(), devSD.verification,
             in.lookups * sizeof(unsigned long)).wait();
    unsigned long long checksum = std::accumulate(host_verif.begin(),
                                                  host_verif.end(),
                                                  0ULL);
    return checksum;
}

// Split‑sampling variant (kernel_id == 1)
// ------------------------------------------------------------
unsigned long long run_split_sampling(sycl::queue &q, const Inputs &in, SimulationData &devSD) {
    devSD.p_energy_samples = sycl::malloc_device<double>(in.lookups, q);
    devSD.mat_samples = sycl::malloc_device<int>(in.lookups, q);
    const int nthreads = 32;
    const int nblocks = (in.lookups + nthreads - 1) / nthreads;
    q.submit([&](sycl::handler &cgh) {
        cgh.parallel_for<class sampling_kernel>(sycl::nd_range<1>(sycl::range<1>(nblocks * nthreads),
                                                                sycl::range<1>(nthreads)),
            [=](sycl::nd_item<1> it) {
                int i = it.get_global_id(0);
                if (i >= in.lookups) return;
                uint64_t seed = STARTING_SEED;
                seed = fast_forward_LCG(seed, 2ULL * i);
                double p_energy = LCG_random_double(&seed);
                int mat = pick_mat(&seed);
                devSD.p_energy_samples[i] = p_energy;
                devSD.mat_samples[i] = mat;
            });
    }).wait();
    q.submit([&](sycl::handler &cgh) {
        cgh.parallel_for<class lookup_split_kernel>(sycl::nd_range<1>(sycl::range<1>(nblocks * nthreads),
                                                                    sycl::range<1>(nthreads)),
            [=](sycl::nd_item<1> it) {
                int i = it.get_global_id(0);
                if (i >= in.lookups) return;
                double macro_xs[5] = {0};
                calculate_macro_xs(devSD.p_energy_samples[i],
                                   devSD.mat_samples[i],
                                   in.n_isotopes, in.n_gridpoints,
                                   devSD.num_nucs, devSD.concs,
                                   devSD.unionized_energy_array,
                                   devSD.index_grid,
                                   devSD.nuclide_grid,
                                   devSD.mats,
                                   macro_xs,
                                   in.grid_type, in.hash_bins,
                                   devSD.max_num_nucs);
                double maxv = -1.0;
                int max_idx = 0;
                for (int j = 0; j < 5; ++j)
                    if (macro_xs[j] > maxv) { maxv = macro_xs[j]; max_idx = j; }
                devSD.verification[i] = static_cast<unsigned long>(max_idx + 1);
            });
    }).wait();
    std::vector<unsigned long> host_verif(in.lookups);
    q.memcpy(host_verif.data(), devSD.verification,
             in.lookups * sizeof(unsigned long)).wait();
    unsigned long long checksum = std::accumulate(host_verif.begin(),
                                                  host_verif.end(),
                                                  0ULL);
    return checksum;
}

// ------------------------------------------------------------
// Main driver
// ------------------------------------------------------------
int main(int argc, char *argv[]) {
    Inputs in{};
    in.simulation_method = EVENT_BASED;
    in.nthreads = 1;
    in.n_isotopes = 355;
    in.n_gridpoints = 11303;
    in.particles = 500000;
    in.lookups = 34;
    in.grid_type = UNIONIZED;
    in.hash_bins = 10000;
    in.binary_mode = NONE;
    in.kernel_id = 0;
    in.HM = (char *)malloc(6);
    strcpy(in.HM, "large");

    SimulationData hostSD;
    if (in.binary_mode == READ) hostSD = binary_read(in);
    else hostSD = grid_init_do_not_profile(in);
    if (in.binary_mode == WRITE) binary_write(in, hostSD);

    // Use explicit CUDA selector to ensure CUDA backend with USM support
    sycl::queue q{sycl::cuda_selector{}, sycl::property::queue::in_order{}};

    SimulationData devSD = move_to_device(q, in, hostSD);

    double t0 = get_time();
    unsigned long long checksum = 0;
    if (in.kernel_id == 0) checksum = run_baseline(q, in, devSD);
    else if (in.kernel_id == 1) checksum = run_split_sampling(q, in, devSD);
    else { printf("Unsupported kernel_id %d\n", in.kernel_id); return 1; }
    double t1 = get_time();

    checksum %= 999983ULL;
    printf("Runtime: %.3f s\n", t1 - t0);
    printf("Checksum: %llu\n", checksum);

    release_device_memory(q, devSD);
    free(hostSD.num_nucs);
    free(hostSD.concs);
    free(hostSD.mats);
    free(hostSD.nuclide_grid);
    free(hostSD.index_grid);
    free(hostSD.unionized_energy_array);
    free(in.HM);
    return 0;
}
