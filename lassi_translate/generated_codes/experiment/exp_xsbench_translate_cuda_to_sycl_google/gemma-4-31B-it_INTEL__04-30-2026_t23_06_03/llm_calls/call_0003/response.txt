inline uint64_t fast_forward_LCG(uint64_t seed, uint64_t n) {
    const uint64_t m = 9223372036854775808ULL;
    uint64_t a = 2806196910506780709ULL;
    uint64_t c = 1ULL;
    n = n % m;
    uint64_t a_new = 1;
    uint64_t c_new = 0;
    while (n > 0) {
        if (n & 1) {
            a_new *= a;
            c_new = c_new * a + c;
        }
        c *= (a + 1);
        a *= a;
        n >>= 1;
    }
    return (a_new * seed + c_new) % m;
}

inline double LCG_random_double(uint64_t* seed) {
    const uint64_t m = 9223372036854775808ULL;
    const uint64_t a = 2806196910506780709ULL;
    const uint64_t c = 1ULL;
    *seed = (a * (*seed) + c) % m;
    return (double)(*seed) / (double)m;
}

inline long grid_search_nuclide(long n, double quarry, NuclideGridPoint* A, long low, long high) {
    long lowerLimit = low;
    long upperLimit = high;
    long length = upperLimit - lowerLimit;
    while (length > 1) {
        long examinationPoint = lowerLimit + (length / 2);
        if (A[examinationPoint].energy > quarry)
            upperLimit = examinationPoint;
        else
            lowerLimit = examinationPoint;
        length = upperLimit - lowerLimit;
    }
    return lowerLimit;
}

inline long grid_search(long n, double quarry, double* A) {
    long lowerLimit = 0;
    long upperLimit = n - 1;
    long length = upperLimit - lowerLimit;
    while (length > 1) {
        long examinationPoint = lowerLimit + (length / 2);
        if (A[examinationPoint] > quarry)
            upperLimit = examinationPoint;
        else
            lowerLimit = examinationPoint;
        length = upperLimit - lowerLimit;
    }
    return lowerLimit;
}

inline int pick_mat(uint64_t* seed) {
    double dist[12] = {0.140, 0.052, 0.275, 0.134, 0.154, 0.064, 0.066, 0.055, 0.008, 0.015, 0.025, 0.013};
    double roll = LCG_random_double(seed);
    for (int i = 0; i < 12; i++) {
        double running = 0;
        for (int j = i; j >= 0; j--) running += dist[j];
        if (roll < running) return i;
    }
    return 0;
}

inline void calculate_micro_xs(double p_energy, int nuc, long n_isotopes, long n_gridpoints,
                               double* egrid, int* index_data, NuclideGridPoint* nuclide_grids,
                               long idx, double* xs_vector, int grid_type, int hash_bins) {
    double f;
    NuclideGridPoint *low, *high;
    if (grid_type == NUCLIDE) {
        idx = grid_search_nuclide(n_gridpoints, p_energy, &nuclide_grids[nuc * n_gridpoints], 0, n_gridpoints - 1);
        low = (idx == n_gridpoints - 1) ? &nuclide_grids[nuc * n_gridpoints + idx - 1] : &nuclide_grids[nuc * n_gridpoints + idx];
    } else if (grid_type == UNIONIZED) {
        int grid_idx = index_data[idx * n_isotopes + nuc];
        low = (grid_idx == n_gridpoints - 1) ? &nuclide_grids[nuc * n_gridpoints + grid_idx - 1] : &nuclide_grids[nuc * n_gridpoints + grid_idx];
    } else {
        int u_low = index_data[idx * n_isotopes + nuc];
        int u_high = (idx == hash_bins - 1) ? n_gridpoints - 1 : index_data[(idx + 1) * n_isotopes + nuc] + 1;
        double e_low = nuclide_grids[nuc * n_gridpoints + u_low].energy;
        double e_high = nuclide_grids[nuc * n_gridpoints + u_high].energy;
        int lower;
        if (p_energy <= e_low) lower = 0;
        else if (p_energy >= e_high) lower = n_gridpoints - 1;
        else lower = grid_search_nuclide(n_gridpoints, p_energy, &nuclide_grids[nuc * n_gridpoints], u_low, u_high);
        low = (lower == n_gridpoints - 1) ? &nuclide_grids[nuc * n_gridpoints + lower - 1] : &nuclide_grids[nuc * n_gridpoints + lower];
    }
    high = low + 1;
    f = (high->energy - p_energy) / (high->energy - low->energy);
    xs_vector[0] = high->total_xs - f * (high->total_xs - low->total_xs);
    xs_vector[1] = high->elastic_xs - f * (high->elastic_xs - low->elastic_xs);
    xs_vector[2] = high->absorbtion_xs - f * (high->absorbtion_xs - low->absorbtion_xs);
    xs_vector[3] = high->fission_xs - f * (high->fission_xs - low->fission_xs);
    xs_vector[4] = high->nu_fission_xs - f * (high->nu_fission_xs - low->nu_fission_xs);
}

inline void calculate_macro_xs(double p_energy, int mat, long n_isotopes, long n_gridpoints, int* num_nucs,
                               double* concs, double* egrid, int* index_data, NuclideGridPoint* nuclide_grids,
                               int* mats, double* macro_xs_vector, int grid_type, int hash_bins, int max_num_nucs) {
    long idx = -1;
    for (int k = 0; k < 5; k++) macro_xs_vector[k] = 0;
    if (grid_type == UNIONIZED) idx = grid_search(n_isotopes * n_gridpoints, p_energy, egrid);
    else if (grid_type == HASH) idx = (long)(p_energy / (1.0 / hash_bins));

    for (int j = 0; j < num_nucs[mat]; j++) {
        double xs_vector[5];
        int p_nuc = mats[mat * max_num_nucs + j];
        double conc = concs[mat * max_num_nucs + j];
        calculate_micro_xs(p_energy, p_nuc, n_isotopes, n_gridpoints, egrid, index_data, nuclide_grids, idx, xs_vector, grid_type, hash_bins);
        for (int k = 0; k < 5; k++) macro_xs_vector[k] += xs_vector[k] * conc;
    }
}

unsigned long long run_event_based_simulation_baseline(Inputs in, SimulationData GSD, int mype, sycl::queue& q) {
    if (mype == 0) printf("Running baseline event-based simulation...\n");
    int nthreads = 256;
    int nblocks = (int)ceil((double)in.lookups / nthreads);

    q.submit([&](sycl::handler& h) {
        h.parallel_for(sycl::range<1>(in.lookups), [=](sycl::id<1> idx) {
            int i = idx[0];
            uint64_t seed = STARTING_SEED;
            seed = fast_forward_LCG(seed, 2 * i);
            double p_energy = LCG_random_double(&seed);
            int mat = pick_mat(&seed);
            double macro_xs_vector[5];
            calculate_macro_xs(p_energy, mat, in.n_isotopes, in.n_gridpoints, GSD.num_nucs, GSD.concs, GSD.unionized_energy_array, GSD.index_grid, GSD.nuclide_grid, GSD.mats, macro_xs_vector, in.grid_type, in.hash_bins, GSD.max_num_nucs);
            double max_val = -1.0;
            int max_idx = 0;
            for (int j = 0; j < 5; j++) {
                if (macro_xs_vector[j] > max_val) {
                    max_val = macro_xs_vector[j];
                    max_idx = j;
                }
            }
            GSD.verification[i] = max_idx + 1;
        });
    }).wait();

    unsigned long long sum = 0;
    std::vector<unsigned long> host_ver(in.lookups);
    q.memcpy(host_ver.data(), GSD.verification, in.lookups * sizeof(unsigned long)).wait();
    for (auto v : host_ver) sum += v;
    return sum;
}

void sampling_kernel_impl(Inputs in, SimulationData GSD, sycl::queue& q) {
    q.submit([&](sycl::handler& h) {
        h.parallel_for(sycl::range<1>(in.lookups), [=](sycl::id<1> idx) {
            int i = idx[0];
            uint64_t seed = STARTING_SEED;
            seed = fast_forward_LCG(seed, 2 * i);
            GSD.p_energy_samples[i] = LCG_random_double(&seed);
            GSD.mat_samples[i] = pick_mat(&seed);
        });
    }).wait();
}

unsigned long long run_event_based_simulation_optimization_4(Inputs in, SimulationData GSD, int mype, sycl::queue& q) {
    if (mype == 0) printf("Simulation Kernel:\"Optimization 4 - All Material Lookup Kernels + Material Sort\"\n");
    GSD.p_energy_samples = sycl::malloc_device<double>(in.lookups, q);
    GSD.mat_samples = sycl::malloc_device<int>(in.lookups, q);
    sampling_kernel_impl(in, GSD, q);
    std::vector<int> mat_h(in.lookups), energy_h(in.lookups); // Simplified sort for translation
    std::vector<double> energy_d_h(in.lookups);
    q.memcpy(mat_h.data(), GSD.mat_samples, in.lookups * sizeof(int)).wait();
    q.memcpy(energy_d_h.data(), GSD.p_energy_samples, in.lookups * sizeof(double)).wait();
    
    std::vector<size_t> p(in.lookups);
    for(size_t i=0; i<p.size(); ++i) p[i] = i;
    std::sort(p.begin(), p.end(), [&](size_t i, size_t j){
        if(mat_h[i] != mat_h[j]) return mat_h[i] < mat_h[j];
        return energy_d_h[i] < energy_d_h[j];
    });

    std::vector<int> sorted_mat(in.lookups);
    std::vector<double> sorted_energy(in.lookups);
    for(size_t i=0; i<p.size(); ++i) {
        sorted_mat[i] = mat_h[p[i]];
        sorted_energy[i] = energy_d_h[p[i]];
    }
    q.memcpy(GSD.mat_samples, sorted_mat.data(), in.lookups * sizeof(int)).wait();
    q.memcpy(GSD.p_energy_samples, sorted_energy.data(), in.lookups * sizeof(double)).wait();

    int n_lookups_per_material[12] = {0};
    for(int m : sorted_mat) n_lookups_per_material[m]++;

    int offset = 0;
    for (int m = 0; m < 12; m++) {
        int n_l = n_lookups_per_material[m];
        q.submit([&](sycl::handler& h) {
            h.parallel_for(sycl::range<1>(n_l), [=](sycl::id<1> idx) {
                int i = idx[0] + offset;
                double macro_xs_vector[5];
                calculate_macro_xs(GSD.p_energy_samples[i], m, in.n_isotopes, in.n_gridpoints, GSD.num_nucs, GSD.concs, GSD.unionized_energy_array, GSD.index_grid, GSD.nuclide_grid, GSD.mats, macro_xs_vector, in.grid_type, in.hash_bins, GSD.max_num_nucs);
                double max_val = -1.0;
                int max_idx = 0;
                for (int j = 0; j < 5; j++) {
                    if (macro_xs_vector[j] > max_val) {
                        max_val = macro_xs_vector[j];
                        max_idx = j;
                    }
                }
                GSD.verification[i] = max_idx + 1;
            });
        }).wait();
        offset += n_l;
    }
    unsigned long long sum = 0;
    std::vector<unsigned long> host_ver(in.lookups);
    q.memcpy(host_ver.data(), GSD.verification, in.lookups * sizeof(unsigned long)).wait();
    for (auto v : host_ver) sum += v;
    sycl::free(GSD.p_energy_samples, q);
    sycl::free(GSD.mat_samples, q);
    return sum;
}

    q.memcpy(GSD.p_energy_samples, sorted_energy.data(), in.lookups * sizeof(double)).wait();

    int n_lookups_per_material[12] = {0};
    for(int m : sorted_mat) n_lookups_per_material[m]++;

    int offset = 0;
    for (int m = 0; m < 12; m++) {
        int n_l = n_lookups_per_material[m];
        q.submit([&](sycl::handler& h) {
            h.parallel_for(sycl::range<1>(n_l), [=](sycl::id<1> idx) {
                int i = idx[0] + offset;
                double macro_xs_vector[5];
                calculate_macro_xs(GSD.p_energy_samples[i], m, in.n_isotopes, in.n_gridpoints, GSD.num_nucs, GSD.concs, GSD.unionized_energy_array, GSD.index_grid, GSD.nuclide_grid, GSD.mats, macro_xs_vector, in.grid_type, in.hash_bins, GSD.max_num_nucs);
                double max_val = -1.0;
                int max_idx = 0;
                for (int j = 0; j < 5; j++) {
                    if (macro_xs_vector[j] > max_val) {
                        max_val = macro_xs_vector[j];
                        max_idx = j;
                    }
                }
                GSD.verification[i] = max_idx + 1;
            });
        }).wait();
        offset += n_l;
    }
    unsigned long long sum = 0;
    std::vector<unsigned long> host_ver(in.lookups);
    q.memcpy(host_ver.data(), GSD.verification, in.lookups * sizeof(unsigned long)).wait();
    for (auto v : host_ver) sum += v;
    sycl::free(GSD.p_energy_samples, q);
    sycl::free(GSD.mat_samples, q);
    return sum;
}

// Memory and Grid Init
SimulationData move_simulation_data_to_device(Inputs in, int mype, SimulationData SD, sycl::queue& q) {
    if (mype == 0) printf("Allocating and moving simulation data to GPU memory space...\n");
    SimulationData GSD = SD;
    size_t total_sz = 0;
    auto alloc_copy = [&](void** ptr, const void* src, size_t sz) {
        *ptr = sycl::malloc_device(sz, q);
        q.memcpy(*ptr, src, sz).wait();
        total_sz += sz;
    };
    alloc_copy((void**)&GSD.num_nucs, SD.num_nucs, SD.length_num_nucs * sizeof(int));
    alloc_copy((void**)&GSD.concs, SD.concs, SD.length_concs * sizeof(double));
    alloc_copy((void**)&GSD.mats, SD.mats, SD.length_mats * sizeof(int));
    alloc_copy((void**)&GSD.unionized_energy_array, SD.unionized_energy_array, SD.length_unionized_energy_array * sizeof(double));
    alloc_copy((void**)&GSD.index_grid, SD.index_grid, SD.length_index_grid * sizeof(int));
    alloc_copy((void**)&GSD.nuclide_grid, SD.nuclide_grid, SD.length_nuclide_grid * sizeof(NuclideGridPoint));
    GSD.verification = sycl::malloc_device<unsigned long>(in.lookups, q);
    GSD.length_verification = in.lookups;
    total_sz += in.lookups * sizeof(unsigned long);
    if (mype == 0) printf("GPU Intialization complete. Allocated %.0lf MB of data on GPU.\n", total_sz / 1024.0 / 1024.0);
    return GSD;
}

void release_device_memory(SimulationData GSD, sycl::queue& q) {
    sycl::free(GSD.num_nucs, q);
    sycl::free(GSD.concs, q);
    sycl::free(GSD.mats, q);
    sycl::free(GSD.unionized_energy_array, q);
    sycl::free(GSD.index_grid, q);
    sycl::free(GSD.nuclide_grid, q);
    sycl::free(GSD.verification, q);
}

SimulationData grid_init_do_not_profile(Inputs in, int mype) {
    SimulationData SD;
    size_t nbytes = 0;
    uint64_t seed = 42;
    if (mype == 0) printf("Intializing nuclide grids...\n");
    SD.length_nuclide_grid = in.n_isotopes * in.n_gridpoints;
    SD.nuclide_grid = (NuclideGridPoint*)malloc(SD.length_nuclide_grid * sizeof(NuclideGridPoint));
    nbytes += SD.length_nuclide_grid * sizeof(NuclideGridPoint);
    for (int i = 0; i < SD.length_nuclide_grid; i++) {
        SD.nuclide_grid[i].energy = LCG_random_double(&seed);
        SD.nuclide_grid[i].total_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].elastic_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].absorbtion_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].fission_xs = LCG_random_double(&seed);
        SD.nuclide_grid[i].nu_fission_xs = LCG_random_double(&seed);
    }
    for (int i = 0; i < in.n_isotopes; i++)
        qsort(&SD.nuclide_grid[i * in.n_gridpoints], in.n_gridpoints, sizeof(NuclideGridPoint), NGP_compare);

    if (in.grid_type == NUCLIDE) {
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = 0;
    } else if (in.grid_type == UNIONIZED) {
        if (mype == 0) printf("Intializing unionized grid...\n");
        SD.length_unionized_energy_array = in.n_isotopes * in.n_gridpoints;
        SD.unionized_energy_array = (double*)malloc(SD.length_unionized_energy_array * sizeof(double));
        nbytes += SD.length_unionized_energy_array * sizeof(double);
        for (int i = 0; i < SD.length_unionized_energy_array; i++) SD.unionized_energy_array[i] = SD.nuclide_grid[i].energy;
        qsort(SD.unionized_energy_array, SD.length_unionized_energy_array, sizeof(double), double_compare);
        SD.length_index_grid = (long)SD.length_unionized_energy_array * in.n_isotopes;
        SD.index_grid = (int*)malloc(SD.length_index_grid * sizeof(int));
        nbytes += SD.length_index_grid * sizeof(int);
        int* idx_low = (int*)calloc(in.n_isotopes, sizeof(int));
        double* energy_high = (double*)malloc(in.n_isotopes * sizeof(double));
        for (int i = 0; i < in.n_isotopes; i++) energy_high[i] = SD.nuclide_grid[i * in.n_gridpoints + 1].energy;
        for (long e = 0; e < SD.length_unionized_energy_array; e++) {
            double unionized_energy = SD.unionized_energy_array[e];
            for (long i = 0; i < in.n_isotopes; i++) {
                if (unionized_energy < energy_high[i]) SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                else if (idx_low[i] == in.n_gridpoints - 2) SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                else {
                    idx_low[i]++;
                    SD.index_grid[e * in.n_isotopes + i] = idx_low[i];
                    energy_high[i] = SD.nuclide_grid[i * in.n_gridpoints + idx_low[i] + 1].energy;
                }
            }
        }
        free(idx_low); free(energy_high);
    } else if (in.grid_type == HASH) {
        if (mype == 0) printf("Intializing hash grid...\n");
        SD.length_unionized_energy_array = 0;
        SD.length_index_grid = (long)in.hash_bins * in.n_isotopes;
        SD.index_grid = (int*)malloc(SD.length_index_grid * sizeof(int));
        nbytes += SD.length_index_grid * sizeof(int);
        double du = 1.0 / in.hash_bins;
        for (long e = 0; e < in.hash_bins; e++) {
            double energy = e * du;
            for (long i = 0; i < in.n_isotopes; i++)
                SD.index_grid[e * in.n_isotopes + i] = grid_search_nuclide(in.n_gridpoints, energy, SD.nuclide_grid + i * in.n_gridpoints, 0, in.n_gridpoints - 1);
        }
    }
    if (mype == 0) printf("Intializing material data...\n");
    SD.num_nucs = load_num_nucs(in.n_isotopes);
    SD.length_num_nucs = 12;
    SD.mats = load_mats(SD.num_nucs, in.n_isotopes, &SD.max_num_nucs);
    SD.length_mats = SD.length_num_nucs * SD.max_num_nucs;
    SD.concs = load_concs(SD.num_nucs, SD.max_num_nucs);
    SD.length_concs = SD.length_mats;
    if (mype == 0) printf("Intialization complete. Allocated %.0lf MB of data on CPU.\n", nbytes / 1024.0 / 1024.0);
    return SD;
}

Inputs read_CLI(int argc, char* argv[]) {
    Inputs input;
    input.simulation_method = HISTORY_BASED;
    input.nthreads = 1;
    input.n_isotopes = 355;
    input.n_gridpoints = 11303;
    input.particles = 500000;
    input.lookups = 34;
    input.grid_type = UNIONIZED;
    input.hash_bins = 10000;
    input.binary_mode = NONE;
    input.kernel_id = 0;
    input.HM = strdup("large");
    int user_g = 0;
    int default_lookups = 1, default_particles = 1;
    for (int i = 1; i < argc; i++) {
        char* arg = argv[i];
        if (strcmp(arg, "-g") == 0 && ++i < argc) { user_g = 1; input.n_gridpoints = atol(argv[i]); }
        else if (strcmp(arg, "-m") == 0 && ++i < argc) {
            if (strcmp(argv[i], "history") == 0) input.simulation_method = HISTORY_BASED;
            else if (strcmp(argv[i], "event") == 0) {
                input.simulation_method = EVENT_BASED;
                if (default_lookups && default_particles) { input.lookups *= input.particles; input.particles = 0; }
            } else print_CLI_error();
        } else if (strcmp(arg, "-l") == 0 && ++i < argc) { input.lookups = atoi(argv[i]); default_lookups = 0; }
        else if (strcmp(arg, "-h") == 0 && ++i < argc) input.hash_bins = atoi(argv[i]);
        else if (strcmp(arg, "-p") == 0 && ++i < argc) { input.particles = atoi(argv[i]); default_particles = 0; }
        else if (strcmp(arg, "-s") == 0 && ++i < argc) input.HM = argv[i];
        else if (strcmp(arg, "-G") == 0 && ++i < argc) {
            if (strcmp(argv[i], "unionized") == 0) input.grid_type = UNIONIZED;
            else if (strcmp(argv[i], "nuclide") == 0) input.grid_type = NUCLIDE;
            else if (strcmp(argv[i], "hash") == 0) input.grid_type = HASH;
            else print_CLI_error();
        } else if (strcmp(arg, "-b") == 0 && ++i < argc) {
            if (strcmp(argv[i], "read") == 0) input.binary_mode = READ;
            else if (strcmp(argv[i], "write") == 0) input.binary_mode = WRITE;
            else print_CLI_error();
        } else if (strcmp(arg, "-k") == 0 && ++i < argc) input.kernel_id = atoi(argv[i]);
        else print_CLI_error();
    }
    if (strcasecmp(input.HM, "small") == 0) input.n_isotopes = 68;
    else if (strcasecmp(input.HM, "XL") == 0 && user_g == 0) input.n_gridpoints = 238847;
    else if (strcasecmp(input.HM, "XXL") == 0 && user_g == 0) input.n_gridpoints = (int)(238847 * 2.1);
    return input;
}

int double_compare(const void* a, const void* b) {
    double A = *(double*)a, B = *(double*)b;
    return (A > B) - (A < B);
}

int NGP_compare(const void* a, const void* b) {
    NuclideGridPoint A = *(NuclideGridPoint*)a, B = *(NuclideGridPoint*)b;
    return (A.energy > B.energy) - (A.energy < B.energy);
}

double rn_v(void) {
    static unsigned long seed = 1337;
    unsigned long a = 16807, m = 2147483647;
    seed = (a * seed) % m;
    return (double)seed / m;
}

size_t estimate_mem_usage(Inputs in) {
    size_t all_nuclide_grids = (size_t)in.n_isotopes * in.n_gridpoints * sizeof(NuclideGridPoint);
    size_t size_UEG = (size_t)in.n_isotopes * in.n_gridpoints * sizeof(double) + (size_t)in.n_isotopes * in.n_gridpoints * in.n_isotopes * sizeof(int);
    size_t size_hash_grid = (size_t)in.hash_bins * in.n_isotopes * sizeof(int);
    size_t memtotal = (in.grid_type == UNIONIZED) ? all_nuclide_grids + size_UEG : (
