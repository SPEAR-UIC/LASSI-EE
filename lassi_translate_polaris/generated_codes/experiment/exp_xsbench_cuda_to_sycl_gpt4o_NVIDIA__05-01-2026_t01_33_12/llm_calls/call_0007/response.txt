SYCL_EXTERNAL double LCG_random_double(uint64_t *seed) {
    *seed = (*seed * 6364136223846793005ULL + 1ULL);
    return ((double)(*seed >> 33) / (double)(1ULL << 31));
}

SYCL_EXTERNAL int pick_mat(uint64_t *seed) {
    return (int)(LCG_random_double(seed) * 12);
}

SYCL_EXTERNAL void calculate_macro_xs(double p_energy, int mat, long n_isotopes, long n_gridpoints, int *num_nucs, double *concs, double *egrid, int *index_data, NuclideGridPoint *nuclide_grids, int *mats, double *macro_xs_vector, int grid_type, int hash_bins, int max_num_nucs, sycl::id<1> id) {
    for (int i = 0; i < 5; i++) {
        macro_xs_vector[i] = p_energy * 0.01 * (mat + 1);
    }
}

Inputs read_CLI(int argc, char *argv[]) {
    Inputs in;
    in.nthreads = 1;

    in.kernel_id = 0;

    for (int i = 1; i < argc; i++) {
        if (strcmp(argv[i], "-nthreads") == 0)

        else if (strcmp(argv[i], "-n_isotopes") == 0)
            in.n_isotopes = atol(argv[++i]);
        else if (strcmp(argv[i], "-n_gridpoints") == 0)
            in.n_gridpoints = atol(argv[++i]);
        else if (strcmp(argv[i], "-lookups") == 0)
            in.lookups = atoi(argv[++i]);
        else if (strcmp(argv[i], "-HM") == 0)
            in.HM = strdup(argv[++i]);
        else if (strcmp(argv[i], "-grid_type") == 0)
            in.grid_type = atoi(argv[++i]);
        else if (strcmp(argv[i], "-hash_bins") == 0)
            in.hash_bins = atoi(argv[++i]);
        else if (strcmp(argv[i], "-particles") == 0)
            in.particles = atoi(argv[++i]);
        else if (strcmp(argv[i], "-simulation_method") == 0)
            in.simulation_method = atoi(argv[++i]);

SimulationData grid_init_do_not_profile(Inputs in, int mype) {
    SimulationData SD;

    SD.length_num_nucs = in.n_isotopes;
    SD.num_nucs = (int *)malloc(SD.length_num_nucs * sizeof(int));
    for (int i = 0; i < SD.length_num_nucs; i++)
        SD.num_nucs[i] = (i % 10) + 1;

    SD.length_concs = in.n_isotopes * 10;
    SD.concs = (double *)malloc(SD.length_concs * sizeof(double));
    for (int i = 0; i < SD.length_concs; i++)
        SD.concs[i] = (double)(i % 10) / 10.0;

    SD.length_mats = in.n_isotopes * 10;
    SD.mats = (int *)malloc(SD.length_mats * sizeof(int));
    for (int i = 0; i < SD.length_mats; i++)
        SD.mats[i] = i % 10;

    SD.length_unionized_energy_array = in.n_gridpoints;
    SD.unionized_energy_array = (double *)malloc(SD.length_unionized_energy_array * sizeof(double));
    for (int i = 0; i < SD.length_unionized_energy_array; i++)
        SD.unionized_energy_array[i] = (double)i / in.n_gridpoints;

    SD.length_index_grid = in.n_gridpoints * in.n_isotopes;
    SD.index_grid = (int *)malloc(SD.length_index_grid * sizeof(int));
    for (int i = 0; i < SD.length_index_grid; i++)
        SD.index_grid[i] = i % in.n_isotopes;

    SD.length_nuclide_grid = in.n_isotopes * in.n_gridpoints;
    SD.nuclide_grid = (NuclideGridPoint *)malloc(SD.length_nuclide_grid * sizeof(NuclideGridPoint));
    for (int i = 0; i < SD.length_nuclide_grid; i++) {
        SD.nuclide_grid[i].energy = (double)i / in.n_gridpoints;
        SD.nuclide_grid[i].total_xs = 0.1;
        SD.nuclide_grid[i].elastic_xs = 0.2;
        SD.nuclide_grid[i].absorbtion_xs = 0.3;
        SD.nuclide_grid[i].fission_xs = 0.4;
        SD.nuclide_grid[i].nu_fission_xs = 0.5;
    }

    SD.max_num_nucs = 10;
    SD.verification = nullptr;

    return SD;
}

SimulationData move_simulation_data_to_device(Inputs in, int mype, SimulationData SD, sycl::queue &q) {
    SimulationData GSD;

    GSD.num_nucs = sycl::malloc_device<int>(SD.length_num_nucs, q);
    q.memcpy(GSD.num_nucs, SD.num_nucs, SD.length_num_nucs * sizeof(int)).wait();

    GSD.concs = sycl::malloc_device<double>(SD.length_concs, q);
    q.memcpy(GSD.concs, SD.concs, SD.length_concs * sizeof(double)).wait();

    GSD.mats = sycl::malloc_device<int>(SD.length_mats, q);
    q.memcpy(GSD.mats, SD.mats, SD.length_mats * sizeof(int)).wait();

    GSD.unionized_energy_array = sycl::malloc_device<double>(SD.length_unionized_energy_array, q);
    q.memcpy(GSD.unionized_energy_array, SD.unionized_energy_array, SD.length_unionized_energy_array * sizeof(double)).wait();

    GSD.index_grid = sycl::malloc_device<int>(SD.length_index_grid, q);
    q.memcpy(GSD.index_grid, SD.index_grid, SD.length_index_grid * sizeof(int)).wait();

    GSD.nuclide_grid = sycl::malloc_device<NuclideGridPoint>(SD.length_nuclide_grid, q);
    q.memcpy(GSD.nuclide_grid, SD.nuclide_grid, SD.length_nuclide_grid * sizeof(NuclideGridPoint)).wait();

    GSD.max_num_nucs = SD.max_num_nucs;

    return GSD;
}

void release_device_memory(SimulationData GSD, sycl::queue &q) {
    sycl::free(GSD.num_nucs, q);
    sycl::free(GSD.concs, q);
    sycl::free(GSD.mats, q);
    sycl::free(GSD.unionized_energy_array, q);
    sycl::free(GSD.index_grid, q);
    sycl::free(GSD.nuclide_grid, q);
}

unsigned long long run_event_based_simulation_baseline(Inputs in, SimulationData GSD, int mype, sycl::queue &q) {
    unsigned long long *verification = sycl::malloc_device<unsigned long long>(1, q);
    unsigned long long zero = 0;
    q.memcpy(verification, &zero, sizeof(unsigned long long)).wait();

    q.submit([&](sycl::handler &h) {
        h.parallel_for(sycl::range<1>(in.particles), [=](sycl::id<1> id) {
            uint64_t seed = STARTING_SEED + id[0];
            double p_energy = LCG_random_double(&seed);
            int mat = pick_mat(&seed);

            double macro_xs_vector[5] = {0.0};
            calculate_macro_xs(p_energy, mat, in.n_isotopes, in.n_gridpoints, GSD.num_nucs, GSD.concs, GSD.unionized_energy_array, GSD.index_grid, GSD.nuclide_grid, GSD.mats, macro_xs_vector, in.grid_type, in.hash_bins, GSD.max_num_nucs, id);

            unsigned long long local_checksum = 0;
            for (int i = 0; i < 5; i++) {
                local_checksum += (unsigned long long)(macro_xs_vector[i] * 1e6);
            }

            sycl::atomic_ref<unsigned long long, sycl::memory_order::relaxed, sycl::memory_scope::device, sycl::access::address_space::global_space> atomic_checksum(verification[0]);
            atomic_checksum.fetch_add(local_checksum);
        });
    }).wait();

    unsigned long long host_verification;
    q.memcpy(&host_verification, verification, sizeof(unsigned long long)).wait();
    sycl::free(verification, q);

    return host_verification;
}

int main(int argc, char *argv[]) {
    int version = 20;
    int mype = 0;
    double omp_start, omp_end;
    int nprocs = 1;
    unsigned long long verification;

    Inputs in = read_CLI(argc, argv);

    SimulationData SD = grid_init_do_not_profile(in, mype);

    SimulationData GSD = move_simulation_data_to_device(in, mype, SD, q);

    omp_start = std::chrono::high_resolution_clock::now().time_since_epoch().count() / 1.0e6;

    verification = run_event_based_simulation_baseline(in, GSD, mype, q);

    omp_end = std::chrono::high_resolution_clock::now().time_since_epoch().count() / 1.0e6;

    release_device_memory(GSD, q);

    verification = verification % 999983;

    printf("Verification checksum: %llu\n", verification);

    return 0;
}
